"""Offline finalization of a user-stopped warm study; never launch native Codex."""
from pathlib import Path
import json
from benchmarks import run_warm as w

output=Path('/workspace/Memglide/.local/warm-v2')
with w.durable.experiment_lock(output):
 metadata=json.loads((output/'metadata.json').read_text())
 state=json.loads((output/'state.json').read_text())
 plan=json.loads((output/'plan.json').read_text())
 assert not metadata.get('finished_at')
 w.durable.validate_frozen(metadata,w.frozen_manifest(plan['tasks'],metadata['frozen_manifest']['settings']))
 w.durable.verify_repositories(metadata['repositories'])
 w.validate_saved_state(output,plan['runs'],state)
 sealed={p:w.common.digest(p.read_bytes()) for p in (output/'runs').glob('*/result.json')}
 completed=[r for r in state['originals'].values() if r['status']=='completed']
 assert len(completed)==80 and max(r['order'] for r in completed)==80
 w.common.recover_interrupted(output,state,metadata)
 reason='User requested stop after 80 completed assignments; one already-started 81st original interrupted and retained'
 for identifier,row in list(state['originals'].items()):
  if row.get('pending'):
   state['originals'][identifier]=w.legacy.not_run(row,reason)
 metadata['user_requested_stop']={'completed_assignment_limit':80,'reason':reason,'request_text':'stop after 80 and give results','offline_recovery_only':True,'no_original_reissued':True}
 metadata['finished_at']=w.legacy.utc_now()
 metadata['memglide_source_after']=w.capture_provenance(w.ROOT)
 metadata['source_unchanged_during_experiment']=metadata['memglide_source_before']['source_tree_sha256']==metadata['memglide_source_after']['source_tree_sha256'] and not metadata.get('source_guard_failure')
 assert metadata['source_unchanged_during_experiment']
 w.validate_saved_state(output,plan['runs'],state)
 for p,expected in sealed.items():assert w.common.digest(p.read_bytes())==expected
 w.durable.append_log(output,'experiment.stopped_by_user',completed_assignments=80,interrupted_originals=1,unrun_assignments=135)
 w.durable.checkpoint(output,state,metadata)
 report=w.build_report(list(state['originals'].values()),plan['tasks'],metadata)
 w.write_report(report,output)
 w.durable.append_log(output,'experiment.finished',report_sha256=w.common.digest((output/'report.json').read_bytes()),user_requested_stop=True)
 print('Finalized without model requests:',{s:sum(r['status']==s for r in state['originals'].values()) for s in ('completed','interrupted','not_run')})
