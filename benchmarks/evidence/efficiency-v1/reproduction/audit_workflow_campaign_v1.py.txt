"""Independent financial accounting for both retained three-arm workflow studies.

No model requests, application imports, repository execution, raw native/grade
reads, credentials, databases or native-state reads. This relies on and binds
both independently audited finished studies. Different corpora are never pooled
for efficacy, latency, retrieval quality or a causal savings claim.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import math
import sys
from datetime import datetime, timezone
from decimal import Decimal
from pathlib import Path

ROOT=Path('/workspace/Memglide')
KEYS=('native_reported_price_equivalent_usd','possible_write_price_lower_usd','possible_write_price_upper_usd')
ARMS=('A','B','C')


def sha(data):
    return hashlib.sha256(data).hexdigest()


def unique(pairs):
    result={}
    for key,value in pairs:
        if key in result: raise ValueError('Duplicate JSON key')
        result[key]=value
    return result


def read(path):
    path=Path(path)
    if not path.is_absolute(): path=ROOT/path
    if not path.resolve().is_relative_to(ROOT): raise ValueError('Input escapes project')
    for component in (path,*path.parents):
        if component==ROOT: break
        if component.is_symlink() or component.name in {'codex-state','auth.json','memglide.db'}: raise ValueError('Private state/symlinks excluded')
    if not path.is_file() or path.stat().st_size>128*1024*1024: raise ValueError('Bounded regular JSON required')
    return path.read_bytes()


def document(path):
    return json.loads(read(path),object_pairs_hook=unique)


def amount(value):
    if value is None: return None
    if isinstance(value,bool): raise ValueError('Boolean is not a cost')
    result=Decimal(str(value))
    if not result.is_finite() or result<0: raise ValueError('Negative/nonfinite cost')
    return result


def aggregate(rows):
    results={}
    for arm in ARMS:
        selected=[r for r in rows if r['configuration']==arm]
        successes=sum(r['success'] for r in selected)
        totals={}
        for key in KEYS:
            values=[r['cost'][key] for r in selected]
            totals[key]=sum(values,Decimal(0)) if selected and all(v is not None for v in values) else None
        results[arm]={'attempts':len(selected),'successful_assignments':successes,'failed_assignments':len(selected)-successes,'unknown_cost_assignments':sum(r['cost'][KEYS[0]] is None for r in selected),'cost_total_usd':totals,'cost_per_successful_task_usd':{k:v/successes if v is not None and successes else None for k,v in totals.items()}}
    return results


class Audit:
    def __init__(self,arguments):
        self.args=arguments; self.errors=[]; self.checks=0

    def check(self,value,code):
        self.checks+=1
        if not value: self.errors.append({'code':code})

    def number(self,actual,expected,code):
        if expected is None:
            self.check(actual is None,code)
        elif type(actual) not in (int,float) or not math.isfinite(actual):
            self.check(False,code)
        else:
            self.check(abs(Decimal(str(actual))-Decimal(str(expected)))<=max(Decimal('1e-10'),abs(Decimal(str(expected)))*Decimal('1e-10')),code)

    def run(self):
        # Process exits must be durably observed before opening either receipt.
        for name in ('formal-exit.json','confirmation-exit.json'):
            receipt=document(ROOT/'.local/efficiency-v1'/name)
            if not (type(receipt.get('returncode')) is int and type(receipt.get('finished_unix')) in (int,float) and type(receipt.get('started_unix')) in (int,float) and receipt['finished_unix']>=receipt['started_unix']):
                raise RuntimeError('Finished driver exit receipt missing or invalid')
            self.check(receipt['returncode']==0,'driver_returned_zero:'+name)
        campaign_bytes=read(self.args.assessment); campaign=json.loads(campaign_bytes,object_pairs_hook=unique)
        self.check(campaign.get('assessment_version')=='efficiency-campaign-1','campaign_assessment_version')
        self.check(campaign.get('actual_billed_cost_usd') is None,'actual_billing_remains_null')
        rows=[]; studies={}
        for name in ('formal','confirmation'):
            receipt_path=getattr(self.args,name+'_audit'); receipt_bytes=read(receipt_path); receipt=json.loads(receipt_bytes,object_pairs_hook=unique)
            report_path=getattr(self.args,name+'_report'); report_bytes=read(report_path)
            price_path=getattr(self.args,name+'_assessment'); price_bytes=read(price_path)
            auditor_path=ROOT/'.local/efficiency-v1'/('audit_workflow_v1.py' if name=='formal' else 'audit_workflow_v2.py')
            self.check(receipt.get('pass') is True and receipt.get('errors')==[] and receipt.get('planned_originals')==60 and receipt.get('verified_triplets')==20,'passing_independent_finished_study:'+name)
            self.check(receipt.get('source_report_sha256')==sha(report_bytes),'exact_primary_report_binding:'+name)
            self.check(receipt.get('assessment_verification',{}).get('pass') is True and receipt['assessment_verification'].get('errors')==[] and receipt['assessment_verification'].get('assessment_sha256')==sha(price_bytes),'exact_independently_verified_assessment:'+name)
            self.check(receipt.get('auditor_sha256')==sha(read(auditor_path)),'exact_independent_auditor_source:'+name)
            reported=campaign['studies'][name]
            expected_study={'source_report_sha256':sha(report_bytes),'assessment_sha256':sha(price_bytes),'frozen_git_head':receipt['expected_frozen_head'],'attempts':sum(r.get('attempted') is True for r in receipt['runs']),'successful_assignments':sum(r.get('successful_completion') is True for r in receipt['runs'])}
            expected_study['failed_assignments']=expected_study['attempts']-expected_study['successful_assignments']
            for key,value in expected_study.items(): self.check(reported.get(key)==value,'study_provenance_outcome:'+name+':'+key)
            studies[name]={**expected_study,'independent_audit_sha256':sha(receipt_bytes),'independent_auditor_sha256':receipt['auditor_sha256']}
            self.check(len(receipt['runs'])==60,'all_original_rows_in_receipt:'+name)
            for row in receipt['runs']:
                self.check(row.get('attempted') is True,'all_actual_attempts_retained:'+name+':'+row['run_id'])
                rows.append({'study':name,'run_id':row['run_id'],'configuration':row['configuration'],'success':row.get('successful_completion') is True,'cost':{key:amount(row['cost'][key]) for key in KEYS}})
        section=campaign['campaign']; originals=section['runs']
        expected={(r['study'],r['run_id']):r for r in rows}; actual={(r['study'],r['run_id']):r for r in originals}
        self.check(len(rows)==len(expected)==len(originals)==len(actual)==120,'one_hundred_twenty_original_attempts_once')
        self.check(len({r['run_id'] for r in rows})==120,'fresh_confirmation_task_run_ids_disjoint')
        self.check(set(actual)==set(expected),'all_failed_and_successful_originals_retained')
        self.check(section.get('original_assignments')==120 and section.get('originals_per_arm')==40,'planned_campaign_population_120_40')
        for key,row in expected.items():
            other=actual[key]
            self.check(other.get('configuration')==row['configuration'] and other.get('success') is row['success'],'original_success_and_arm:'+str(key))
            for field,value in row['cost'].items(): self.number(other['cost'].get(field),value,'original_cost:'+str(key)+':'+field)
        expected_aggregate=aggregate(rows)
        self.check(set(section['configurations'])==set(ARMS),'all_three_financial_arm_summaries')
        for arm,summary in expected_aggregate.items():
            other=section['configurations'][arm]
            for key in ('attempts','successful_assignments','failed_assignments','unknown_cost_assignments'):
                self.check(other.get(key)==summary[key],'arm_denominator:'+arm+':'+key)
            self.check(summary['attempts']==40,'forty_attempts_each_arm:'+arm)
            for kind in ('cost_total_usd','cost_per_successful_task_usd'):
                for field,value in summary[kind].items(): self.number(other[kind].get(field),value,'arm_all_attempt_spend:'+arm+':'+kind+':'+field)
        return {'schema_version':'independent-workflow-campaign-accounting-1','created_at_utc':datetime.now(timezone.utc).isoformat(),'pass':not self.errors,'errors':self.errors,'checked_assertions':self.checks,'campaign_assessment_sha256':sha(campaign_bytes),'auditor_sha256':sha(Path(__file__).read_bytes()),'studies':studies,'planned_originals':120,'originals_per_arm':40,'configurations':expected_aggregate,'runs':rows,'actual_billed_cost_usd':None,'scope':'Financial attempt accounting only across different corpora. No pooled efficacy, latency, retrieval-quality, or causal-benefit comparison.'}


def preflight():
    rows=[{'configuration':arm,'success':success,'cost':dict(zip(KEYS,(Decimal('.1'),Decimal('.08'),Decimal('.12'))))} for arm in ARMS for success in (False,True)]
    result=aggregate(rows)
    assert all(r['attempts']==2 and r['successful_assignments']==1 and r['failed_assignments']==1 and r['cost_per_successful_task_usd'][KEYS[0]]==Decimal('.2') for r in result.values())
    rows[0]['cost'][KEYS[0]]=None
    assert aggregate(rows)['A']['cost_total_usd'][KEYS[0]] is None and aggregate(rows)['A']['unknown_cost_assignments']==1
    for row in rows: row['success']=False
    assert all(r['cost_per_successful_task_usd'][KEYS[1]] is None for r in aggregate(rows).values())
    return {'pass':True,'mode':'synthetic_preflight_only','no_study_or_campaign_data_opened':True,'auditor_sha256':sha(Path(__file__).read_bytes())}


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--formal-audit',type=Path)
    parser.add_argument('--confirmation-audit',type=Path)
    parser.add_argument('--formal-report',type=Path,default=ROOT/'benchmarks/codex-workflow-ab.json')
    parser.add_argument('--confirmation-report',type=Path,default=ROOT/'benchmarks/codex-workflow-confirmation.json')
    parser.add_argument('--formal-assessment',type=Path,default=ROOT/'benchmarks/workflow-assessment.json')
    parser.add_argument('--confirmation-assessment',type=Path,default=ROOT/'benchmarks/workflow-confirmation-assessment.json')
    parser.add_argument('--assessment',type=Path,default=ROOT/'benchmarks/efficiency-assessment.json')
    parser.add_argument('--output-prefix',type=Path,required=True)
    parser.add_argument('--preflight',action='store_true')
    args=parser.parse_args(); prefix=args.output_prefix.resolve()
    if not prefix.is_relative_to(ROOT/'.local/efficiency-v1'): raise ValueError('Audit outputs must remain under .local/efficiency-v1')
    outputs=[prefix.with_suffix(suffix) for suffix in ('.json','.md','.invocation.json')]
    if any(p.exists() for p in outputs): raise FileExistsError('Fresh output prefix required')
    prefix.parent.mkdir(parents=True,exist_ok=True)
    with outputs[2].open('x') as out: json.dump({'started_at_utc':datetime.now(timezone.utc).isoformat(),'argv':sys.argv,'auditor_sha256':sha(Path(__file__).read_bytes()),'preflight_only':args.preflight},out,indent=2); out.write('\n')
    try:
        result=preflight() if args.preflight else Audit(args).run()
    except Exception as error:
        result={'pass':False,'errors':[{'auditor_exception':type(error).__name__+': '+str(error)}],'checked_assertions':0,'auditor_sha256':sha(Path(__file__).read_bytes())}
    with outputs[0].open('x') as out: json.dump(result,out,indent=2,default=str); out.write('\n')
    with outputs[1].open('x') as out:
        out.write('# Independent campaign financial accounting audit\n\nPass: **'+str(result['pass'])+'**.\n\n')
        out.write('All original failed and successful attempt costs enter the numerator. Only strict successful assignments enter the denominator. Different corpora are not pooled for efficacy or latency. Actual billing is unknown.\n')
        for error in result.get('errors',[]): out.write('\n- '+json.dumps(error)+'\n')
    print(json.dumps({key:value for key,value in result.items() if key in {'pass','mode','checked_assertions','planned_originals','originals_per_arm','errors'}}))
    return int(not result['pass'])


if __name__=='__main__': raise SystemExit(main())
