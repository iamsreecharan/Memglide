def frozen_fixture_literals() -> tuple[list[bytes], str]:
    global XML_FIXTURES
    if XML_FIXTURES is not None:
        return XML_FIXTURES
    source = subprocess.check_output(
        ["git", "show", f"{XML_FROZEN_COMMIT}:tests/test_security_storage.py"],
        cwd=ROOT,
        stderr=subprocess.PIPE,
    )

    def literal(node: ast.AST) -> str | int:
        if isinstance(node, ast.Constant) and type(node.value) in {str, int}:
            return node.value
        if isinstance(node, ast.BinOp):
            a, b = literal(node.left), literal(node.right)
            if isinstance(node.op, ast.Add) and isinstance(a, str) and isinstance(b, str):
                return a + b
            if (
                isinstance(node.op, ast.Mult)
                and isinstance(a, str)
                and type(b) is int
                and 0 <= b <= 1000
            ):
                return a * b
        raise ValueError

    values = []
    for node in ast.walk(ast.parse(source)):
        try:
            value = literal(node)
        except ValueError:
            continue
        if isinstance(value, str):
            values.append(value.encode())
    XML_FIXTURES = values, sha(source)
    return XML_FIXTURES


def scan_frozen_synthetic_xml(path: Path, data: bytes) -> None:
    relative = str(path.relative_to(ROOT))
    if sha(data) != XML_APPROVALS[relative]:
        fail("Exact frozen validation XML changed", path)
    fixtures, source_sha = frozen_fixture_literals()
    try:
        tree = ET.fromstring(data)
    except ET.ParseError:
        fail("Malformed exact-approved validation XML", path)
    actual_names = Counter((e.get("classname"), e.get("name")) for e in tree.iter("testcase"))
    spans = []
    tag_pattern = re.compile(rb"""<testcase\b(?:[^"'<>]|"[^"]*"|'[^']*')*>""", re.DOTALL)
    name_pattern = re.compile(
        rb"""\bname\s*=\s*(?:"(?P<double>[^"]*)"|'(?P<single>[^']*)')""", re.DOTALL
    )
    for tag in tag_pattern.finditer(data):
        opening = tag.group()
        try:
            element = ET.fromstring(
                opening if opening.rstrip().endswith(b"/>") else opening[:-1] + b"/>"
            )
        except ET.ParseError:
            fail("Malformed approved testcase opening tag", path)
        name, classname = element.get("name", ""), element.get("classname", "")
        identity = (classname, name)
        if not actual_names[identity]:
            continue
        actual_names[identity] -= 1
        if classname != "tests.test_security_storage" or name.split("[")[0] not in {
            "test_detected_credentials_are_excluded",
            "test_additional_credential_exclusions",
        }:
            continue
        attribute = name_pattern.search(opening)
        if attribute is None:
            fail("Missing validated synthetic testcase name", path)
        group = "double" if attribute.group("double") is not None else "single"
        spans.append((tag.start() + attribute.start(group), tag.start() + attribute.end(group)))
    matches = [m for pattern in SECRET_PATTERNS for m in pattern.finditer(data)]
    if len(matches) != 7:
        fail("Unexpected frozen synthetic fixture match count", path)
    masked = bytearray(data)
    for match in matches:
        if not any(
            start <= match.start() and match.end() <= end for start, end in spans
        ) or not any(match.group() in fixture for fixture in fixtures):
            fail("Credential pattern outside an exact public synthetic testcase name", path)
        masked[match.start() : match.end()] = b"_" * (match.end() - match.start())
    # Scan the original byte stream with only proven match spans masked. XML
    # comments, processing instructions, attributes and diagnostics stay intact.
    if any(pattern.search(masked) for pattern in SECRET_PATTERNS):
        fail("Credential pattern outside the frozen testcase-name exception", path)
    XML_SCAN_EXCEPTIONS[relative] = {
        "path": relative,
        "sha256": sha(data),
        "pattern_matches": len(matches),
        "matched_sha256": [sha(match.group()) for match in matches],
        "fixture_source": f"{XML_FROZEN_COMMIT}:tests/test_security_storage.py",
        "fixture_source_sha256": source_sha,
        "original_xml_bytes_preserved": True,
        "all_other_original_bytes_scanned": True,
    }


def fail(message: str, path: Path | None = None) -> None:
    # Never print file content, matched credentials, or parser exception excerpts.
    suffix = f": {path.relative_to(ROOT)}" if path is not None else ""
    raise SystemExit(message + suffix)


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def canonical(value: Any) -> bytes:
    return json.dumps(value, sort_keys=True, separators=(",", ":")).encode()


def safe_source(path: Path) -> None:
    if not path.is_relative_to(ROOT) or ".." in path.parts or path.absolute() != path.resolve():
        fail("Source is outside the canonical repository path")
    if path == ROOT / ".local/efficiency-v1/browser.json":
        fail("Raw browser configuration may contain credentials; use only a sanitized receipt")
    if (
        FORBIDDEN_PARTS.intersection(path.relative_to(ROOT).parts)
        or path.suffix in FORBIDDEN_SUFFIXES
    ):
        fail("Refusing excluded source", path)
    current = path
    while current != ROOT:
        if current.is_symlink():
            fail("Refusing a symlink in the source path", path)
        current = current.parent
    if not path.is_file():
        fail("Required evidence is missing", path)


def scan_text_bytes(source_path: Path, data: bytes) -> None:
    """Scan exact UTF-8 bytes; source_path is the manifest's original ROOT path.

    Bundle builders may use this API on bytes read from public copies, passing
    ROOT / manifest_entry["source"]. The two XML exceptions remain bound to
    exact source path, original SHA256 and frozen public AST fixture literals.
    """
    if not source_path.is_relative_to(ROOT):
        fail("Scanner source path is outside the repository")
    try:
        data.decode("utf-8")
    except UnicodeError:
        fail("Refusing nontext evidence", source_path)
    if b"\0" in data:
        fail("Refusing binary evidence", source_path)
    if str(source_path.relative_to(ROOT)) in XML_APPROVALS:
        scan_frozen_synthetic_xml(source_path, data)
    elif any(pattern.search(data) for pattern in SECRET_PATTERNS):
        fail("Potential credential pattern; publication stopped", source_path)


def read_text_bytes(path: Path) -> bytes:
    safe_source(path)
    data = path.read_bytes()
    scan_text_bytes(path, data)
    return data


def no_duplicate_keys(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
    result: dict[str, Any] = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("Duplicate key")
        result[key] = value
    return result


def read_json(path: Path) -> Any:
    try:
        return json.loads(read_text_bytes(path), object_pairs_hook=no_duplicate_keys)
    except (ValueError, UnicodeError):
        fail("Invalid evidence JSON", path)


def identifier(value: Any) -> str:
    if not isinstance(value, str) or not SAFE_ID.fullmatch(value):
        fail("Unsafe or missing attempt identifier")
    return value


@dataclass(frozen=True)
class Artifact:
    source: Path
    destination: Path
    size: int
    sha256: str


class Collection:
    def __init__(self) -> None:
        self.files: dict[str, Artifact] = {}
        self.missing: list[dict[str, str]] = []
        self.experiments: dict[str, Any] = {}

    def add(self, source: Path, destination: Path, *, required: bool = False) -> None:
        if destination.is_absolute() or ".." in destination.parts:
            fail("Unsafe destination")
        if (
            FORBIDDEN_PARTS.intersection(destination.parts)
            or destination.suffix in FORBIDDEN_SUFFIXES
        ):
            fail("Excluded destination")
        if not source.exists() and not source.is_symlink():
            if required:
                fail("Required evidence is missing", source)
            self.missing.append(
                {
                    "source": str(source.relative_to(ROOT)),
                    "reason": "Not emitted; inspect run status/language before interpreting absence",
                }
            )
            return
        key = str(destination)
        if key in self.files:
            fail("Duplicate destination")
        data = read_text_bytes(source)
        self.files[key] = Artifact(source, destination, len(data), sha(data))

    def attempt(self, source: Path, destination: Path) -> None:
        for name in RUN_FILES:
            self.add(
                source / name,
                destination / name,
                required=name in {"attempt-start.json", "result.json"},
            )
        for name in ("process-identity.json", "launch-grant.json"):
            path = source / name
            if path.exists():
                identity = read_json(path)
                if set(identity) != {"pid", "process_group", "start_ticks"} or not all(
                    isinstance(v, int) and not isinstance(v, bool) and v > 0
                    for v in identity.values()
                ):
                    fail("Unexpected process identity schema; review before publishing", path)

    def experiment(
        self, label: str, source_relative: str, expected: int, version: str, corpus: str
    ) -> None:
        source, destination = ROOT / source_relative, Path(label)
        metadata = read_json(source / "metadata.json")
        if not metadata.get("finished_at"):
            fail(
                "Experiment is not finished; wait for root authorization", source / "metadata.json"
            )
        plan, state = read_json(source / "plan.json"), read_json(source / "state.json")
        runs, supplemental = (
            read_json(source / "runs.json"),
            read_json(source / "supplemental-runs.json"),
        )
        report = read_json(source / "report.json")
        planned = [identifier(row.get("run_id")) for row in plan["runs"]]
        if len(planned) != expected or len(set(planned)) != expected:
            fail("Unexpected plan size or duplicated original assignment", source / "plan.json")
        if set(state["originals"]) != set(planned):
            fail("State does not retain the complete original plan", source / "state.json")
        if len(runs) != expected or {identifier(r.get("run_id")) for r in runs} != set(planned):
            fail("Run projection does not retain every original", source / "runs.json")
        if len(report["runs"]) != expected or {
            identifier(r.get("run_id")) for r in report["runs"]
        } != set(planned):
            fail("Report does not retain every original", source / "report.json")
        if (
            runs != list(state["originals"].values())
            or supplemental != state["supplemental"]
            or supplemental != report.get("supplemental_runs", [])
        ):
            fail("State/projections/supplemental report disagree", source / "state.json")
        if metadata.get("experiment_version") != version:
            fail("Unexpected experiment version", source / "metadata.json")
        tasks = plan["tasks"]
        task_ids = {identifier(task["id"]) for task in tasks}
        if len(tasks) != 10 or len(task_ids) != 10:
            fail("Expected the complete ten-task fresh workflow corpus", source / "plan.json")
        if (
            metadata.get("configuration_labels") != LABELS
            or report.get("configuration_labels") != LABELS
        ):
            fail("Unexpected three-arm treatment labels")
        if metadata.get("corpus") != corpus or metadata.get("repeats") != 2:
            fail("Wrong fresh corpus or repeat count")
        paired = Counter(
            (row["task_id"], row["repeat"], row["configuration"]) for row in plan["runs"]
        )
        if paired != {
            (task_id, repeat, arm): 1 for task_id in task_ids for repeat in (1, 2) for arm in LABELS
        }:
            fail("Expected exactly twenty complete original task/repeat triplets")
        selection_counts = {}
        for arm in LABELS:
            selection_counts[arm] = dict(
                Counter(
                    row.get("selection_stratum")
                    for row in plan["runs"]
                    if row["configuration"] == arm
                )
            )
            if selection_counts[arm] != {"explicit_focus": 16, "caller_context_supplied": 4}:
                fail("Unexpected original arm selection strata")
            if any(
                row.get("context_format") != (None if arm == "A" else "compact-prefix")
                for row in plan["runs"]
                if row["configuration"] == arm
            ):
                fail("Unexpected assigned context format")
        for name in TOP_FILES:
            self.add(source / name, destination / name, required=True)
        for name in INVOCATION_FILES:
            self.add(source / "preflight" / name, destination / "preflight" / name)

        # Only the narrowly named attempt-log directory and six-digit JSON files
        # are enumerated. No arbitrary directory contents are copied.
        logdir = source / "attempt-log"
        if logdir.is_symlink() or not logdir.is_dir():
            fail("Missing or unsafe immutable attempt log")
        names = sorted(p.name for p in logdir.iterdir())
        if any(not re.fullmatch(r"[0-9]{6}\.json", name) for name in names):
            fail("Unexpected file in immutable attempt log")
        records, previous = [], None
        for index, name in enumerate(names, 1):
            path = logdir / name
            record = read_json(path)
            body = {key: value for key, value in record.items() if key != "sha256"}
            if (
                name != f"{index:06d}.json"
                or body.get("sequence") != index
                or body.get("previous_sha256") != previous
                or sha(canonical(body)) != record.get("sha256")
            ):
                fail("Immutable attempt hash chain failed", path)
            previous = record["sha256"]
            records.append(record)
            self.add(path, destination / "attempt-log" / name, required=True)
        if (
            not records
            or records[-1].get("event") != "experiment.finished"
            or records[-1].get("report_sha256") != sha(read_text_bytes(source / "report.json"))
        ):
            fail("Experiment-finished seal is missing or invalid", source / "report.json")
        for record in records:
            event = record.get("event")
            if event == "attempt.started":
                path = source / "runs" / identifier(record.get("attempt_id")) / "attempt-start.json"
                if sha(canonical(read_json(path))) != record.get("start_sha256"):
                    fail("Immutable attempt-start seal failed", path)
            if event in {"attempt.finished", "attempt.interrupted"}:
                path = source / "runs" / identifier(record.get("attempt_id")) / "result.json"
                if sha(read_text_bytes(path)) != record.get("result_sha256"):
                    fail("Immutable result seal failed", path)

        attempts = [r for r in runs if r.get("attempted")] + supplemental
        attempt_ids = [identifier(r.get("run_id")) for r in attempts]
        if len(attempt_ids) != len(set(attempt_ids)):
            fail("Duplicate original/supplemental attempt identifier")
        if len(records) != 2 + 2 * len(attempt_ids):
            fail("Unexpected durable log event count")
        if Counter(
            record.get("attempt_id")
            for record in records
            if record.get("event") == "attempt.started"
        ) != Counter(attempt_ids):
            fail("Every invocation must have exactly one durable start seal")
        if Counter(
            record.get("attempt_id")
            for record in records
            if record.get("event") in {"attempt.finished", "attempt.interrupted"}
        ) != Counter(attempt_ids):
            fail("Every invocation must have exactly one durable terminal seal")
        for row in attempts:
            run_id = identifier(row["run_id"])
            directory = source / "runs" / run_id
            if Path(row["artifacts"]).absolute() != directory:
                fail("Attempt artifact pointer escaped its designated run directory")
            sealed = directory / "result.json"
            if read_json(sealed) != row:
                fail("State row differs from immutable result", sealed)
            self.attempt(directory, destination / "runs" / run_id)
        # Preserve whitelisted text from any unregistered partial attempt too;
        # it stays explicitly separate from original/supplemental result rows.
        unregistered = []
        rundir = source / "runs"
        if rundir.exists():
            if rundir.is_symlink():
                fail("Unsafe runs directory")
            for directory in sorted(rundir.iterdir()):
                run_id = identifier(directory.name)
                if directory.is_symlink() or not directory.is_dir():
                    fail("Unexpected object in runs directory")
                if run_id not in attempt_ids:
                    unregistered.append(run_id)
                    self.attempt(
                        directory, destination / "unregistered-partial-invocations" / run_id
                    )
        self.experiments[label] = {
            "source": source_relative,
            "planned_originals": expected,
            "original_status_counts": {
                status: sum(r.get("status") == status for r in runs)
                for status in sorted({r.get("status") for r in runs})
            },
            "supplemental_attempts": len(supplemental),
            "unregistered_partial_invocations": unregistered,
            "treatment": "A standard native Codex / B production compact source-only / C compact complete source plus caller-selected support",
            "planned_selection_strata_by_arm": selection_counts,
            "observed_preparation_modes_by_arm": {
                arm: dict(
                    Counter(
                        row.get("preload_treatment_fidelity", {}).get("mode", "unknown")
                        for row in runs
                        if row["configuration"] == arm
                    )
                )
                for arm in LABELS
            },
            "finished_at": metadata["finished_at"],
            "attempt_log_records": len(records),
            "last_attempt_log_sha256": previous,
            "source_unchanged_during_experiment": metadata.get(
                "source_unchanged_during_experiment"
            ),
        }

    def corpus_validation(self) -> None:
        for stage in CORPUS_STAGES:
            source = ROOT / ".local/efficiency-v1" / stage
            if source.is_symlink() or not source.is_dir():
                fail("Unsafe or missing corpus validation directory")
            self.add(
                source / "validation.json",
                Path("corpus/controls") / stage / "validation.json",
                required=True,
            )
            for case in sorted(source.iterdir()):
                if case.name == "validation.json":
                    continue
                if case.is_symlink():
                    if case.name.endswith("current"):
                        continue
                    fail("Unexpected corpus control symlink", case)
                identifier(case.name)
                if not case.is_dir():
                    fail("Unexpected corpus control directory object", case)
                for grade_name in CORPUS_GRADE_DIRS:
                    directory = case / grade_name
                    if directory.is_symlink():
                        fail("Unsafe corpus grading artifact directory", directory)
                    if not directory.exists():
                        continue
                    if not directory.is_dir():
                        fail("Unexpected corpus grading object", directory)
                    for name in CORPUS_GRADE_FILES:
                        self.add(
                            directory / name,
                            Path("corpus/controls") / stage / case.name / grade_name / name,
                        )


def verify_frozen_source() -> dict[str, Any]:
    head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT).decode().strip()
    if head != FROZEN_COMMIT:
        fail("Final confirmation frozen Git HEAD changed")
    studies = {}
    for study in EXPERIMENTS:
        expected_head, expected_app = study["git_head"], study["app"]
        source = ROOT / study["source"]
        metadata = read_json(source / "metadata.json")
        freeze = read_json(ROOT / ".local/efficiency-v1" / study["freeze"])
        if read_json(ROOT / ".local/efficiency-v1" / study["exit"]).get("returncode") != 0:
            fail("Study driver did not finish with exit 0")
        if (
            freeze.get("git_head", freeze.get("frozen_commit")) != expected_head
            or freeze["application"]["source_tree_sha256"] != expected_app
        ):
            fail("Study freeze receipt differs from its authorized source")
        for phase in ("before", "after"):
            record = metadata["memglide_source_" + phase]
            if (
                record.get("git_head") != expected_head
                or record.get("source_tree_sha256") != expected_app
                or record.get("source_dirty_entries")
            ):
                fail("Per-study application source attestation differs")
            if sha(json.dumps(record["source_files"], sort_keys=True).encode()) != expected_app:
                fail("Application aggregate does not match study's frozen source map")
            for name, digest in record["source_files"].items():
                if not name.startswith("src/memglide/") or not name.endswith(".py"):
                    fail("Application source map contains an unexpected path")
                data = read_text_bytes(ROOT / name)
                original = subprocess.check_output(
                    ["git", "show", f"{expected_head}:{name}"], cwd=ROOT, stderr=subprocess.PIPE
                )
                if sha(data) != digest or data != original:
                    fail(
                        "Unchanged current application differs from per-study frozen Git bytes",
                        ROOT / name,
                    )
        for name, digest in metadata["frozen_manifest"]["files"].items():
            data = read_text_bytes(ROOT / name)
            original = subprocess.check_output(
                ["git", "show", f"{expected_head}:{name}"], cwd=ROOT, stderr=subprocess.PIPE
            )
            if sha(data) != digest or data != original:
                fail(
                    "Frozen measurement/source input differs from its study's exact Git bytes",
                    ROOT / name,
                )
        if metadata.get("source_unchanged_during_experiment") is not True:
            fail("Study did not attest unchanged measured source")
        studies[study["label"]] = {
            "git_head": expected_head,
            "application_source_tree_sha256": expected_app,
            "formal_driver_returncode": 0,
            "current_source_and_frozen_inputs_verified": True,
        }
    return {"final_git_head": head, "studies": studies, "independent_study_freezes": True}


def verify_historical_evidence() -> dict[str, Any]:
    """Compare prior report/evidence/archive bytes, allowing reviewed helper edits."""
    raw = subprocess.check_output(["git", "ls-tree", "-rz", BASELINE, "--", "benchmarks"], cwd=ROOT)
    entries = []
    for line in raw.split(b"\0"):
        if not line:
            continue
        info, path_raw = line.split(b"\t", 1)
        mode, kind, oid = info.split()
        name = path_raw.decode()
        if kind != b"blob" or mode not in {b"100644", b"100755"}:
            fail("Unexpected historical Git object")
        if name.startswith("benchmarks/evidence/") or not name.endswith(".py"):
            entries.append((name, oid))
    verified, archives = [], []
    process = subprocess.Popen(
        ["git", "cat-file", "--batch"],
        cwd=ROOT,
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )
    assert process.stdin is not None and process.stdout is not None
    try:
        for relative, oid in entries:
            path = ROOT / relative
            if path.absolute() != path.resolve() or not path.is_file():
                fail("Missing or unsafe immutable historical artifact", path)
            process.stdin.write(oid + b"\n")
            process.stdin.flush()
            header = process.stdout.readline().split()
            if len(header) != 3 or header[0] != oid or header[1] != b"blob":
                fail("Frozen historical Git blob unavailable")
            original = process.stdout.read(int(header[2]))
            if process.stdout.read(1) != b"\n":
                fail("Invalid frozen Git blob boundary")
            current = path.read_bytes()
            if current != original:
                fail("Immutable historical report/evidence/archive changed", path)
            record = {"path": relative, "bytes": len(current), "sha256": sha(current)}
            verified.append(record)
            if path.suffix == ".zip":
                archives.append(record)
    finally:
        process.stdin.close()
        process.stdout.close()
        if process.wait() != 0:
            fail("Historical Git object reader failed")
    if len(archives) < 4:
        fail("Expected at least four immutable historical ZIPs")
    return {
        "base_commit": BASELINE,
        "scope": "All pre-iteration tracked benchmarks reports/configs/evidence/archives; authorized Python helper edits excluded unless under immutable evidence",
        "unchanged_files": len(verified),
        "unchanged_bytes": sum(row["bytes"] for row in verified),
        "sha256_of_sorted_file_records": sha(canonical(verified)),
        "whole_original_zip_files": archives,
    }


def finished_guard() -> None:
    """Read only both completion receipts/metadata before any raw study input."""
    for study in EXPERIMENTS:
        receipt = read_json(ROOT / ".local/efficiency-v1" / study["exit"])
        metadata = read_json(ROOT / study["source"] / "metadata.json")
        if receipt.get("returncode") != 0 or not metadata.get("finished_at"):
            fail(
                "Both study driver/metadata pairs must attest completion before raw evidence is inspected"
            )


def verify_independent_audit(
    study: dict[str, Any], audit_path: Path, auditor_path: Path
) -> dict[str, Any]:
    audit = read_json(audit_path)
    report_path = ROOT / study["source"] / "report.json"
    assessment_bytes = read_text_bytes(ROOT / (study["assessment"] + ".json"))
    assessment = json.loads(assessment_bytes)
    verification = audit.get("assessment_verification", {})
    if (
        audit.get("pass") is not True
        or audit.get("errors")
        or audit.get("expected_frozen_head") != study["git_head"]
        or audit.get("expected_app_sha256") != study["app"]
        or audit.get("source_report_sha256") != sha(read_text_bytes(report_path))
        or audit.get("auditor_sha256") != sha(read_text_bytes(auditor_path))
        or audit.get("planned_originals") != 60
        or audit.get("verified_triplets") != 20
        or not isinstance(audit.get("checked_assertions"), int)
        or audit["checked_assertions"] <= 0
    ):
        fail("Independent study audit does not bind its complete frozen sixty-assignment study")
    if (
        verification.get("pass") is not True
        or verification.get("errors")
        or verification.get("assessment_sha256") != sha(assessment_bytes)
        or not isinstance(verification.get("checked_assertions"), int)
        or verification["checked_assertions"] <= 0
        or audit.get("pricing_sha256")
        != sha(read_text_bytes(ROOT / "benchmarks/cost-pricing.json"))
    ):
        fail("Independent study numeric comparison does not bind final assessment/pricing")
    if assessment.get("actual_billed_cost_usd") is not None:
        fail("Unavailable actual charges must remain null")
    return {
        "source": str(audit_path.relative_to(ROOT)),
        "published_path": "audit/" + study["label"] + "-independent-audit.json",
        "sha256": sha(read_text_bytes(audit_path)),
        "auditor_sha256": sha(read_text_bytes(auditor_path)),
        "pass": True,
        "checked_assertions": audit["checked_assertions"],
        "assessment_verification": verification,
        "source_report_sha256": audit["source_report_sha256"],
        "pricing_sha256": audit["pricing_sha256"],
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--expected-head", required=True, help="Final confirmation freeze")
    parser.add_argument("--expected-app", required=True)
    parser.add_argument("--expected-v1-head", required=True)
    parser.add_argument("--expected-v1-app", required=True)
    parser.add_argument("--expected-scanner-sha256", required=True)
    parser.add_argument("--extras-json", type=Path, required=True)
    parser.add_argument(
        "--audit-json", type=Path, required=True, help="Confirmation final independent audit"
    )
    parser.add_argument("--auditor", type=Path, required=True)
    parser.add_argument("--v1-audit-json", type=Path, required=True)
    parser.add_argument("--v1-auditor", type=Path, required=True)
    parser.add_argument("--campaign-audit-json", type=Path, required=True)
    parser.add_argument("--campaign-auditor", type=Path, required=True)
    parser.add_argument("--publish", action="store_true")
    args = parser.parse_args()
    global FROZEN_COMMIT, FROZEN_APP, XML_FROZEN_COMMIT, XML_APPROVALS
    if not all(
        re.fullmatch(r"[0-9a-f]{40}", value)
        for value in (args.expected_head, args.expected_v1_head)
    ) or not all(
        re.fullmatch(r"[0-9a-f]{64}", value)
        for value in (args.expected_app, args.expected_v1_app, args.expected_scanner_sha256)
    ):
        fail("Expected full independent frozen Git/application/scanner SHA values")
    if sha(Path(__file__).read_bytes()) != args.expected_scanner_sha256:
        fail("Scanner differs from explicitly reviewed source identity")
    FROZEN_COMMIT, FROZEN_APP, XML_FROZEN_COMMIT = (
        args.expected_head,
        args.expected_app,
        args.expected_v1_head,
    )
    EXPERIMENTS[0]["git_head"], EXPERIMENTS[0]["app"] = args.expected_v1_head, args.expected_v1_app
    EXPERIMENTS[1]["git_head"], EXPERIMENTS[1]["app"] = args.expected_head, args.expected_app
    finished_guard()
    output = DEFAULT_OUTPUT
    if output.resolve() != output or output.exists() or output.is_symlink():
        fail("Evidence target must be fresh and nonsymlink; no overwrite supported")
    selected_paths = [
        path.absolute()
        for path in (
            args.extras_json,
            args.audit_json,
            args.auditor,
            args.v1_audit_json,
            args.v1_auditor,
            args.campaign_audit_json,
            args.campaign_auditor,
        )
    ]
    if any(not path.is_relative_to(ROOT / ".local/efficiency-v1") for path in selected_paths):
        fail("Configuration/audit inputs must be below the ignored efficiency study root")
    extras_path = args.extras_json.absolute()
    extras = read_json(extras_path)
    XML_APPROVALS = extras.get("xml_approvals", {})
    if not isinstance(XML_APPROVALS, dict) or any(
        not isinstance(value, str)
        or not re.fullmatch(r"[0-9a-f]{64}", value)
        or not path.startswith(".local/efficiency-v1/")
        or not path.endswith(".xml")
        for path, value in XML_APPROVALS.items()
    ):
        fail("Unsafe exact XML synthetic-fixture approval")
    source_frozen = verify_frozen_source()
    audit_inputs = (
        (EXPERIMENTS[0], args.v1_audit_json.absolute(), args.v1_auditor.absolute()),
        (EXPERIMENTS[1], args.audit_json.absolute(), args.auditor.absolute()),
    )
    audits = {
        study["label"]: verify_independent_audit(study, audit_path, auditor_path)
        for study, audit_path, auditor_path in audit_inputs
    }
    campaign_path, campaign_auditor = (
        args.campaign_audit_json.absolute(),
        args.campaign_auditor.absolute(),
    )
    campaign = read_json(campaign_path)
    campaign_assessment_bytes = read_text_bytes(ROOT / "benchmarks/efficiency-assessment.json")
    if (
        campaign.get("pass") is not True
        or campaign.get("errors")
        or campaign.get("campaign_assessment_sha256") != sha(campaign_assessment_bytes)
        or campaign.get("auditor_sha256") != sha(read_text_bytes(campaign_auditor))
        or campaign.get("planned_originals") != 120
        or campaign.get("originals_per_arm") != 40
        or not isinstance(campaign.get("checked_assertions"), int)
        or campaign["checked_assertions"] <= 0
    ):
        fail("Independent campaign accounting does not bind all 120 originals and final narrative")
    for study in EXPERIMENTS:
        binding = campaign.get("studies", {}).get(study["label"], {})
        if (
            binding.get("source_report_sha256") != audits[study["label"]]["source_report_sha256"]
            or binding.get("assessment_sha256")
            != audits[study["label"]]["assessment_verification"]["assessment_sha256"]
        ):
            fail("Campaign accounting does not bind both separate immutable study outputs")
    campaign_proof = {
        "source": str(campaign_path.relative_to(ROOT)),
        "published_path": "audit/campaign-independent-audit.json",
        "sha256": sha(read_text_bytes(campaign_path)),
        "auditor_sha256": sha(read_text_bytes(campaign_auditor)),
        "campaign_assessment_sha256": sha(campaign_assessment_bytes),
        "pass": True,
        "checked_assertions": campaign["checked_assertions"],
    }
    collection = Collection()
    for study in EXPERIMENTS:
        collection.experiment(
            study["label"], study["source"], study["expected"], study["version"], study["corpus"]
        )
    for source, destination in EXTRA_FILES:
        collection.add(ROOT / source, Path(destination), required=True)
    expected_extra_hashes = extras.get("selected_source_sha256", {})
    if (
        not isinstance(expected_extra_hashes, dict)
        or set(expected_extra_hashes) != {entry.get("source") for entry in extras["files"]}
        or any(
            not isinstance(digest, str) or not re.fullmatch(r"[0-9a-f]{64}", digest)
            for digest in expected_extra_hashes.values()
        )
    ):
        fail("Explicit extra artifacts require a complete reviewed source/SHA map")
    for entry in extras["files"]:
        if (
            set(entry) - {"source", "destination", "required"}
            or not isinstance(entry.get("source"), str)
            or not isinstance(entry.get("destination"), str)
        ):
            fail("Invalid explicit artifact specification")
        selected = ROOT / entry["source"]
        if (
            not selected.is_relative_to(ROOT / ".local/efficiency-v1")
            and entry["source"] not in EXPLICIT_LEGACY_HELPERS
        ):
            fail("Extra originals must be inside efficiency root or exact legacy helper allowlist")
        if sha(read_text_bytes(selected)) != expected_extra_hashes[entry["source"]]:
            fail("Extra artifact differs from reviewed original bytes", selected)
        collection.add(selected, Path(entry["destination"]), required=entry.get("required", True))
    collection.add(extras_path, Path("reproduction/explicit-artifacts.json"), required=True)
    for study, audit_path, auditor_path in audit_inputs:
        collection.add(audit_path, Path(audits[study["label"]]["published_path"]), required=True)
        collection.add(
            auditor_path, Path("reproduction") / (auditor_path.name + ".txt"), required=True
        )
    collection.add(campaign_path, Path(campaign_proof["published_path"]), required=True)
    collection.add(
        campaign_auditor, Path("reproduction") / (campaign_auditor.name + ".txt"), required=True
    )
    collection.corpus_validation()
    historical = verify_historical_evidence()
    files = [collection.files[key] for key in sorted(collection.files)]
    manifest = {
        "schema_version": "efficiency-campaign-2",
        "scope": "Two separately frozen ten-task/two-repeat 60-assignment studies; all 120 originals retained, initial strictfailures not replaced and efficacy not pooled",
        "experiments": collection.experiments,
        "source_freeze": source_frozen,
        "historical_preservation": historical,
        "independent_audits": audits,
        "campaign_audit": campaign_proof,
        "whitelist": {
            "top": TOP_FILES,
            "preflight": INVOCATION_FILES,
            "runs": RUN_FILES,
            "extras": EXTRA_FILES,
            "explicit_extra_artifacts": extras,
            "corpus_validation_stages": CORPUS_STAGES,
            "corpus_grading_directories": CORPUS_GRADE_DIRS,
            "corpus_grading_files": CORPUS_GRADE_FILES,
            "attempt_log": "Only contiguous six-digit JSON records with verified hash chain",
        },
        "never_selected": sorted(FORBIDDEN_PARTS)
        + sorted(FORBIDDEN_SUFFIXES)
        + ["raw external research bodies", "untracked file contents", "environment dumps"],
        "missing_optional_files": collection.missing,
        "synthetic_fixture_scan_exceptions": sorted(
            XML_SCAN_EXCEPTIONS.values(), key=lambda row: row["path"]
        ),
        "limitations": [
            "Manifest excludes itself and post-copy publication audit to avoid circular checksums.",
            "Both study protocols and source snapshots remain independent; all initial strict failures and spend remain visible.",
            "Campaign cost per success includes failed prior-study attempts; it is operational accounting, not pooled causal efficacy.",
            "Credential scanning is pattern-based and cannot prove absence of every possible secret.",
            "Exact Git-AST-bound synthetic XML exceptions preserve original bytes and scan all other original spans.",
            "Common prompts and delivery attestations are preserved; private stdin descriptors, native state, auth, SQLite, fixture repositories/worktrees and untracked file contents are excluded.",
        ],
        "files": [
            {
                "path": str(a.destination),
                "source": str(a.source.relative_to(ROOT)),
                "bytes": a.size,
                "sha256": a.sha256,
            }
            for a in files
        ],
    }
    manifest_bytes = json.dumps(manifest, indent=2).encode() + b"\n"
    scan_text_bytes(output / "manifest.json", manifest_bytes)
    if args.publish:
        output.mkdir(parents=True, exist_ok=False)
        for artifact in files:
            data = read_text_bytes(artifact.source)
            if len(data) != artifact.size or sha(data) != artifact.sha256:
                fail("Evidence changed after complete pre-copy validation", artifact.source)
            target = output / artifact.destination
            target.parent.mkdir(parents=True, exist_ok=True)
            with target.open("xb") as stream:
                stream.write(data)
        with (output / "manifest.json").open("xb") as stream:
            stream.write(manifest_bytes)
        for artifact in files:
            copied = (output / artifact.destination).read_bytes()
            if (
                len(copied) != artifact.size
                or sha(copied) != artifact.sha256
                or copied != read_text_bytes(artifact.source)
            ):
                fail("Published source/target bytes differ", artifact.source)
        if verify_historical_evidence() != historical or verify_frozen_source() != source_frozen:
            fail("Frozen sources or historical evidence changed during publication")
        receipt = {
            "pass": True,
            "selected_files": len(files),
            "bytes": sum(a.size for a in files),
            "manifest_sha256": sha(manifest_bytes),
            "every_source_and_target_sha256_verified": True,
            "all_120_originals_retained": True,
            "durable_log_and_result_seals_verified": True,
            "credential_pattern_scan": "passed",
            "independent_audits": audits,
            "campaign_audit": campaign_proof,
            "historical_preservation_rechecked_after_copy": historical,
            "source_freeze_rechecked_after_copy": source_frozen,
            "synthetic_fixture_scan_exceptions": manifest["synthetic_fixture_scan_exceptions"],
        }
        receipt_bytes = json.dumps(receipt, indent=2).encode() + b"\n"
        scan_text_bytes(output / "publication-audit.json", receipt_bytes)
        with (output / "publication-audit.json").open("xb") as stream:
            stream.write(receipt_bytes)
    print(
        json.dumps(
            {
                "mode": "published" if args.publish else "check_only",
                "output": str(output.relative_to(ROOT)),
                "files": len(files),
                "bytes": sum(a.size for a in files),
                "manifest_sha256": sha(manifest_bytes),
                "scanner_sha256": args.expected_scanner_sha256,
                "originals": 120,
                "triplets_per_study": 20,
                "independent_audits_pass": True,
                "missing_optional_files": len(collection.missing),
                "credential_pattern_scan": "passed",
                "synthetic_fixture_exceptions": len(XML_SCAN_EXCEPTIONS),
                "source_freeze": source_frozen,
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
