#!/usr/bin/env python3
"""Verify and package completed, publicly copied workflow evidence into a fresh ZIP.

All input identities are explicit arguments. This helper makes no model requests,
reads no live experiment state, never overwrites, and verifies/scans every selected
byte before opening the output archive. By default it checks selected inputs without creating an archive.
"""

import hashlib

import zipfile

EVIDENCE = ROOT / "benchmarks/evidence/efficiency-v1"
OUTPUT = ROOT / "benchmarks/memglide-efficiency-report.zip"
RECEIPT = ROOT / "benchmarks/efficiency-report-bundle.json"
LABELS = {"A": "Standard Codex", "B": "Memglide source-only", "C": "Memglide source plus support"}
REQUIRED_REPORTS = (

    "benchmarks/codex-workflow-ab.json",
    "benchmarks/workflow-assessment.md",
    "benchmarks/workflow-assessment.json",
    "benchmarks/codex-workflow-confirmation.md",

FORBIDDEN = {
    "auth.json",
    "codex-state",
    "native-state",
    "worktrees",

    "state",

    "config.toml",

    "env.json",
}
BAD_SUFFIXES = {".db", ".sqlite", ".sqlite3", ".tar", ".tgz", ".gz", ".zip", ".whl", ".source"}


def fail(message: str) -> None:
    raise SystemExit(message)


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def encoded(value: object) -> bytes:
    return (json.dumps(value, indent=2, sort_keys=True) + "\n").encode()


def unique_json(data: bytes) -> object:
    def pairs(items):
        result = {}
        for key, value in items:
            if key in result:
                fail("Duplicate JSON key")
            result[key] = value
        return result

    return json.loads(data, object_pairs_hook=pairs)


def safe_name(value: str) -> str:
    path = PurePosixPath(value)
    if (
        not value
        or "\\" in value
        or path.is_absolute()
        or str(path) != value
        or any(part in {"", ".", ".."} | FORBIDDEN for part in path.parts)
        or path.suffix.lower() in BAD_SUFFIXES
    ):
        fail("Unsafe or excluded archive member name")
    return value


def reject_symlinks(path: Path) -> None:
    if not path.is_relative_to(ROOT) or path.absolute() != path.resolve():
        fail("Selected path is outside the canonical repository or contains a symlink")
    if not path.is_file():
        fail("Selected artifact is not a regular file")


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--expected-head", required=True)
    parser.add_argument("--expected-app", required=True)
    parser.add_argument("--expected-v1-head", required=True)
    parser.add_argument("--expected-v1-app", required=True)
    parser.add_argument("--expected-builder-sha256", required=True)
    parser.add_argument("--manifest-sha256", required=True)
    parser.add_argument("--scanner-sha256", required=True)
    parser.add_argument("--evidence-files", type=int, required=True)
    parser.add_argument("--evidence-bytes", type=int, required=True)
    parser.add_argument(
        "--final-files-json",
        type=Path,
        required=True,
        help="Reviewed exact source/SHA list for every finalized report and optional current documentation",
    )
    parser.add_argument("--publish", action="store_true")
    args = parser.parse_args()
    if not all(
        re.fullmatch(r"[0-9a-f]{40}", value)
        for value in (args.expected_head, args.expected_v1_head)
    ) or not all(
        re.fullmatch(r"[0-9a-f]{64}", item)
        for item in (
            args.expected_app,
            args.expected_v1_app,
            args.expected_builder_sha256,
            args.manifest_sha256,
            args.scanner_sha256,
        )
    ):
        fail("Expected full frozen Git and SHA-256 identities")
    builder_path = Path(__file__).absolute()
    if sha(builder_path.read_bytes()) != args.expected_builder_sha256:
        fail("Builder differs from explicitly reviewed source identity")
    if OUTPUT.exists() or OUTPUT.is_symlink() or RECEIPT.exists() or RECEIPT.is_symlink():
        fail("Archive and receipt must both be fresh; overwrites are prohibited")
    if OUTPUT.parent.resolve() != OUTPUT.parent:
        fail("Archive parent contains a symlink")
    manifest_path = EVIDENCE / "manifest.json"
    reject_symlinks(manifest_path)
    manifest_bytes = manifest_path.read_bytes()
    if sha(manifest_bytes) != args.manifest_sha256:
        fail("Evidence manifest differs from independent publication receipt")
    manifest = unique_json(manifest_bytes)
    study_specs = (
        {
            "label": "formal",
            "head": args.expected_v1_head,
            "app": args.expected_v1_app,
            "report": "benchmarks/codex-workflow-ab",
            "assessment": "benchmarks/workflow-assessment",
        },
        {
            "label": "confirmation",
            "head": args.expected_head,
            "app": args.expected_app,
            "report": "benchmarks/codex-workflow-confirmation",
            "assessment": "benchmarks/workflow-confirmation-assessment",
        },
    )
    if (
        manifest.get("schema_version") != "efficiency-campaign-2"
        or manifest["source_freeze"].get("final_git_head") != args.expected_head
    ):
        fail("Evidence final freeze or schema differs")
    for study in study_specs:
        frozen = manifest["source_freeze"]["studies"][study["label"]]
        if (
            frozen.get("git_head") != study["head"]
            or frozen.get("application_source_tree_sha256") != study["app"]
        ):
            fail("Evidence per-study source freeze differs")
    records = manifest["files"]
    if (
        len(records) != args.evidence_files
        or sum(row["bytes"] for row in records) != args.evidence_bytes
    ):
        fail("Evidence count/bytes differ from independent publisher receipt")
    scanner_records = [
        row for row in records if row["path"] == "reproduction/publish_evidence.py.txt"
    ]
    if len(scanner_records) != 1 or scanner_records[0]["sha256"] != args.scanner_sha256:
        fail("Pinned scanner record differs")
    scanner_path = EVIDENCE / scanner_records[0]["path"]
    reject_symlinks(scanner_path)
    scanner_source = scanner_path.read_bytes()
    if (
        sha(scanner_source) != args.scanner_sha256
        or len(scanner_source) != scanner_records[0]["bytes"]
    ):
        fail("Pinned scanner source differs")
    guard = types.ModuleType("frozen_efficiency_bundle_scanner")
    guard.__file__ = str(scanner_path)
    sys.modules[guard.__name__] = guard
    exec(compile(scanner_source, str(scanner_path), "exec"), guard.__dict__)
    guard.ROOT = ROOT
    guard.FROZEN_COMMIT = args.expected_head
    guard.XML_FROZEN_COMMIT = args.expected_v1_head
    guard.FROZEN_APP = args.expected_app
    guard.XML_APPROVALS = manifest["whitelist"]["explicit_extra_artifacts"].get("xml_approvals", {})
    members, origins, identities = {}, {}, {}

    def scan(data: bytes, label: str) -> None:
        try:
            data.decode("utf-8")
        except UnicodeError:
            fail("Selected artifact is not UTF-8 text: " + label)
        if b"\0" in data or any(pattern.search(data) for pattern in guard.SECRET_PATTERNS):
            fail("Potential credential or binary content: " + label)

    def read(path: Path) -> bytes:
        reject_symlinks(path)
        guard.safe_source(path)
        data = path.read_bytes()
        identities[path] = (len(data), sha(data))
        return data

    def add(name: str, data: bytes, origin: dict, *, already_scanned=False) -> None:
        safe_name(name)
        if name in members:
            fail("Duplicate selected archive member")
        if not already_scanned:
            scan(data, name)
        members[name], origins[name] = data, origin

    expected = {"manifest.json", "publication-audit.json"}
    for record in records:
        relative = safe_name(record["path"])
        if relative in expected:
            fail("Duplicate or reserved evidence manifest path")
        expected.add(relative)
        path = EVIDENCE / relative
        data = read(path)
        if len(data) != record["bytes"] or sha(data) != record["sha256"]:
            fail("Published evidence size/hash differs: " + relative)
        source = ROOT / record["source"]
        if source.absolute() != source.resolve() or not source.is_relative_to(ROOT):
            fail("Unsafe original source path in manifest")
        guard.scan_text_bytes(source, data)
        add(
            str(path.relative_to(ROOT)),
            data,
            {
                "kind": "manifested_public_evidence",
                "source": str(path.relative_to(ROOT)),
                "original_source": record["source"],
            },
            already_scanned=True,
        )
    actual = set()
    for directory, directories, files in os.walk(EVIDENCE, followlinks=False):
        for name in directories:
            path = Path(directory) / name
            if path.is_symlink() or path.resolve() != path:
                fail("Symlink in public evidence tree")
        for name in files:
            path = Path(directory) / name
            reject_symlinks(path)
            actual.add(str(path.relative_to(EVIDENCE)))
    if actual != expected:
        fail("Public evidence has missing or unmanifested files")
    exceptions = sorted(guard.XML_SCAN_EXCEPTIONS.values(), key=lambda row: row["path"])
    if exceptions != sorted(
        manifest["synthetic_fixture_scan_exceptions"], key=lambda row: row["path"]
    ):
        fail("Synthetic XML exceptions differ from independent publisher record")
    if read(manifest_path) != manifest_bytes:
        fail("Evidence manifest changed during selection")
    add(
        str(manifest_path.relative_to(ROOT)),
        manifest_bytes,
        {"kind": "pinned_evidence_manifest", "sha256": args.manifest_sha256},
    )
    audit_path = EVIDENCE / "publication-audit.json"
    publication_bytes = read(audit_path)
    publication = unique_json(publication_bytes)
    if (
        publication.get("pass") is not True
        or publication.get("manifest_sha256") != args.manifest_sha256
        or publication.get("selected_files") != args.evidence_files
        or publication.get("bytes") != args.evidence_bytes
        or publication.get("all_120_originals_retained") is not True
    ):
        fail("Publication audit does not validate selected evidence")
    add(str(audit_path.relative_to(ROOT)), publication_bytes, {"kind": "publication_audit"})
    final_config_path = args.final_files_json.absolute()
    if not final_config_path.is_relative_to(ROOT / ".local/efficiency-v1"):
        fail("Reviewed final artifact list must be inside ignored efficiency root")
    final_config_bytes = read(final_config_path)
    final_config = unique_json(final_config_bytes)
    finalized_names = set()
    for record in final_config["files"]:
        if set(record) != {"source", "sha256"} or not re.fullmatch(
            r"[0-9a-f]{64}", record["sha256"]
        ):
            fail("Invalid exact final artifact identity")
        relative = safe_name(record["source"])
        if (
            relative in finalized_names
            or relative.startswith(".local/")
            or relative.startswith("benchmarks/evidence/")
        ):
            fail("Duplicate or unsafe finalized artifact")
        finalized_names.add(relative)
        data = read(ROOT / relative)
        if sha(data) != record["sha256"]:
            fail("Final artifact differs from reviewed identity: " + relative)
        add(relative, data, {"kind": "finalized_report_or_documentation", "source": relative})
    if not set(REQUIRED_REPORTS) <= finalized_names:
        fail("Final artifact list omits a required report/document")
    add(
        "reproduction/final-artifacts.json",
        final_config_bytes,
        {"kind": "reviewed_exact_final_artifact_list"},
    )
    assessments = {}
    independent_proofs = manifest["independent_audits"]
    for study in study_specs:
        label = study["label"]
        report_bytes = members[study["report"] + ".json"]
        report = unique_json(report_bytes)
        assessment_bytes = members[study["assessment"] + ".json"]
        assessment = unique_json(assessment_bytes)
        assessments[label] = assessment
        evidence_prefix = "benchmarks/evidence/efficiency-v1/" + label
        if (
            report_bytes != members[evidence_prefix + "/report.json"]
            or members[study["report"] + ".md"] != members[evidence_prefix + "/report.md"]
        ):
            fail("Study primary copies differ from immutable public evidence")
        if (
            len(report["tasks"]) != 10
            or len(report["runs"]) != 60
            or report.get("configuration_labels") != LABELS
        ):
            fail("Study primary scope or treatment labels differ")
        for phase in ("before", "after"):
            source = report["metadata"]["memglide_source_" + phase]
            if (
                source["git_head"] != study["head"]
                or source["source_tree_sha256"] != study["app"]
                or source.get("source_dirty_entries")
            ):
                fail("Study App source identity differs")
        frozen_names = (
            set(report["metadata"]["frozen_manifest"]["files"])
            | set(report["metadata"]["memglide_source_before"]["source_files"])
            | set(REQUIRED_TESTS)
            | {
                "pyproject.toml",
                "uv.lock",
                "frontend/package.json",
                "frontend/package-lock.json",
                "benchmarks/__init__.py",
                "LICENSE",
            }
        )
        if label == "confirmation":
            test_paths = (
                subprocess.check_output(
                    ["git", "ls-tree", "-r", "--name-only", study["head"], "--", "tests"],
                    cwd=ROOT,
                    stderr=subprocess.PIPE,
                )
                .decode()
                .splitlines()
            )
            frozen_names.update(
                name for name in test_paths if name.endswith("_v2.py") and "workflow" in name
            )
        for relative in sorted(frozen_names):
            safe_name(relative)
            data = subprocess.check_output(
                ["git", "show", study["head"] + ":" + relative], cwd=ROOT, stderr=subprocess.PIPE
            )
            add(
                "frozen-source/" + label + "/" + relative + ".txt",
                data,
                {
                    "kind": "frozen_git_source_text",
                    "study": label,
                    "commit": study["head"],
                    "path": relative,
                },
            )
        provenance = assessment["provenance"]
        source_prefix = "frozen-source/" + label + "/"
        if (
            provenance.get("primary_report_sha256") != sha(report_bytes)
            or provenance.get("pricing_document_sha256")
            != sha(members["benchmarks/cost-pricing.json"])
            or provenance.get("workflow_estimator_sha256")
            != sha(members[source_prefix + "benchmarks/workflow_cost.py.txt"])
            or provenance.get("shared_estimator_sha256")
            != sha(members[source_prefix + "benchmarks/estimate_cost.py.txt"])
        ):
            fail("Study accounting does not bind its frozen estimators/report/pricing")
        if (
            assessment.get("actual_billed_cost_usd") is not None
            or assessment.get("unique_attempts") != 60
        ):
            fail("Unavailable study billing or original attempt count differs")
        for row in assessment["runs"]:
            if row["cost"].get("actual_billed_cost_usd") is not None:
                fail("Unavailable attempt billing was replaced")
            native_name = evidence_prefix + "/runs/" + safe_name(row["run_id"]) + "/events.jsonl"
            if row.get("events_sha256") is not None and (
                native_name not in members or row["events_sha256"] != sha(members[native_name])
            ):
                fail("Accounting row native event identity differs")
        independent = independent_proofs[label]
        audit_member = "benchmarks/evidence/efficiency-v1/" + independent["published_path"]
        independent_bytes = members[audit_member]
        audit = unique_json(independent_bytes)
        verification = audit.get("assessment_verification", {})
        if (
            independent.get("pass") is not True
            or sha(independent_bytes) != independent["sha256"]
            or audit.get("pass") is not True
            or audit.get("errors")
            or verification.get("pass") is not True
            or verification.get("errors")
            or verification.get("assessment_sha256") != sha(assessment_bytes)
            or audit.get("source_report_sha256") != sha(report_bytes)
            or audit.get("pricing_sha256") != provenance["pricing_document_sha256"]
            or audit.get("expected_frozen_head") != study["head"]
            or audit.get("expected_app_sha256") != study["app"]
        ):
            fail("Final independent study audit does not bind selected final bytes")
        if report["metadata"].get("source_unchanged_during_experiment") is not True:
            fail("Per-study source was not retained during measurements")
    campaign_document_bytes = members["benchmarks/efficiency-assessment.json"]
    campaign_document = unique_json(campaign_document_bytes)
    campaign_proof = manifest["campaign_audit"]
    campaign_audit_bytes = members[
        "benchmarks/evidence/efficiency-v1/" + campaign_proof["published_path"]
    ]
    campaign_audit = unique_json(campaign_audit_bytes)
    if (
        campaign_proof.get("pass") is not True
        or sha(campaign_audit_bytes) != campaign_proof["sha256"]
        or campaign_audit.get("pass") is not True
        or campaign_audit.get("errors")
        or campaign_audit.get("campaign_assessment_sha256") != sha(campaign_document_bytes)
        or campaign_audit.get("planned_originals") != 120
        or campaign_audit.get("originals_per_arm") != 40
    ):
        fail("Final campaign audit does not bind all 120 attempts and final narrative")
    for study in study_specs:
        binding = campaign_audit.get("studies", {}).get(study["label"], {})
        if (
            binding.get("source_report_sha256")
            != independent_proofs[study["label"]]["source_report_sha256"]
            or binding.get("assessment_sha256")
            != independent_proofs[study["label"]]["assessment_verification"]["assessment_sha256"]
        ):
            fail("Campaign accounting does not bind both separate study outputs")
    start = """# Memglide efficiency campaign: complete reports and evidence

Start with [the campaign assessment](benchmarks/efficiency-assessment.md), [the fresh confirmation assessment](benchmarks/workflow-confirmation-assessment.md), [the original study](benchmarks/workflow-assessment.md), [the protocol-contract issue](benchmarks/workflow-contract-issue.md), [validation](benchmarks/efficiency-validation.md), [research](benchmarks/efficiency-research.md) and [the architecture](docs/efficiency-architecture.md).

This archive retains two separately frozen studies and all 120 original invocations. Each study uses ten fresh Python/TypeScript tasks, two repeats and three matched configurations: A standard Codex, B complete caller-selected production source, C that source plus complete caller-selected supporting tests. The original study's60 strict failures remain failures: models appended regression tests to byte-protected existing test files under an ambiguous contract. Their passing behavioral/smoke/type checks do not override the protected-file grading rule. The new corpus and explicit instruction contract form a fresh confirmation study, not a replacement or retrospective regrade. Read both protocols and all outcomes. Efficacy, latency and source-benefit comparisons are reported separately by study.

Campaign accounting retains spend from all 120 attempts, including every initial failure, with40original attempts per arm. It is an operational cost-per-success calculation, not pooled causal efficacy across different task corpora/protocols. All dollar figures are API-list-price equivalents, not actual provider bills or subscription charges. Missing usage remains unknown. Command items/file-change items are proxies, not provider model calls or actual filesystem reads. No auxiliary retrieval model or native MCP server is configured in these source-preload studies. Long-term memory and large-repository benefits are outside this experiment.

The prespecified support-promotion gate excludes supplied-context controls and requires focused cost and full-time means AND medians to match or improve both other arms overall and separately in each repeat, with complete matched correctness/fidelity. All gate checks and failures remain visible. A finite-corpus acceptance rule does not establish universal improvements.

All native streams, common prompts, commands, preparation attestations, grading output, patches and durable attempt/result seals are under `benchmarks/evidence/efficiency-v1/formal/` and `confirmation/`. Auth, native state, databases, fixture repositories/worktrees, raw browser/environment configuration, arbitrary full external research bodies and untracked file contents are excluded. Engineering receipts include initial failures, interrupted validation and final verification. No original outcome is replaced or hidden.

Each study's frozen application and transitive benchmark/corpus/estimator/protocol source is reference text under `frozen-source/formal/` or `frozen-source/confirmation/`. Exact original paths and independent Git commits appear in `MEMBER-MANIFEST.json`. This is not a full executable source checkout. To recompute one study's accounting without model requests, restore that study's `frozen-source/STUDY/benchmarks/*.py.txt` to matching `benchmarks/*.py` paths, then use Python 3.11 or newer from the extracted root and choose a fresh output prefix:

```sh
python -m benchmarks.workflow_cost --report benchmarks/codex-workflow-ab.json --pricing benchmarks/cost-pricing.json --events-root benchmarks/evidence/efficiency-v1/formal/runs --output-prefix NEW_ORIGINAL_PREFIX
python -m benchmarks.workflow_cost --report benchmarks/codex-workflow-confirmation.json --pricing benchmarks/cost-pricing.json --events-root benchmarks/evidence/efficiency-v1/confirmation/runs --output-prefix NEW_CONFIRMATION_PREFIX
```

Creation timestamps can differ. Compare numerical fields, gate checks and bound input hashes. Pairwise views reference the same attempts and never create additional spend. [LICENSE](LICENSE) accompanies the source excerpts.

`MEMBER-MANIFEST.json` lists every other member's exact size, SHA-256 and origin. The external `benchmarks/efficiency-report-bundle.json` binds that inventory, ZIP bytes, CRC and each member hash, avoiding circular self-hashes. All selected original text bytes are credential-pattern scanned before packaging. Narrow synthetic pytest XML exceptions require exact original hashes and matching public frozen Git literals inside approved testcase-name spans; every other XML byte remains scanned and original bytes are preserved. Pattern scanning cannot prove absence of every possible secret. Historical reports/evidence/archives were compared byte-for-byte to pre-iteration Git blobs and remain unchanged.
""".encode()
    add("START-HERE.md", start, {"kind": "generated_final_guide"})
    add(
        "reproduction/build_bundle.py.txt",
        read(builder_path),
        {"kind": "explicit_builder_source", "sha256": args.expected_builder_sha256},
    )
    inventory = {
        "schema_version": 1,
        "scope": "Two separately frozen ten-task/two-repeat studies; all 120 originals and failed spend retained; no actual billing claim or efficacy pooling",
        "study_source_freezes": manifest["source_freeze"],
        "published_evidence_manifest_sha256": args.manifest_sha256,
        "pinned_scanner_sha256": args.scanner_sha256,
        "self_excluded_from_members": "MEMBER-MANIFEST.json; external receipt records its hash",
        "synthetic_fixture_scan_exceptions": exceptions,
        "files": [
            {"path": name, "bytes": len(data), "sha256": sha(data), "origin": origins[name]}
            for name, data in sorted(members.items())
        ],
    }
    inventory_bytes = encoded(inventory)
    add("MEMBER-MANIFEST.json", inventory_bytes, {"kind": "generated_member_inventory"})
    for path, identity in identities.items():
        reject_symlinks(path)
        data = path.read_bytes()
        if (len(data), sha(data)) != identity:
            fail("Selected bytes changed during packaging")
    if not args.publish:
        print(
            json.dumps(
                {
                    "mode": "check_only",
                    "archive_member_count": len(members),
                    "uncompressed_bytes": sum(map(len, members.values())),
                    "all_selected_bytes_scanned": True,
                    "member_manifest_sha256": sha(inventory_bytes),
                    "builder_sha256": args.expected_builder_sha256,
                }
            )
        )
        return
    with zipfile.ZipFile(OUTPUT, "x", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for name, data in sorted(members.items()):
            info = zipfile.ZipInfo(name, date_time=(2026, 10, 19, 0, 0, 0))
            info.compress_type = zipfile.ZIP_DEFLATED
            info.external_attr = 0o100644 << 16
            archive.writestr(info, data, compress_type=zipfile.ZIP_DEFLATED, compresslevel=9)
    with zipfile.ZipFile(OUTPUT) as archive:
        if (
            archive.testzip() is not None
            or len(archive.namelist()) != len(members)
            or set(archive.namelist()) != set(members)
        ):
            fail("ZIP CRC, duplicate or exact-member verification failed")
        for name, expected_bytes in members.items():
            actual_bytes = archive.read(name)
            if len(actual_bytes) != len(expected_bytes) or sha(actual_bytes) != sha(expected_bytes):
                fail("ZIP member size/SHA verification failed")
    data = OUTPUT.read_bytes()
    receipt = {
