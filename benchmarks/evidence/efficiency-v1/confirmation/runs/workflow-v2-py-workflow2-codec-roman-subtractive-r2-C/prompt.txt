```python
def roman_numeral(value):
    symbols=((1000,"M"),(500,"D"),(100,"C"),(50,"L"),(10,"X"),(5,"V"),(1,"I"))
    parts=[]
    for amount,symbol in symbols:
        count,value=divmod(value,amount)
        parts.append(symbol*count)
    return "".join(parts)
```
