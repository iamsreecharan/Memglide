"""Count secure source reads while selecting memory; no model requests."""

from __future__ import annotations

import hashlib
import json
import tempfile
from pathlib import Path
from unittest.mock import patch

import memglide.memory.engine as memory_module
from memglide.service import Service


with tempfile.TemporaryDirectory(prefix="memglide-memory-selection-") as directory:
    root = Path(directory)
    repo = root / "repository"
    repo.mkdir()
    service = Service(root / "state" / "state.db")
    try:
        service.db.execute(
            "INSERT INTO projects(id,path,name,created_at) VALUES('project',?,'fixture',0)",
            (str(repo),),
        )
        for name in ["relevant.py", "shared.py", *(f"other{i}.py" for i in range(10))]:
            (repo / name).write_text("def fixture():\n    return True\n")
        session = service.memory.start_session("project", "verify")
        session_id = session["id"]
        saved = []
        for i in range(40):
            path = "shared.py" if i < 30 else f"other{i - 30}.py"
            saved.append(
                service.memory.save_memory(
                    session_id,
                    "observation",
                    f"Migration planning observation {i}",
                    "inspected fixture",
                    [{"path": path}],
                )
            )
        for i in range(15):
            saved.append(
                service.memory.save_memory(
                    session_id,
                    "verified_fix" if i % 2 else "observation",
                    f"Verify behavior observation {i}",
                    "inspected fixture",
                    [{"path": "relevant.py"}],
                )
            )
        for i, memory in enumerate(saved):
            service.db.execute(
                "UPDATE memories SET created_at=?,updated_at=?,stale=? WHERE id=?",
                (1000 + i, 1000 + i, int(i in {40, 41, 42}), memory["id"]),
            )
        reads = []
        original_read = memory_module.read_source

        def counted_read(repository: str, path: str, max_size: int):
            reads.append(path)
            return original_read(repository, path, max_size)

        with patch.object(memory_module, "read_source", counted_read):
            selected = service.context._select_memories(session_id, "verify", "verify")
        projection = [{key: value for key, value in memory.items() if key != "id"} for memory in selected]
        encoded = json.dumps(projection, sort_keys=True, ensure_ascii=False).encode()
        print(
            json.dumps(
                {
                    "saved_memories": len(saved),
                    "irrelevant_memories": 40,
                    "potentially_relevant_memories": 15,
                    "selected_memories": len(selected),
                    "secure_source_read_invocations": len(reads),
                    "unique_source_paths_read": sorted(set(reads)),
                    "selected_projection_sha256": hashlib.sha256(encoded).hexdigest(),
                    "selected_projection": projection,
                    "measurement_scope": "Application read_source calls, not operating-system reads or model token/cost telemetry",
                },
                indent=2,
            )
        )
    finally:
        service.close()
