from pathlib import Path
from datetime import datetime, timezone
from concurrent.futures import ThreadPoolExecutor
import hashlib, json, urllib.request, urllib.error
ROOT=Path('.local/efficiency-v1/research')
COMMIT='ff6aec96948b70d94983af2641a6b67c94faeff5'
SOURCES={
 'openai-prompt-caching':'https://developers.openai.com/api/docs/guides/prompt-caching',
 'openai-latency':'https://developers.openai.com/api/docs/guides/latency-optimization',
 'openai-cost-optimization':'https://developers.openai.com/api/docs/guides/cost-optimization',
 'openai-conversation-state':'https://developers.openai.com/api/docs/guides/conversation-state',
 'anthropic-context':'https://www.anthropic.com/engineering/effective-context-engineering-for-ai-agents',
 'anthropic-tools':'https://www.anthropic.com/engineering/writing-tools-for-agents',
 'codex-exec-lib':f'https://raw.githubusercontent.com/openai/codex/{COMMIT}/codex-rs/exec/src/lib.rs',
 'codex-exec-cli':f'https://raw.githubusercontent.com/openai/codex/{COMMIT}/codex-rs/exec/src/cli.rs',
 'codex-compact':f'https://raw.githubusercontent.com/openai/codex/{COMMIT}/codex-rs/core/src/compact.rs',
 'codex-compact-remote':f'https://raw.githubusercontent.com/openai/codex/{COMMIT}/codex-rs/core/src/compact_remote.rs',
 'codex-client':f'https://raw.githubusercontent.com/openai/codex/{COMMIT}/codex-rs/core/src/client.rs',
}
def fetch(item):
 name,url=item
 record={'id':name,'url':url,'retrieved_at_utc':datetime.now(timezone.utc).isoformat(),'http_status':None}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Memglide-research/1.0'}),timeout=45) as response:
   body=response.read(12_000_001)
   if len(body)>12_000_000: raise ValueError('response exceeds research cap')
   record.update(http_status=response.status,final_url=response.geturl(),body_bytes=len(body),body_sha256=hashlib.sha256(body).hexdigest())
   (ROOT/(name+'.source')).write_bytes(body)
 except urllib.error.HTTPError as exc:
  body=exc.read(2000)
  record.update(http_status=exc.code,error_type=type(exc).__name__,error_body_bytes=len(body),error_body_sha256=hashlib.sha256(body).hexdigest())
 except Exception as exc:
  record.update(error_type=type(exc).__name__,error=str(exc))
 (ROOT/(name+'.fetch.json')).write_text(json.dumps(record,indent=2)+'\n')
 return record
with ThreadPoolExecutor(max_workers=5) as executor:
 records=list(executor.map(fetch,SOURCES.items()))
(ROOT/'fetch-log.json').write_text(json.dumps(records,indent=2)+'\n')
print(json.dumps([{k:v for k,v in r.items() if k in ('id','http_status','body_bytes','error_type')} for r in records],indent=2))
