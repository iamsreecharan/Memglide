#!/usr/bin/env python3
"""Audit context selection against a frozen baseline using synthetic memory views.

This verifier executes the real old/new selector methods and query tokenizer, but
supplies precomputed memory freshness views. Its call counts are simulated
_memory_view invocations, never measured source reads or application benchmarks.
It does not exercise filesystem freshness, session lifecycle, or publication.
Only the ignored sibling JSON audit artifact is written.
"""

from __future__ import annotations

import ast
import copy
import hashlib
import json
import random
import runpy
import subprocess
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


BASELINE_COMMIT = "448f70555698b56f2f7b257a84628106846fb55e"
SEED = 4310
FIXTURE_COUNT = 1200
ROOT = Path(__file__).resolve().parents[2]
CONTEXT_PATH = "src/memglide/core/context.py"
ENGINE_PATH = "src/memglide/memory/engine.py"
QUERY_PATH = "src/memglide/core/query.py"
RESULT_PATH = Path(__file__).with_suffix(".json")


def sha256(value: str) -> str:
    return hashlib.sha256(value.encode("utf-8")).hexdigest()


def baseline_source(path: str) -> str:
    return subprocess.check_output(
        ["git", "show", f"{BASELINE_COMMIT}:{path}"], cwd=ROOT, text=True
    )


def method(source: str, class_name: str, method_name: str) -> ast.FunctionDef:
    tree = ast.parse(source)
    cls = next(
        node for node in tree.body if isinstance(node, ast.ClassDef) and node.name == class_name
    )
    return next(
        node
        for node in cls.body
        if isinstance(node, ast.FunctionDef) and node.name == method_name
    )


def canonical_ast(node: ast.AST) -> str:
    return ast.dump(node, include_attributes=False)


def selector(source: str) -> Any:
    selected = copy.deepcopy(method(source, "ContextEngine", "_select_memories"))
    selected.name = "select"
    namespace: dict[str, Any] = {"Any": Any}
    module = ast.fix_missing_locations(ast.Module(body=[selected], type_ignores=[]))
    exec(compile(module, "<extracted-selector>", "exec"), namespace)
    return namespace["select"]


class FakeMemory:
    """Supply synthetic views while counting simulated view invocations."""

    def __init__(self, rows: list[dict[str, Any]], views: dict[str, dict[str, Any]]):
        self.rows = rows
        self.views = views
        self.simulated_view_calls = 0

    def view(self, row: dict[str, Any]) -> dict[str, Any]:
        self.simulated_view_calls += 1
        return copy.deepcopy(self.views[row["id"]])

    def recall_memory(self, session_id: str) -> list[dict[str, Any]]:
        return [self.view(row) for row in self.rows]

    def _recall_context_candidates(self, session_id: str, include: Any) -> list[dict[str, Any]]:
        return [self.view(row) for row in self.rows if include(row)]


def main() -> None:
    baseline_context = baseline_source(CONTEXT_PATH)
    current_context = (ROOT / CONTEXT_PATH).read_text()
    baseline_engine = baseline_source(ENGINE_PATH)
    current_engine = (ROOT / ENGINE_PATH).read_text()
    query_source = (ROOT / QUERY_PATH).read_text()
    old_select, new_select = selector(baseline_context), selector(current_context)
    tokens = runpy.run_path(str(ROOT / QUERY_PATH))["tokens"]

    class Context:
        _words = staticmethod(lambda value: set(tokens(value)))

        def __init__(self, memory: FakeMemory):
            self.memory = memory

    rng = random.Random(SEED)
    categories = [
        "observation", "error", "failed_attempt", "verified_fix", "open_issue", "decision",
        "checkpoint",
    ]
    contents = [
        "HTTPStatus caches", "status cache", "only provenance", "authenticate_user tokens",
        "completely unrelated", "the and a", "μικρός Buses", "cache timeout error",
        "attempt previous",
    ]
    queries = [
        "HTTPStatuses", "cache", "authenticateUser", "missing words", "the and a", "",
        "buses", "previous attempts", "tokENS unrelated",
    ]
    old_calls = new_calls = candidate_rows = selected_rows = 0
    larger_cases = empty_term_cases = prior_attempt_cases = 0
    for case in range(FIXTURE_COUNT):
        rows: list[dict[str, Any]] = []
        views: dict[str, dict[str, Any]] = {}
        for index in range(rng.randint(0, 70)):
            raw = {
                "id": f"m-{index:03}",
                "session_id": "session",
                "category": rng.choice(categories),
                "content": rng.choice(contents),
                "provenance": "cache authenticate_user",
                "stale": rng.choice([False, False, True]),
                "updated_at": rng.choice([1, 2, 3]),
                "version": "v",
                "sources": [],
            }
            view = {**raw, "stale": raw["stale"] or rng.choice([False, False, True])}
            view["freshness"] = (
                "Stale" if view["stale"] else rng.choice(["Current", "Requires inspection"])
            )
            rows.append(raw)
            views[raw["id"]] = view
        rng.shuffle(rows)
        query = rng.choice(queries)
        task = rng.choice(queries)
        intent = rng.choice([None, "prior_attempts", "session_changes"])
        old_memory, new_memory = FakeMemory(rows, views), FakeMemory(rows, views)
        original = old_select(Context(old_memory), "session", query, task, intent)
        current = new_select(Context(new_memory), "session", query, task, intent)
        assert original == current, (case, query, task, intent)
        old_calls += old_memory.simulated_view_calls
        new_calls += new_memory.simulated_view_calls
        candidate_rows += len(rows)
        selected_rows += len(current)
        larger_cases += len(rows) > 10
        empty_term_cases += not bool(tokens(query + " " + task))
        prior_attempt_cases += intent == "prior_attempts"

    old_recall_ast = canonical_ast(method(baseline_engine, "MemoryEngine", "recall_memory"))
    new_recall_ast = canonical_ast(method(current_engine, "MemoryEngine", "recall_memory"))
    assert old_recall_ast == new_recall_ast, "Public recall_memory AST changed"
    result = {
        "status": "passed",
        "executed_at_utc": datetime.now(timezone.utc).isoformat(),
        "baseline_commit": BASELINE_COMMIT,
        "seed": SEED,
        "fixtures": {
            "total": FIXTURE_COUNT,
            "identical_selection_outputs": FIXTURE_COUNT,
            "candidate_rows_total": candidate_rows,
            "selected_rows_total": selected_rows,
            "more_than_ten_candidates": larger_cases,
            "empty_normalized_term_cases": empty_term_cases,
            "prior_attempt_intent_cases": prior_attempt_cases,
            "candidate_count_range": [0, 70],
        },
        "simulated_memory_view_calls": {"baseline": old_calls, "prefiltered": new_calls},
        "actual_source_reads_measured": False,
        "public_recall_ast_equal": old_recall_ast == new_recall_ast,
        "source_sha256": {
            CONTEXT_PATH: {"baseline": sha256(baseline_context), "current": sha256(current_context)},
            ENGINE_PATH: {"baseline": sha256(baseline_engine), "current": sha256(current_engine)},
            QUERY_PATH: {"current": sha256(query_source)},
        },
        "method_ast_sha256": {
            "ContextEngine._select_memories": {
                "baseline": sha256(canonical_ast(method(baseline_context, "ContextEngine", "_select_memories"))),
                "current": sha256(canonical_ast(method(current_context, "ContextEngine", "_select_memories"))),
            },
            "MemoryEngine.recall_memory": {
                "baseline": sha256(old_recall_ast), "current": sha256(new_recall_ast),
            },
        },
        "limitations": [
            "Freshness views are precomputed synthetic fixtures; view-call counts are simulated.",
            "No real source freshness reads, lifecycle operations, or publication are exercised.",
            "The current real tokenizer is shared by both extracted selector methods.",
            "No performance, token-use, or total-system quality claim follows from these fixtures.",
        ],
    }
    RESULT_PATH.write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(json.dumps({
        "status": result["status"],
        "fixtures": result["fixtures"],
        "simulated_memory_view_calls": result["simulated_memory_view_calls"],
        "public_recall_ast_equal": result["public_recall_ast_equal"],
        "result_path": str(RESULT_PATH),
    }, indent=2))


if __name__ == "__main__":
    main()
