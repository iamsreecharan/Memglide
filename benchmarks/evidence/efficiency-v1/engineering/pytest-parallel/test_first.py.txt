import os,time
from pathlib import Path
def test_first(tmp_path):
 marker=tmp_path/'alive';marker.write_text('present')
 Path(os.environ['PROOF_READY']).write_text(str(tmp_path))
 deadline=time.monotonic()+15
 while not Path(os.environ['PROOF_FINISH']).exists() and time.monotonic()<deadline:time.sleep(.02)
 assert marker.exists(), 'another pytest process removed the active fixture directory'
