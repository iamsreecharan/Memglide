KEY='native_reported_price_equivalent_usd'

def num(v,d=4): return f'{v:,.{d}f}' if type(v) in (int,float) and math.isfinite(v) else 'unavailable'
def change(base,candidate):
    if type(base) not in (int,float) or type(candidate) not in (int,float) or base<=0: return 'unavailable'
    pct=(candidate/base-1)*100
    return f'{abs(pct):.2f}% '+('higher' if pct>0 else 'lower' if pct<0 else 'unchanged')
def table(summaries):
    lines=['| Arm | Attempts | Strict successes | Estimated USD/success | Mean full seconds | Median full seconds | Mean uncached input | Mean output |','|---|---:|---:|---:|---:|---:|---:|---:|']
    for arm,s in summaries.items():
        m=s['metrics']; lines.append(f"| {arm} | {s['attempts']} | {s['successful_assignments']} | {num(s['cost_per_successful_task_usd'][KEY],7)} | {num(m['end_to_end_wall_seconds']['mean'],3)} | {num(m['end_to_end_wall_seconds']['median'],3)} | {num(m['uncached_input_tokens']['mean'],1)} | {num(m['output_tokens']['mean'],1)} |")
    return lines
