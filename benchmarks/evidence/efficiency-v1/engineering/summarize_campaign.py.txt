"""Derive all-attempt campaign accounting from two sealed, separately reported studies."""
from pathlib import Path
from decimal import Decimal
from datetime import datetime,timezone
import hashlib,json,math

ROOT=Path(__file__).resolve().parents[2]
KEYS=('native_reported_price_equivalent_usd','possible_write_price_lower_usd','possible_write_price_upper_usd')
SPECS={'formal':('codex-workflow-ab','workflow-assessment','93acf63e1583bd66239b22dc53c2aaf196e3a35a'),'confirmation':('codex-workflow-confirmation','workflow-confirmation-assessment','85ebff2229efd62be2e3c0e179c3c8eae0f0cade')}

def number(v): return type(v) in (int,float) and math.isfinite(v) and v>=0

def main():
    base=ROOT/'.local/efficiency-v1'
    for receipt in ('formal-exit.json','confirmation-exit.json'):
        v=json.loads((base/receipt).read_text()); assert v['returncode']==0 and v['finished_unix']>=v['started_unix']
    target=ROOT/'benchmarks/efficiency-assessment.json'; assert not target.exists()
    studies={}; rows=[]; inputs={}
    for study,(primary,cost,head) in SPECS.items():
        p=ROOT/f'benchmarks/{primary}.json'; c=ROOT/f'benchmarks/{cost}.json'
        r=json.loads(p.read_text()); a=json.loads(c.read_text()); assert len(r['runs'])==len(a['runs'])==60
        assert r['metadata']['frozen_manifest']['git_head']==head and r['metadata']['source_unchanged_during_experiment'] is True
        assert a['provenance']['primary_report_sha256']==hashlib.sha256(p.read_bytes()).hexdigest()
        successes=sum(x['success'] is True for x in a['runs'])
        studies[study]={'source_report_sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'assessment_sha256':hashlib.sha256(c.read_bytes()).hexdigest(),'frozen_git_head':head,'attempts':60,'successful_assignments':successes,'failed_assignments':60-successes,'primary_report':str(p.relative_to(ROOT)),'cost_assessment':str(c.relative_to(ROOT)),'promotion_gate_passed':a['promotion_gate']['passed']}
        inputs[study]=a
        rows.extend({'study':study,'run_id':x['run_id'],'configuration':x['configuration'],'success':x['success'],'cost':{k:x['cost'][k] for k in KEYS}} for x in a['runs'])
    assert len(rows)==len({x['run_id'] for x in rows})==120
    assert studies['formal']['successful_assignments']==0
    summaries={}
    for arm in ('A','B','C'):
        selected=[x for x in rows if x['configuration']==arm]; assert len(selected)==40
        successes=sum(x['success'] is True for x in selected)
        totals={k:float(sum(Decimal(str(x['cost'][k])) for x in selected)) if all(number(x['cost'][k]) for x in selected) else None for k in KEYS}
        summaries[arm]={'attempts':40,'successful_assignments':successes,'failed_assignments':40-successes,'unknown_cost_assignments':sum(not number(x['cost'][KEYS[0]]) for x in selected),'cost_total_usd':totals,'cost_per_successful_task_usd':{k:float(Decimal(str(v))/successes) if v is not None and successes else None for k,v in totals.items()}}
    gate=inputs['confirmation']['promotion_gate']
    value={'assessment_version':'efficiency-campaign-1','generated_at_utc':datetime.now(timezone.utc).isoformat(),'scope':'Two separately frozen fresh-task studies. Campaign accounting includes every attempt and failure; different task corpora are not pooled for efficacy, token or latency conclusions.','actual_billed_cost_usd':None,'studies':studies,'campaign':{'original_assignments':120,'originals_per_arm':40,'configurations':summaries,'runs':rows},'confirmation_promotion_gate':gate,'support_default':'Explicitly optional; no automatic promotion','implementation':{'application_source_tree_sha256':'43032dcabab006ec346b77d2032da9b7016adfea2ff013cf414d25f67f61f176','verified_automated_cases':1443,'actual_local_memory_reader_calls_before':55,'actual_local_memory_reader_calls_after':15,'synthetic_selection_equivalence_cases':1200,'component_measurement_is_model_cost_saving':False},'limitations':['All dollar amounts are cited standard API list-price equivalents, not actual charges or subscription savings.','Native-reported zero cache writes can reflect upstream omission; possible-write bounds are scenarios, not confidence intervals.','Original60 protected-file failures and all their attempted spend remain included; none was regraded.','Only the fresh confirmation study tests the clarified contract; its outcomes must remain separate from the initial corpus.','Workloads are small purpose-built repositories; these results do not establish production or cross-session-memory gains.','All native arms use zero configured MCP calls. Real MCP functionality passed separate installed-package checks; model efficiency through MCP is not measured.','Provider model-call counts, actual filesystem read counts and model-server resources remain unknown. Read-command and patch counts are proxies.','Local CPU is a measured lower bound; child RSS excludes parent indexing and can double-count shared pages. No local-dollar conversion is invented.']}
    target.write_text(json.dumps(value,indent=2)+'\n')
    print(json.dumps({'attempts':120,'strict_successes':sum(x['success'] is True for x in rows),'confirmation_gate_passed':gate['passed'],'campaign_per_arm':summaries}))

if __name__=='__main__': main()
