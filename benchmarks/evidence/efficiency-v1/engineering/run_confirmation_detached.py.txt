from pathlib import Path
import json
import sys
import time
import traceback

root = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(root))
from benchmarks.run_workflow_v2 import main

receipt = root / '.local/efficiency-v1/confirmation-exit.json'
if receipt.exists():
    raise SystemExit('Refusing to overwrite a formal exit receipt')
started = time.time()
try:
    code = main(sys.argv[1:])
except BaseException:
    traceback.print_exc()
    code = 70
value = {'returncode': code, 'started_unix': started, 'finished_unix': time.time()}
temporary = receipt.with_suffix('.tmp')
temporary.write_text(json.dumps(value, indent=2) + '\n')
temporary.replace(receipt)
raise SystemExit(code)
