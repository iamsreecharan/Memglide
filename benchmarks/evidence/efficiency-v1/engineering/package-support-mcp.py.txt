CHECKS: dict[str, bool] = {}
sha = lambda value: hashlib.sha256(value).hexdigest()
def check(label: str, condition: bool) -> None:
    CHECKS[label] = bool(condition)
    assert condition, label

check('isolated_installed_imports', bool(sys.flags.isolated) and Path(memglide.__file__).resolve().is_relative_to(Path(sys.prefix).resolve()))
versions = {name: importlib.metadata.version(name) for name in ['memglide', 'fastapi', 'uvicorn', 'pydantic', 'mcp', 'tree-sitter', 'tree-sitter-javascript', 'tree-sitter-typescript', 'watchdog', 'pathspec', 'typer', 'rich']}
check('exact_direct_dependency_versions', versions == {'memglide':'0.1.0','fastapi':'0.143.2','uvicorn':'0.42.0','pydantic':'2.12.5','mcp':'2.3.0','tree-sitter':'0.25.2','tree-sitter-javascript':'0.25.0','tree-sitter-typescript':'0.23.2','watchdog':'6.0.0','pathspec':'1.0.4','typer':'0.24.1','rich':'14.3.3'})

def bodies(result: dict, prompt: str) -> dict[str, str]:
    if result['context_format'] == 'compact-prefix':
        suffix = TASK_SEPARATOR + prompt
        assert result['prompt'].endswith(suffix)
        envelope = json.loads(result['prompt'][:-len(suffix)])
    else:
        prefix = prompt + '\n\n' + _EXPLANATION
        assert result['prompt'].startswith(prefix)
        envelope = json.loads(result['prompt'][len(prefix):])
    actual = {row['path']: row['content'] for row in envelope['memglide_sources']}
    for metadata in result['files']:
        assert metadata['sha256'] == sha(actual[metadata['path']].encode())
    return actual

native_observations = []
def native_case(label: str, db: Path, project: str, prompt: str, expected: dict, format_: str) -> None:
    folder = STATE / label
    folder.mkdir(mode=0o700)
    fake, capture, marker, report = [folder / name for name in ['codex', 'stdin.bin', 'once.txt', 'report.json']]
    fake.write_text(f'#!{sys.executable}\nfrom pathlib import Path\nimport sys\nPath({str(marker)!r}).open("x").write("one execution\\n")\nPath({str(capture)!r}).write_bytes(sys.stdin.buffer.read())\nraise SystemExit(31)\n')
    fake.chmod(0o700)
    argv = [str(Path(sys.executable).parent/'memglide-codex'), '--codex-executable', str(fake), '--project', project, '--db', str(db), '--focus', 'calc.py', '--support', 'tests/test_calc.py', '--preparation-report', str(report), '--context-format', format_, '--', '--json', '--model', 'dummy-no-model-request']
    completed = subprocess.run(argv, input=prompt.encode(), cwd='/tmp', capture_output=True, timeout=30)
    assert completed.returncode == 31, completed.stderr.decode(errors='replace')
    assert not completed.stdout and not completed.stderr
    private = json.loads(report.read_text())
    check(label + '_single_native_exact_payload', marker.read_text() == 'one execution\n' and capture.read_bytes() == expected['prompt'].encode())
    check(label + '_private_v3_classifications', private['version'] == 3 and private['preparation']['focus_paths'] == ['calc.py'] and private['preparation']['support_paths'] == ['tests/test_calc.py'])
    check(label + '_private_mode_hash_and_no_source', private['preparation']['mode'] == expected['mode'] and private['preparation']['files'] == expected['files'] and private['native_stdin']['sha256'] == sha(capture.read_bytes()) and 'return value *' not in report.read_text() and report.stat().st_mode & 0o777 == 0o600)
    native_observations.append({'case':label,'report_version':private['version'],'mode':private['preparation']['mode'],'reason':private['preparation']['reason'],'native_execs':1,'model_requests':0})

async def check_mcp(repo: Path, db: Path, project: str) -> dict:
    canary = STATE/'hostile-cwd-imported'
    hostile = repo/'memglide'
    hostile.mkdir()
    (hostile/'__init__.py').write_text(f'from pathlib import Path\nPath({str(canary)!r}).write_text("executed")\nraise RuntimeError("repository shadow imported")\n')
    async def connected(profile: str):
        arguments = ['-I', '-m', 'memglide.cli', '--db', str(db), 'mcp']
        if profile == 'agent':
            arguments += ['--profile', 'agent', '--project', project]
        return StdioServerParameters(command=sys.executable,args=arguments,cwd=str(repo),env={**os.environ,'PYTHONPATH':str(repo)})
    parameters = await connected('default')
    async with stdio_client(parameters) as (read,write), ClientSession(read,write) as client:
        await client.initialize()
        tools = {tool.name for tool in (await client.list_tools()).tools}
        check('real_default_mcp_shared_tools', {'start_session','save_memory','get_context','get_context_passport','end_session'} <= tools)
        started = await client.call_tool('start_session',{'project_id':project,'task':'Review multiply implementation'})
        assert not started.is_error and started.structured_content
        session = started.structured_content['id']
        remembered = await client.call_tool('save_memory',{'session_id':session,'category':'observation','content':'multiply has approved exact source','provenance':'installed wheel smoke','sources':[{'path':'calc.py','hash':sha((repo/'calc.py').read_bytes())}]})
        assert not remembered.is_error, remembered
        recalled = await client.call_tool('get_context',{'session_id':session,'query':'multiply','level':2,'budget':4000})
        assert not recalled.is_error and recalled.structured_content
        payload = recalled.structured_content
        check('real_default_mcp_retrieval_and_memory', any(item['path']=='calc.py' for item in payload['items']) and bool(payload['memories']))
        passport = await client.call_tool('get_context_passport',{'session_id':session,'retrieval_id':payload['retrieval_id']})
        check('real_default_mcp_verified_passport', not passport.is_error and passport.structured_content['verified_source_hashes']['calc.py']==sha((repo/'calc.py').read_bytes()))
        ended = await client.call_tool('end_session',{'session_id':session})
        assert not ended.is_error
        rejected = await client.call_tool('get_context',{'session_id':session,'query':'multiply','level':2,'budget':4000})
        check('real_default_mcp_ended_session_rejected', bool(rejected.is_error))
    parameters = await connected('agent')
    async with stdio_client(parameters) as (read,write), ClientSession(read,write) as client:
        await client.initialize()
        tools = {tool.name for tool in (await client.list_tools()).tools}
        check('real_agent_mcp_exact_two_tools', tools == {'fetch_context','verify_context'})
        fetched = await client.call_tool('fetch_context',{'query':'multiply'})
        assert not fetched.is_error and len(fetched.content)==1 and fetched.structured_content is None
        payload = json.loads(fetched.content[0].text)
        check('real_agent_mcp_source_hash_and_single_payload', payload['source_hashes']['calc.py']==sha((repo/'calc.py').read_bytes()) and any('return value * 3' in item['snippet'] for item in payload['items']))
        source = repo/'calc.py'
        before = source.stat()
        source.write_text(source.read_text().replace('value * 3','value * 4'))
        os.utime(source,ns=(before.st_atime_ns,before.st_mtime_ns))
        verified = await client.call_tool('verify_context',{})
        stale = json.loads(verified.content[0].text)
        check('real_agent_mcp_same_size_restored_mtime_stale', not verified.is_error and stale['status']=='Stale' and 'calc.py' in stale['stale_files'])
        refreshed = await client.call_tool('fetch_context',{'query':'multiply'})
        current = json.loads(refreshed.content[0].text)
        check('real_agent_mcp_refreshed_source', current['source_hashes']['calc.py']==sha(source.read_bytes()) and any('return value * 4' in item['snippet'] for item in current['items']))
        foreign = await client.call_tool('verify_context',{'retrieval_id':'foreign-retrieval'})
        check('real_agent_mcp_foreign_evidence_rejected', bool(foreign.is_error))
    check('real_mcp_hostile_cwd_pythonpath_not_imported', not canary.exists())
    return {'profiles':['default','agent'],'model_requests':0}

with tempfile.TemporaryDirectory(prefix='memglide-efficiency-installed-',dir='/tmp') as temp:
    repo = Path(temp)
    subprocess.run(['git','-c',f'core.hooksPath={os.devnull}','init','-q',str(repo)],check=True,capture_output=True)
    (repo/'tests').mkdir()
    source, support = repo/'calc.py', repo/'tests/test_calc.py'
    source.write_text('def multiply(value):\n    return value * 2\n')
    support.write_text('from calc import multiply\nassert multiply(3) == 6\n')
    untrusted_marker = STATE/'source-was-executed'
    (repo/'never_execute.py').write_text(f'from pathlib import Path\nPath({str(untrusted_marker)!r}).write_text("executed")\n')
    db = STATE/'state.db'
    database = Database(db)
    project = Indexer(database).add_project(str(repo))['id']
    prompt = 'Fix multiply. Preserve exact task bytes λ\r\n$HOME `not-a-shell`\n'
    args = {'project_id':project,'db_path':db,'focus_paths':['calc.py'],'support_paths':['tests/test_calc.py']}
    initial = prepare_prompt(prompt,**args,context_format='compact-prefix')
    check('support_union_complete_and_classified', initial['mode']=='preloaded' and bodies(initial,prompt)=={'calc.py':source.read_text(),'tests/test_calc.py':support.read_text()} and initial['focus_paths']==['calc.py'] and initial['support_paths']==['tests/test_calc.py'])
    for path, old, new in [(source,'value * 2','value * 3'),(support,'== 6','== 9')]:
        before = path.stat()
        path.write_text(path.read_text().replace(old,new))
        os.utime(path,ns=(before.st_atime_ns,before.st_mtime_ns))
        assert path.stat().st_size==before.st_size and path.stat().st_mtime_ns==before.st_mtime_ns
    compact = prepare_prompt(prompt,**args,context_format='compact-prefix')
    check('support_and_focus_fresh_without_reindex', bodies(compact,prompt)=={'calc.py':source.read_text(),'tests/test_calc.py':support.read_text()} and compact['files'] != initial['files'])
    native_case('support-compact',db,project,prompt,compact,'compact-prefix')
    legacy = prepare_prompt(prompt,**args,context_format='legacy')
    check('support_legacy_complete_source', bodies(legacy,prompt)=={'calc.py':source.read_text(),'tests/test_calc.py':support.read_text()})
    native_case('support-legacy',db,project,prompt,legacy,'legacy')
    (repo/'.memglideignore').write_text('tests/\n')
    ignored = prepare_prompt(prompt,**args,context_format='compact-prefix')
    check('current_support_ignore_discards_all_context', ignored['mode']=='passthrough' and ignored['prompt']==prompt and ignored['files']==[] and ignored['reason']=='source_excluded')
    native_case('support-ignored',db,project,prompt,ignored,'compact-prefix')
    (repo/'.memglideignore').unlink()
    for case in ['missing','symlink']:
        support.unlink()
        if case=='symlink':
            support.symlink_to(source)
        declined = prepare_prompt(prompt,**args,context_format='compact-prefix')
        check('support_' + case + '_discards_all_context', declined['mode']=='passthrough' and declined['prompt']==prompt and declined['files']==[] and declined['context_bytes']==0)
        if support.is_symlink():
            support.unlink()
        support.write_text('from calc import multiply\nassert multiply(3) == 9\n')
    small = prepare_prompt(prompt,**args,context_format='compact-prefix',max_context_bytes=100)
    check('combined_envelope_budget_discards_all_context',small['mode']=='passthrough' and small['prompt']==prompt and small['files']==[])
    absent = STATE/'not-created'/'state.db'
    bypass = prepare_prompt(prompt,**{**args,'db_path':absent},context_format='compact-prefix',context_supplied=True)
    check('support_context_supplied_no_state_access',bypass['mode']=='passthrough' and bypass['prompt']==prompt and not absent.parent.exists())
    try:
        prepare_prompt(prompt,**{**args,'db_path':absent,'support_paths':['../outside.py']},mode='off')
    except ValueError:
        check('support_traversal_rejected_before_state_access',not absent.parent.exists())
    else:
        raise AssertionError('traversal accepted')
    check('index_and_source_preparation_never_execute_source',not untrusted_marker.exists())
    mcp = asyncio.run(check_mcp(repo,db,project))

summary = {'passed':True,'functional_check_count':len(CHECKS),'functional_checks':CHECKS,'versions':versions,'native_observations':native_observations,'mcp':mcp,'model_requests':0,'scope':'Installed wheel, isolated imports, dummy native stdin transport and real local MCP; functional checks are not pytest cases or benchmark results.'}
with (BASE/'package-support-mcp-v2.json').open('x') as stream:
    json.dump(summary,stream,indent=2)
    stream.write('\n')
print(json.dumps(summary,indent=2))
