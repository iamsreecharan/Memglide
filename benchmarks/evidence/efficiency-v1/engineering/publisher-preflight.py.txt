"""Ignored no-model publication guard proof; never reads live experiment files."""

from __future__ import annotations

import importlib.util
import json
import sys
import tempfile
from pathlib import Path
from unittest.mock import patch
from xml.sax.saxutils import escape

ROOT = Path(__file__).resolve().parents[2]
STUDY = ROOT / ".local/efficiency-v1"
checks = 0


def check(value):
    global checks
    assert value
    checks += 1


def rejected(function, *args):
    global checks
    try:
        function(*args)
    except SystemExit:
        checks += 1
        return
    raise AssertionError("Expected safe rejection")


def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    result = importlib.util.module_from_spec(spec)
    sys.modules[name] = result
    spec.loader.exec_module(result)
    return result


publisher = module("efficiency_publisher_guard_proof", STUDY / "publish_evidence.py")
builder = module("efficiency_builder_guard_proof", STUDY / "build_bundle.py")
publisher.FROZEN_COMMIT = publisher.XML_FROZEN_COMMIT = "93acf63e1583bd66239b22dc53c2aaf196e3a35a"
publisher.FROZEN_APP = "43032dcabab006ec346b77d2032da9b7016adfea2ff013cf414d25f67f61f176"

# Both study pairs must be completed before any event, report or grade is read.
# These are synthetic mappings; no live experiment path is opened.
for v1_code, v1_finished, v2_code, v2_finished, accepted in [
    (0, "finished", 0, "finished", True),
    (0, "finished", 0, None, False),
    (0, "finished", 1, "finished", False),
    (0, "finished", None, "finished", False),
    (0, None, 0, "finished", False),
    (1, "finished", 0, "finished", False),
]:
    seen = []

    def fake_read(path):
        seen.append(path)
        if path.name == "formal-exit.json":
            return {"returncode": v1_code}
        if path.name == "confirmation-exit.json":
            return {"returncode": v2_code}
        check(path.name == "metadata.json")
        return {
            "finished_at": v1_finished if path.parent.name == "formal-workflow-v1" else v2_finished
        }

    with patch.object(publisher, "read_json", fake_read):
        if accepted:
            publisher.finished_guard()
            check(True)
        else:
            rejected(publisher.finished_guard)
    expected = ["formal-exit.json", "metadata.json"]
    if v1_code == 0 and v1_finished:
        expected += ["confirmation-exit.json", "metadata.json"]
    check([path.name for path in seen] == expected)

with tempfile.TemporaryDirectory(prefix="publisher-proof-", dir=STUDY) as temporary:
    directory = Path(temporary)
    normal = directory / "safe.txt"
    normal.write_text("Explicit synthetic text evidence.\n")
    check(publisher.read_text_bytes(normal) == normal.read_bytes())
    duplicate = directory / "duplicate.json"
    duplicate.write_text('{"x":1,"x":2}')
    rejected(publisher.read_json, duplicate)
    rejected(publisher.safe_source, directory / "private.sqlite")
    alias = directory / "alias.txt"
    alias.symlink_to(normal)
    rejected(publisher.safe_source, alias)
    rejected(builder.reject_symlinks, alias)
    collection = publisher.Collection()
    collection.add(normal, Path("proof/safe.txt"), required=True)
    check(len(collection.files) == 1)
    rejected(collection.add, normal, Path("proof/safe.txt"))
    for value in [
        "../escape.txt",
        "/absolute.txt",
        "private/auth.json",
        "private/native-state/log.txt",
        "private/browser.json",
        "x.zip",
    ]:
        rejected(builder.safe_name, value)
        rejected(collection.add, normal, Path(value))
    for data in [b"not utf8\xff", b"has\0binary"]:
        rejected(publisher.scan_text_bytes, normal, data)

    # Construct only synthetic public fixtures found by the exact frozen test
    # source AST. Never read credentials, environment or browser configuration.
    fixtures, fixture_sha = publisher.frozen_fixture_literals()
    candidates = [
        value
        for value in fixtures
        if sum(len(list(pattern.finditer(value))) for pattern in publisher.SECRET_PATTERNS) == 1
    ]
    check(bool(candidates))
    names = [candidates[index % len(candidates)].decode() for index in range(7)]
    synthetic = (
        "<testsuite>"
        + "".join(
            '<testcase classname="tests.test_security_storage" name="test_detected_credentials_are_excluded['
            + escape(value, {'"': "&quot;"})
            + ']"/>'
            for value in names
        )
        + "</testsuite>"
    ).encode()
    xml = directory / "synthetic.xml"
    xml.write_bytes(synthetic)
    rejected(publisher.scan_text_bytes, xml, synthetic)
    relative = str(xml.relative_to(ROOT))
    publisher.XML_APPROVALS[relative] = publisher.sha(synthetic)
    publisher.scan_text_bytes(xml, synthetic)
    check(len(publisher.XML_SCAN_EXCEPTIONS) == 1)
    record = publisher.XML_SCAN_EXCEPTIONS[relative]
    check(record["pattern_matches"] == 7)
    check(record["fixture_source_sha256"] == fixture_sha)
    check(record["original_xml_bytes_preserved"] is True)
    # Changing bytes, a new comment or a wrong classname cannot inherit an
    # approved testcase-name exception.
    rejected(publisher.scan_text_bytes, xml, synthetic + b" ")
    injected = synthetic.replace(
        b"</testsuite>",
        b"<system-out>" + escape(candidates[0].decode()).encode() + b"</system-out></testsuite>",
    )
    publisher.XML_APPROVALS[relative] = publisher.sha(injected)
    rejected(publisher.scan_text_bytes, xml, injected)
    changed_class = synthetic.replace(b"tests.test_security_storage", b"tests.unrelated_class")
    publisher.XML_APPROVALS[relative] = publisher.sha(changed_class)
    rejected(publisher.scan_text_bytes, xml, changed_class)

    "builder_sha256": publisher.sha((STUDY / "build_bundle.py").read_bytes()),
    "historical_preservation": historical,
    "limitations": [
        "Synthetic two-study guard proof; actual confirmation/public ZIP checks remain pending completion and parent authorization."
    ],
}
output = STUDY / "publisher-preflight-two-study-v1.json"
with output.open("x") as stream:
    stream.write(json.dumps(result, indent=2) + "\n")
print(json.dumps({key: value for key, value in result.items() if key != "historical_preservation"}))
