"""Run new regressions against preserved HEAD methods without editing source."""

from __future__ import annotations

import hashlib
import json
import subprocess
from pathlib import Path

import pytest

import memglide.core.context as context_module
import memglide.core.indexer as indexer_module


provenance = {}
for module, class_name, method_name, path in (
    (context_module, "ContextEngine", "_get_context", "src/memglide/core/context.py"),
    (indexer_module, "Indexer", "_invalidate", "src/memglide/core/indexer.py"),
):
    frozen = subprocess.run(
        ["git", "show", "HEAD:" + path], capture_output=True, check=True
    ).stdout
    provenance[path] = hashlib.sha256(frozen).hexdigest()
    namespace = dict(vars(module))
    exec(compile(frozen, "frozen-HEAD/" + path, "exec"), namespace)
    old_method = getattr(namespace[class_name], method_name)
    if method_name == "_invalidate":
        old_method = staticmethod(old_method)
    setattr(getattr(module, class_name), method_name, old_method)

Path(".local/efficiency-v1/review-head-regression-provenance.json").write_text(
    json.dumps(
        {
            "head": subprocess.run(
                ["git", "rev-parse", "HEAD"], capture_output=True, check=True, text=True
            ).stdout.strip(),
            "source_sha256": provenance,
            "method_replacements_only": True,
        },
        indent=2,
    )
    + "\n"
)
raise SystemExit(
    pytest.main(
        [
            "tests/test_invalidation_references.py",
            "tests/test_context_atomic_lifecycle.py",
            "-q",
            "--basetemp=.local/efficiency-v1/review-old-methods-pytest-temp",
            "--junitxml=.local/efficiency-v1/review-old-methods-regressions.xml",
        ]
    )
)
