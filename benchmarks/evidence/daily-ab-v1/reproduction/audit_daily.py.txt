"""Independent, standard-library audit of a FINISHED daily A/B experiment.

Never launches Codex, imports Memglide, opens credential/database/native-state
files, changes source, reruns graders, or executes repository code. Reconstructs
forwarded source from frozen Git objects and checks observed native JSON usage.
Every invocation requires a fresh output prefix; failed audits remain intact.
"""

ROOT = Path('/workspace/Memglide')
USAGE = ('input_tokens', 'cached_input_tokens', 'output_tokens')
REPOSITORIES = {'desk-python-daily': 'python', 'catalog-python-daily': 'python',
                'editor-typescript-daily': 'typescript', 'inventory-typescript-daily': 'typescript'}


def sha(value):
    return hashlib.sha256(value).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':')).encode()


def unique_keys(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError('Duplicate JSON key')
        result[key] = value
    return result


def decode(value):
    return json.loads(value, object_pairs_hook=unique_keys)


def read_bytes(path, base):
    path, base = Path(path), Path(base).resolve()
    if not path.is_absolute():
        path = base / path
    if not path.resolve().is_relative_to(base):
        raise ValueError('Audit path escapes approved root')
    if any(p in {'codex-state', 'auth.json', 'memglide.db'} for p in path.parts):
        raise ValueError('Private runtime data is excluded')
    for parent in (path, *path.parents):
        if parent == base:
            break
        if parent.is_symlink():
            raise ValueError('Audit refuses symlinks')
    if not path.is_file() or path.stat().st_size > 128 * 1024 * 1024:
        raise ValueError('Audit input must be a bounded regular file')
    return path.read_bytes()


def read_json(path, base):
    return decode(read_bytes(path, base).decode('utf-8'))


def constant(path, name):
    module = ast.parse(read_bytes(path, ROOT).decode('utf-8'))
    for node in module.body:
        if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == name for t in node.targets):
            return ast.literal_eval(node.value)
    raise ValueError('Expected literal constant absent: ' + name)


def numeric(value):
    return isinstance(value, (float, int)) and not isinstance(value, bool) and math.isfinite(value)


def summary(values, additive=True):
    observed = sorted(value for value in values if numeric(value))
    return {'n': len(observed), 'missing': len(values) - len(observed),
            'mean': statistics.mean(observed) if observed else None,
            'median': statistics.median(observed) if observed else None,
            'p95': observed[max(0, math.ceil(len(observed) * .95) - 1)] if observed else None,
            'known_total': sum(observed) if observed and additive else None,
            'total': sum(observed) if observed and len(observed) == len(values) and additive else None}


def native_stream(data, unknown=False):
    events, bad_lines, items = [], [], {}
    for number, line in enumerate(data.decode('utf-8').splitlines(), 1):
        if not line.strip():
            continue
        try:
            event = decode(line)
            if not isinstance(event, dict) or not isinstance(event.get('type'), str):
                raise ValueError('Invalid event')
            events.append(event)
        except (ValueError, TypeError):
            bad_lines.append(number)
    turns = [row for row in events if row['type'] == 'turn.completed']
    failed = sum(row['type'] == 'turn.failed' for row in events)
    complete = bool(events and events[-1]['type'] in {'turn.completed', 'turn.failed'}
                    and not failed and not bad_lines and not unknown)
    observed = {}
    for key in USAGE:
        values = [(row.get('usage') or {}).get(key) for row in turns]
        observed[key] = sum(values) if values and all(type(v) is int and v >= 0 for v in values) else None
    for number, event in enumerate(events):
        item = event.get('item')
        if isinstance(item, dict) and event['type'] in {'item.started', 'item.updated', 'item.completed'}:
            key = str(item.get('id', f'event-without-id-{number}'))
            items[key] = {**items.get(key, {}), **item, '_lifecycle': event['type']}
    calls = [item for item in items.values() if item.get('type') == 'mcp_tool_call']
    return {'usage': observed.copy() if complete else dict.fromkeys(USAGE),
            'observed_completed_turn_usage': observed, 'complete': complete,
            'completed_turns': len(turns), 'failed_turns': failed,
            'native_error_count': sum(row['type'] in {'error', 'turn.failed'} for row in events),
            'mcp_call_count': len(calls) if events else None,
            'memglide_mcp_call_count': sum(str(row.get('server', '')).lower() == 'memglide' for row in calls) if events else None,
            'bad_lines': bad_lines}


def test_evidence(command, stdout, stderr):
    """Independent test execution proof; zero exit alone does not prove tests ran."""
    marker = re.findall(r'^CORPUS_RESULT=(.+)$', stdout, re.MULTILINE)
    if marker:
        if len(marker) != 1:
            return {}, False
        counts = decode(marker[0])
        valid = all(type(counts.get(k)) is int and counts[k] >= 0 for k in ('tests', 'failures', 'errors', 'skipped'))
        return counts, bool(valid and counts['tests'] == 2 and not any(counts[k] for k in ('failures', 'errors', 'skipped')))
    if '--test' in command:
        counts = {}
        for key in ('tests', 'pass', 'fail', 'cancelled', 'skipped'):
            found = re.findall(r'^# ' + key + r' (\d+)$', stdout, re.MULTILINE)
            if len(found) == 1:
                counts[key] = int(found[0])
        return counts, bool(set(counts) == {'tests', 'pass', 'fail', 'cancelled', 'skipped'}
                            and counts['tests'] == counts['pass'] and counts['tests'] > 0
                            and not any(counts[k] for k in ('fail', 'cancelled', 'skipped')))
    if 'unittest' in command:
        ran = re.findall(r'^Ran (\d+) tests?', stderr, re.MULTILINE)
        counts = {'tests': int(ran[-1])} if ran else {}
        return counts, bool(len(ran) == 1 and re.search(r'^OK\s*$', stderr, re.MULTILINE))
    return {}, True  # TypeScript compilation has no test-case counter.


class Audit:
    def __init__(self, experiment, expected_head, expected_app):
        self.exp, self.expected_head, self.expected_app = experiment, expected_head, expected_app
        self.errors, self.notes, self.check_count = [], [], 0

    def check(self, value, code, run_id=None):
        self.check_count += 1
        if not value:
            self.errors.append({'code': code, **({'run_id': run_id} if run_id else {})})

    def equal_number(self, a, b, code, run_id=None):
        self.check(a == b or numeric(a) and numeric(b) and math.isclose(a, b, rel_tol=1e-10, abs_tol=1e-9), code, run_id)

    def run(self):
        exp = self.exp
        metadata = read_json(exp / 'metadata.json', exp)
        if not metadata.get('finished_at') or not (exp / 'report.json').is_file():
            raise RuntimeError('Experiment is unfinished; native streams were not opened')
        records, previous = [], None
        for number, path in enumerate(sorted((exp / 'attempt-log').glob('*.json')), 1):
            value = read_json(path, exp)
            body = {k: v for k, v in value.items() if k != 'sha256'}
            self.check(path.name == f'{number:06d}.json' and value.get('sequence') == number, 'hashchain_sequence')
            self.check(value.get('previous_sha256') == previous and value.get('sha256') == sha(canonical(body)), 'hashchain_digest')
            records.append(value); previous = value.get('sha256')
        if not records or records[-1].get('event') != 'experiment.finished':
            raise RuntimeError('Durable finish seal absent; native streams were not opened')
        report = read_json(exp / 'report.json', exp)
        self.check(records[-1].get('report_sha256') == sha(read_bytes(exp / 'report.json', exp)), 'sealed_report_digest')
        frozen, settings = metadata['frozen_manifest'], metadata['frozen_manifest']['settings']
        self.check(frozen.get('git_head') == self.expected_head, 'externally_expected_frozen_head')
        current_head = subprocess.check_output(['git', '-C', str(ROOT), 'rev-parse', 'HEAD'], text=True).strip()
        self.check(current_head == self.expected_head, 'current_head_still_frozen')
        for path, expected in frozen['files'].items():
            self.check(sha(read_bytes(ROOT / path, ROOT)) == expected, 'frozen_file:' + path)
        for phase in ('memglide_source_before', 'memglide_source_after'):
            source = metadata.get(phase) or {}
            self.check(source.get('source_tree_sha256') == self.expected_app, 'expected_application_aggregate:' + phase)
            self.check(sha(json.dumps(source.get('source_files'), sort_keys=True).encode()) == self.expected_app, 'application_hash_manifest_self_consistent:' + phase)
        self.check(metadata.get('source_unchanged_during_experiment') is True and not metadata.get('source_guard_failure'), 'source_guard_verified')
        self.check(settings.get('model') == metadata.get('model') and settings.get('effort') == metadata.get('reasoning_effort'), 'frozen_model_and_effort')
        self.check(settings.get('codex_version') == metadata.get('codex_version'), 'frozen_native_cli_version')
        instructions = constant(ROOT / 'benchmarks/run_daily.py', 'COMMON_INSTRUCTIONS')
        explanation = constant(ROOT / 'src/memglide/integrations/preload.py', '_EXPLANATION')
        metric_names = constant(ROOT / 'benchmarks/run_daily.py', 'METRICS')
        self.check(sha(instructions.encode()) == frozen['common_instructions_sha256'], 'common_instructions_hash')
        plan = read_json(exp / 'plan.json', exp)
        runs = read_json(exp / 'runs.json', exp)
        state = read_json(exp / 'state.json', exp)
        supplemental = read_json(exp / 'supplemental-runs.json', exp)
        tasks = {row['id']: row for row in plan['tasks']}
        assignments = {row['run_id']: row for row in plan['runs']}
        by_id = {row['run_id']: row for row in runs}
        reported = {row['run_id']: row for row in report['runs']}
        self.check(len(tasks) == len(plan['tasks']) == 20, 'exact_twenty_unique_tasks')
        self.check(Counter(row['language'] for row in tasks.values()) == {'python': 10, 'typescript': 10}, 'bilingual_task_balance')
        self.check(Counter(row['repository'] for row in tasks.values()) == {key: 5 for key in REPOSITORIES}, 'four_repositories_five_tasks_each')
        self.check(Counter(row['task_kind'] for row in tasks.values()) == {'single-source': 8, 'integration': 8, 'supplied-context': 4}, 'declared_task_strata')
        self.check(len(assignments) == len(plan['runs']) == 80, 'eighty_unique_original_slots')
        self.check(len(runs) == len(by_id) == 80 and set(by_id) == set(assignments), 'all_original_slots_retained')
        self.check(len(report['runs']) == 80 and set(reported) == set(assignments), 'all_original_slots_reported')
        self.check(state['originals'] == by_id, 'authoritative_original_projection')
        self.check(state['supplemental'] == supplemental == [], 'no_supplemental_original_replacement')
        self.check(report['tasks'] == plan['tasks'], 'frozen_tasks_reported_verbatim')
        for key, repository in metadata['repositories'].items():
            directory = exp / 'repositories' / key
            self.check(Path(repository['path']) == directory, 'frozen_repository_path:' + key)
            value = subprocess.check_output(['git', '-C', str(directory), 'rev-parse', 'HEAD', 'HEAD^{tree}'], text=True).splitlines()
            self.check(value == [repository['commit'], repository['tree']], 'frozen_repository_commit_tree:' + key)
        sealed, starts = {}, {}
        for record in records:
            if record['event'] == 'attempt.started':
                starts.setdefault(record['attempt_id'], []).append(record)
            if record['event'] in {'attempt.finished', 'attempt.interrupted'}:
                run_id = record['attempt_id']; sealed.setdefault(run_id, []).append(record)
                self.check(sha(read_bytes(exp / 'runs' / run_id / 'result.json', exp)) == record['result_sha256'], 'sealed_attempt_result', run_id)
        observations, normalized_native, process_states = [], set(), Counter()
        for row in runs:
            run_id = row['run_id']; assignment = assignments[run_id]; task = tasks[row['task_id']]
            self.check(all(row.get(k) == assignment[k] for k in ('task_id', 'configuration', 'repeat', 'order', 'repository', 'language', 'selection_stratum')), 'planned_assignment_identity', run_id)
            if not row.get('attempted'):
                self.check(row.get('status') == 'not_run', 'unattempted_original_explicit', run_id)
                continue
            artifacts = exp / 'runs' / run_id
            self.check(Path(row['artifacts']) == artifacts and Path(row['worktree']) == exp / 'worktrees' / run_id, 'isolated_artifact_worktree_paths', run_id)
            self.check(read_json(artifacts / 'result.json', exp) == row, 'immutable_result_matches_checkpoint', run_id)
            start = read_json(artifacts / 'attempt-start.json', exp)
            self.check(start['run_id'] == run_id and start['status'] == 'running', 'durable_start_record', run_id)
            self.check(bool(sealed.get(run_id)), 'original_has_terminal_log_seal', run_id)
            if starts.get(run_id):
                self.check(len(starts[run_id]) == 1 and starts[run_id][0]['start_sha256'] == sha(canonical(start)), 'durable_start_hash', run_id)
            else:
                self.notes.append({'code': 'start_recovered_before_log_append', 'run_id': run_id})
            identity_path = artifacts / 'process-identity.json'
            if identity_path.exists():
                identity = read_json(identity_path, exp); stat_path = Path('/proc') / str(identity['pid']) / 'stat'
                if not stat_path.exists():
                    status = 'absent'
                else:
                    fields = stat_path.read_text().rsplit(')', 1)[1].split()
                    status = fields[0] if int(fields[19]) == identity['start_ticks'] else 'pid_reused'
                process_states[status] += 1
                self.check(status in {'absent', 'pid_reused', 'Z'}, 'native_process_no_longer_running', run_id)
            base_prompt = (instructions + '\nTask:\n' + task['prompt']
                           + '\n\nCaller-declared task scope (available equally to both configurations):\n'
                           + ('\n'.join('- ' + path for path in task['focus_paths']) or 'No explicit source paths supplied.') + '\n')
            base_digest = sha(base_prompt.encode())
            self.check(base_digest == frozen['common_task_prompt_sha256'][task['id']], 'frozen_common_task_hash', run_id)
            for path in task['focus_paths']:
                self.check(path in task['prompt'], 'scope_also_disclosed_in_base_task', run_id)
            prompt_path = artifacts / 'prompt.txt'
            if prompt_path.exists():
                self.check(read_bytes(prompt_path, exp) == base_prompt.encode() and row.get('common_task_prompt_sha256') == base_digest, 'identical_actual_common_task_input', run_id)
            repository = metadata['repositories'][task['repository']]
            if row.get('starting_commit') is not None:
                self.check(row['starting_commit'] == repository['commit'], 'identical_starting_commit', run_id)
            native = row.get('native_command')
            if native:
                configs, norm, index = {}, [], 0
                while index < len(native):
                    part = native[index]
                    if part == '-c':
                        configs.update(tomllib.loads(native[index + 1])); norm += native[index:index+2]; index += 2
                    elif part in {'-C', '--output-schema', '--output-last-message'}:
                        norm += [part, '<isolated-path>']; index += 2
                    else:
                        norm.append(part); index += 1
                normalized_native.add(tuple(norm))
                self.check(configs.get('mcp_servers') == {} and configs.get('approval_policy') == 'never', 'same_disabled_mcp_and_global_approval', run_id)
                self.check(configs.get('model_reasoning_effort') == settings['effort'] and native[native.index('--model')+1] == settings['model'] and native[native.index('--sandbox')+1] == 'workspace-write', 'same_native_model_effort_sandbox', run_id)
                self.check(all(flag in native for flag in ('--json', '--output-schema', '--ignore-user-config', '--ignore-rules')), 'supported_structured_native_exec', run_id)
                if row['configuration'] == 'A':
                    self.check(read_json(artifacts / 'command.json', exp) == native, 'A_actual_command_is_native', run_id)
            events_path = artifacts / 'events.jsonl'
            parsed = native_stream(read_bytes(events_path, exp), row.get('execution_outcome_unknown') is True) if events_path.exists() else None
            telemetry = row.get('telemetry') or {}
            if parsed is not None:
                self.check(parsed['usage'] == telemetry.get('usage'), 'actual_native_usage_reproduced', run_id)
                self.check(parsed['observed_completed_turn_usage'] == telemetry.get('observed_completed_turn_usage'), 'partial_turn_usage_preserved', run_id)
                for key in ('completed_turns', 'failed_turns', 'mcp_call_count', 'memglide_mcp_call_count'):
                    self.check(parsed[key] == telemetry.get(key), 'actual_native_count:' + key, run_id)
                self.check(parsed['complete'] == telemetry.get('usage_coverage_complete') == telemetry.get('mcp_counts_complete'), 'native_coverage_mask', run_id)
                values = parsed['usage']
                if values['input_tokens'] is not None and values['cached_input_tokens'] is not None:
                    self.check(values['cached_input_tokens'] <= values['input_tokens'], 'cached_input_is_subset', run_id)
            final_path = artifacts / 'final.json'
            final_valid, final_digest = False, None
            if final_path.exists():
                try:
                    final_bytes = read_bytes(final_path, exp)
                    final_value = decode(final_bytes.decode('utf-8'))
                    final_valid = (len(final_bytes) <= 1_000_000 and isinstance(final_value, dict)
                                   and set(final_value) == {'summary', 'tests_run'}
                                   and isinstance(final_value['summary'], str)
                                   and isinstance(final_value['tests_run'], list)
                                   and all(isinstance(item, str) for item in final_value['tests_run']))
                    if final_valid:
                        final_digest = sha(final_bytes)
                except (OSError, ValueError, UnicodeError):
                    final_valid = False
            native_valid = bool(row.get('exit_code') == 0 and row.get('timed_out') is False
                                and parsed and parsed['complete'] and parsed['completed_turns'] > 0
                                and not parsed['failed_turns'] and not parsed['native_error_count']
                                and final_valid and not row.get('execution_outcome_unknown'))
            if 'native_execution_validity' in row:
                self.check(row['native_execution_validity'].get('valid') is native_valid, 'native_validity_matches_exit_stream_schema', run_id)
                self.check(row['native_execution_validity'].get('final_output_sha256') == final_digest, 'final_structured_output_hash', run_id)
            no_mcp = bool(parsed and parsed['complete'] and parsed['mcp_call_count'] == 0)
            fidelity = None if row.get('execution_outcome_unknown') else no_mcp
            mode, decision, context_bytes, source_count, prep = 'baseline', None, 0, 0, {}
            preparation_path = artifacts / 'preparation.json'
            if row['configuration'] == 'B' and preparation_path.exists():
                forwarding = read_json(preparation_path, exp); prep = forwarding['preparation']
                mode, decision = prep['mode'], prep['reason']; files = prep['files']; source_count = len(files)
                if mode == 'preloaded':
                    delivered = [item['path'] for item in files]
                    self.check(len(delivered) == len(set(delivered)) and set(delivered) == set(task['focus_paths']), 'exact_caller_focus_coverage', run_id)
                    self.check(not task['context_supplied'], 'supplied_context_never_reloaded', run_id)
                    sources = []
                    for item in files:
                        relative = item['path']
                        if relative not in task['focus_paths']:
                            raise ValueError('Unexpected source path; Git object was not opened')
                        original = subprocess.check_output(['git', '-C', repository['path'], 'show', repository['commit'] + ':' + relative])
                        metadata_item = {'path': relative, 'sha256': sha(original), 'start_line': 1, 'end_line': len(original.decode('utf-8').splitlines())}
                        self.check(item == metadata_item, 'preload_metadata_matches_full_frozen_git_source', run_id)
                        sources.append({**metadata_item, 'content': original.decode('utf-8')})
                    addition = '\n\n' + explanation + json.dumps({'memglide_sources': sources}, ensure_ascii=False, separators=(',', ':'))
                    reconstructed = base_prompt + addition; context_bytes = len(addition.encode())
                    self.check(context_bytes <= settings['context_max_bytes'], 'complete_envelope_within_frozen_cap', run_id)
                elif mode == 'passthrough':
                    reconstructed = base_prompt
                    self.check(not files and prep['context_bytes'] == 0, 'passthrough_source_free', run_id)
                    self.check(not task['context_supplied'] or decision == 'context_supplied', 'explicit_supplied_context_bypass', run_id)
                else:
                    raise ValueError('Unknown preparation mode')
                expected_digest = sha(reconstructed.encode())
                self.check(prep['task_prompt_sha256'] == base_digest, 'preloader_received_common_task', run_id)
                self.check(prep['prepared_prompt_sha256'] == expected_digest == row.get('actual_model_prompt_sha256'), 'reconstructed_actual_prepared_prompt_hash', run_id)
                self.check(prep['context_bytes'] == context_bytes, 'exact_preparation_context_bytes', run_id)
                self.check(forwarding['native_stdin'] == {'sha256': expected_digest, 'bytes': len(reconstructed.encode()), 'transport': 'temporary_file_descriptor'}, 'forwarded_native_stdin_matches_reconstruction', run_id)
                if native:
                    self.check(forwarding['exec_argv'] == [*native[:-1], '--', '-'] and forwarding['executable'] == native[0], 'B_preserves_all_native_options', run_id)
                # This proves deterministic forwarding records/source equality,
                # not independent server-side receipt or model attention.
            elif row['configuration'] == 'B' and not row.get('execution_outcome_unknown'):
                fidelity = False
            elif row['configuration'] == 'A':
                self.check(row.get('actual_model_prompt_sha256') in {base_digest, None}, 'A_prompt_unchanged', run_id)
            recorded_fidelity = (row.get('preload_treatment_fidelity') or {}).get('valid')
            if row.get('execution_outcome_unknown') or 'preload_treatment_fidelity' in row:
                self.check(recorded_fidelity is fidelity, 'treatment_fidelity_matches_evidence', run_id)
            grade = row.get('grader') or {}; grade_path = artifacts / 'grade.json'
            if grade_path.exists():
                stored_grade = read_json(grade_path, exp)
                expected_projection = dict(stored_grade)
                if row.get('protected_file_violations'):
                    expected_projection['success'] = False
                self.check(grade == expected_projection, 'external_grade_projection', run_id)
                commands = stored_grade.get('commands') or []
                self.check(len(commands) == (2 if row['language'] == 'python' else 3), 'all_behavior_smoke_compile_commands_present', run_id)
                test_proofs = []
                for index, command in enumerate(commands):
                    stdout = read_bytes(command['stdout_path'], exp).decode('utf-8')
                    stderr = read_bytes(command['stderr_path'], exp).decode('utf-8')
                    counts, proof = test_evidence(command['command'], stdout, stderr)
                    if index == 0:
                        proof = proof and counts.get('tests') == 2
                        oracle = artifacts / ('held_out.py' if row['language'] == 'python' else 'held_out.test.mjs')
                        self.check(str(oracle) in command['command'] and oracle.is_file(), 'external_oracle_executed_outside_worktree', run_id)
                    elif index == 1:
                        proof = proof and counts.get('tests', 0) >= 5
                    if 'tests' in counts:
                        self.check(command.get('counts', {}).get('tests') == counts['tests'], 'grade_test_counts_match_raw_logs', run_id)
                    test_proofs.append(bool(proof and command['exit_code'] == 0))
                for item in stored_grade.get('protected_files', []):
                    relative = item['file']; original = subprocess.check_output(['git', '-C', repository['path'], 'show', repository['commit'] + ':' + relative])
                    self.check(sha(original) == item['expected_sha256'], 'protected_fixture_baseline_git_hash', run_id)
                    self.check(item['unchanged'] is (item['actual_sha256'] == item['expected_sha256']), 'protected_fixture_comparison_boolean', run_id)
                grade_truth = (bool(stored_grade.get('protected_files_unchanged')) and all(test_proofs)
                               and not row.get('protected_file_violations'))
                self.check(grade.get('success') is grade_truth, 'grade_requires_actual_completed_tests', run_id)
            successful = bool(row['status'] == 'completed' and grade.get('success') is True and recorded_fidelity is True and native_valid)
            if row['status'] == 'completed':
                self.check(successful and row.get('exit_code') == 0 and row.get('timed_out') is False, 'completion_requires_execution_grade_and_treatment', run_id)
            self.check(row.get('actual_cost_usd') is None and row.get('computed_cost_usd') is None and telemetry.get('model_calls') is None and telemetry.get('repeated_file_reads') is None, 'missing_cost_model_call_actual_io_remain_null', run_id)
            self.check(reported[run_id].get('status') == row['status'] and reported[run_id].get('grader') == grade, 'all_outcomes_reported_unchanged', run_id)
            values = (parsed or {}).get('usage', dict.fromkeys(USAGE)); overhead, setup = row.get('overhead') or {}, row.get('setup') or {}
            known_zero = row['configuration'] == 'A' or row['selection_stratum'] == 'caller_context_supplied'
            setup_default = 0 if known_zero else None
            child_cpu, index_cpu = overhead.get('cpu_seconds'), setup.get('cpu_seconds', setup_default)
            calculated = {**values,
                          'uncached_input_tokens': values['input_tokens'] - values['cached_input_tokens'] if all(numeric(values[k]) for k in ('input_tokens', 'cached_input_tokens')) else None,
                          'wall_time_seconds': row.get('wall_time_seconds'), 'end_to_end_wall_seconds': row.get('end_to_end_wall_seconds'),
                          'native_exec_wall_seconds': row.get('native_exec_wall_seconds'),
                          'approval_index_wall_seconds': setup.get('wall_seconds', setup_default), 'approval_index_cpu_seconds': index_cpu,
                          'context_preparation_wall_seconds': prep.get('preparation_wall_seconds'), 'context_preparation_cpu_seconds': prep.get('preparation_cpu_seconds'),
                          'process_tree_peak_rss_bytes': overhead.get('process_tree_peak_rss_bytes'), 'cpu_seconds': child_cpu,
                          'total_observed_local_cpu_seconds': child_cpu + index_cpu if numeric(child_cpu) and numeric(index_cpu) else None,
                          'mcp_calls': parsed['mcp_call_count'] if parsed and parsed['complete'] else None,
                          'model_calls': None, 'repeated_file_reads': None, 'actual_cost_usd': None, 'computed_cost_usd': None,
                          'observed_file_read_commands': telemetry.get('observed_file_read_commands'),
                          'observed_repeated_read_commands': telemetry.get('observed_repeated_read_commands')}
            for key in metric_names:
                self.equal_number(calculated.get(key), reported[run_id]['metrics'].get(key), 'individual_metric:' + key, run_id)
            observations.append({'run_id': run_id, 'task_id': row['task_id'], 'repeat': row['repeat'], 'configuration': row['configuration'],
                                 'selection_stratum': row['selection_stratum'], 'status': row['status'], 'externally_graded_success': grade.get('success'),
                                 'successful_completion': successful, 'native_execution_valid': native_valid,
                                 'fidelity': recorded_fidelity, 'mode': mode, 'selection_reason': decision,
                                 'source_file_count': source_count, 'context_bytes': context_bytes, 'metrics': calculated})
        self.check(len(normalized_native) <= 1, 'same_non_treatment_native_settings')
        for task_id in tasks:
            assigned = sorted((r for r in assignments.values() if r['task_id'] == task_id), key=lambda r: r['order'])
            self.check(Counter((r['configuration'], r['repeat']) for r in assigned) == {(arm, rep): 1 for arm in ('A', 'B') for rep in (1, 2)}, 'exact_two_repeats_each_arm:' + task_id)
            self.check([r['configuration'] for r in assigned[:2]] == list(reversed([r['configuration'] for r in assigned[2:]])), 'counterbalanced_order:' + task_id)
        observed_by_id = {row['run_id']: row for row in observations}; arms = {}
        for arm in ('A', 'B'):
            assigned = [row for row in runs if row['configuration'] == arm]
            attempted = [row for row in assigned if row.get('attempted')]
            actual = [observed_by_id[row['run_id']] for row in attempted]
            aggregated = report['configurations'][arm]
            successful = sum(row['successful_completion'] for row in actual)
            self.check(aggregated['planned_runs'] == len(assigned) and aggregated['attempted_runs'] == len(attempted), 'reported_original_denominators:' + arm)
            self.check(aggregated['successful_runs'] == successful, 'success_requires_completed_grade_valid:' + arm)
            self.equal_number(aggregated['successful_task_completion_rate'], successful / len(attempted) if attempted else None, 'completion_rate_all_attempts:' + arm)
            metrics = {}
            for key in metric_names:
                metrics[key] = summary([row['metrics'].get(key) for row in actual], additive='rss_bytes' not in key)
                for field, expected in metrics[key].items():
                    self.equal_number(aggregated['metrics'][key][field], expected, 'aggregate:' + arm + ':' + key + ':' + field)
            self.check(aggregated['actual_cost_per_successful_task_usd']['value'] is None and aggregated['computed_cost_per_successful_task_usd']['value'] is None, 'cost_per_success_unknown:' + arm)
            arms[arm] = {'planned': len(assigned), 'attempted': len(attempted), 'completed_and_successful': successful,
                         'grade_passed': sum(row['externally_graded_success'] is True for row in actual),
                         'statuses': dict(Counter(row['status'] for row in assigned)), 'modes': dict(Counter(row['mode'] for row in actual)), 'metrics': metrics}
        pairs = []
        for task_id in tasks:
            for repeat in (1, 2):
                pair = {row['configuration']: row for row in observations if row['task_id'] == task_id and row['repeat'] == repeat}
                if (set(pair) != {'A', 'B'}
                    or any(row['fidelity'] is not True or row['native_execution_valid'] is not True for row in pair.values())
                    or not metadata.get('source_unchanged_during_experiment')):
                    continue
                values = {}
                for key in metric_names:
                    a, b = pair['A']['metrics'][key], pair['B']['metrics'][key]
                    values[key] = {'A': a, 'B': b, 'B_minus_A': b-a if numeric(a) and numeric(b) else None,
                                   'B_over_A': b/a if numeric(a) and numeric(b) and a else None}
                pairs.append({'task_id': task_id, 'repeat': repeat, 'metrics': values})
        self.check(len(pairs) == report['paired_comparison']['valid_pairs'], 'verified_pair_count_reproduced')
        reported_pairs = {(r['task_id'], r['repeat']): r for r in report['paired_comparison']['pairs']}
        for pair in pairs:
            other = reported_pairs.get((pair['task_id'], pair['repeat']))
            self.check(other is not None, 'verified_pair_present:' + pair['task_id'])
            if other:
                for key in metric_names:
                    for field, expected in pair['metrics'][key].items():
                        self.equal_number(other['metrics'][key][field], expected, 'paired_metric:' + key + ':' + field, pair['task_id'])
        self.check({row['run_id'] for row in report['failures']} == {row['run_id'] for row in runs if row['status'] != 'completed'}, 'every_failed_original_visible')
        return {'schema_version': 1, 'audited_at_utc': datetime.now(timezone.utc).isoformat(), 'pass': not self.errors,
                'expected_frozen_head': self.expected_head, 'expected_app_sha256': self.expected_app,
                'source_report_sha256': sha(read_bytes(exp / 'report.json', exp)), 'auditor_sha256': sha(Path(__file__).read_bytes()),
                'checked_assertions': self.check_count, 'errors': self.errors, 'notes': self.notes,
                'task_count': len(tasks), 'planned_originals': len(assignments), 'observed_attempts': len(observations),
                'verified_pairs': len(pairs), 'attempt_log_records': len(records), 'resume_count': metadata.get('resume_count'),
                'native_process_states': dict(process_states), 'arms': arms, 'runs': observations,
                'limitations': ['Known-file utility-edit microbenchmark; four small synthetic repositories and correlated repeats.',
                    'Prepared-prompt reconstruction validates frozen source bytes and launcher forwarding records; it does not prove provider-side receipt or model attention.',
                    'The task prompt is identical before treatment; preloading intentionally changes the actual native prompt.',
                    'No billed costs, cost per successful task, underlying provider model-call counts or actual filesystem I/O trace are supplied.',
                    'Cached input is included in total input; uncached input is derived by subtraction.',
                    'RSS/CPU reproduce sampled local observations, not remote inference overhead; read-command fields remain proxies.',
                    'The auditor never imports Memglide or benchmark code, opens credentials/databases/native state, reruns tests or calls a model.']}


def markdown(value):
    lines = ['# Independent daily A/B audit', '', '**Verdict: ' + ('PASS' if value['pass'] else 'FAIL') + '**.', '',
             'Audit artifacts preserve every invocation, including any initial auditor errors.', '']
    if 'audit_exception_type' in value:
        return '\n'.join(lines + ['Audit could not finish: ' + value['audit_exception_type'] + '.', ''])
    lines += [f"Checked {value['checked_assertions']:,} assertions; {len(value['errors'])} errors and {len(value['notes'])} notes.", '',
              f"{value['task_count']} tasks; {value['planned_originals']} original slots; {value['observed_attempts']} attempts; {value['verified_pairs']} verified pairs.", '',
              '| Arm | Planned | Attempted | Successful completion | Execution statuses |', '|---|---:|---:|---:|---|']
    for arm, row in value['arms'].items():
        lines.append(f"| {arm} | {row['planned']} | {row['attempted']} | {row['completed_and_successful']} | {json.dumps(row['statuses'])} |")
    lines += ['', 'Successful completion requires completed execution, a passing external grade and valid treatment. Failed, invalid and interrupted assignments remain visible.', '',
              '| Metric | A mean | B mean | A median | B median |', '|---|---:|---:|---:|---:|']
    for key in ('input_tokens', 'cached_input_tokens', 'uncached_input_tokens', 'output_tokens', 'end_to_end_wall_seconds', 'mcp_calls'):
        a, b = value['arms']['A']['metrics'][key], value['arms']['B']['metrics'][key]
        lines.append('| ' + key + ' | ' + ' | '.join('unavailable' if n is None else f'{n:.6g}' for n in (a['mean'], b['mean'], a['median'], b['median'])) + ' |')
    lines += ['', '## Limits', ''] + ['- ' + text for text in value['limitations']]
    if value['errors']:
        lines += ['', '## Audit errors', ''] + ['- ' + json.dumps(error) for error in value['errors']]
    return '\n'.join(lines) + '\n'


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--experiment', type=Path, required=True)
    parser.add_argument('--expected-head', required=True)
    parser.add_argument('--expected-app', required=True)
    parser.add_argument('--output-prefix', type=Path, required=True)
    parser.add_argument('--after-finished', action='store_true')
    args = parser.parse_args()
    if not args.after_finished:
        parser.error('Explicit --after-finished is required; streams were not opened')
    prefix = args.output_prefix.absolute()
    outputs = [prefix.with_suffix('.json'), prefix.with_suffix('.md')]
    allowed = (ROOT / '.local/daily-v1').resolve()
    if any(path.exists() or not path.resolve().is_relative_to(allowed) for path in outputs):
        parser.error('Use a fresh output prefix below ignored .local/daily-v1')
    if not re.fullmatch(r'[0-9a-f]{40,64}', args.expected_head) or not re.fullmatch(r'[0-9a-f]{64}', args.expected_app):
        parser.error('Expected frozen commit and application SHA-256 must be explicit hexadecimal digests')
    try:
        value = Audit(args.experiment.resolve(), args.expected_head, args.expected_app).run()
    except Exception as error:
        value = {'schema_version': 1, 'pass': False, 'audit_exception_type': type(error).__name__,
                 'audit_exception_message': str(error), 'audited_at_utc': datetime.now(timezone.utc).isoformat()}
    with outputs[0].open('x', encoding='utf-8') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False); stream.write('\n')
    with outputs[1].open('x', encoding='utf-8') as stream:
        stream.write(markdown(value))
    print(json.dumps({'pass': value['pass'], 'checked': value.get('checked_assertions'), 'errors': len(value.get('errors', [])),
                      'exception': value.get('audit_exception_type'), 'json': str(outputs[0]), 'markdown': str(outputs[1])}))
    return int(not value['pass'])


if __name__ == '__main__':
    raise SystemExit(main())
