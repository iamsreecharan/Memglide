"""Smoke installed preload APIs/console without network or model requests.

Run with the installed virtual environment's Python and -I. This is an
installation smoke, not an additional pytest test suite or token benchmark.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import subprocess
import sys
import tempfile
from pathlib import Path
from typing import Any

import memglide
from memglide.core.indexer import Indexer
from memglide.integrations.preload import prepare_prompt
from memglide.storage.database import Database


def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def envelope(preparation: dict[str, Any]) -> dict[str, Any]:
    text = preparation["prompt"]
    return json.loads(text[text.index('{"memglide_sources"') :])


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--state", type=Path, required=True)
    parser.add_argument("--report", type=Path, required=True)
    options = parser.parse_args()
    assert sys.flags.isolated, "Run this installed-wheel smoke with Python -I"
    package = Path(memglide.__file__).resolve()
    assert package.is_relative_to(Path(sys.prefix).resolve()), "Imported checkout instead of wheel"
    state = options.state.absolute()
    report = options.report.absolute()
    assert not report.exists(), "Choose a new report path; existing evidence is never overwritten"
    assert report.parent.is_dir(), "Create the report parent before running the smoke"
    assert not state.is_relative_to(Path("/tmp")), (
        "Choose --state outside /tmp so the fake installed binary is outside the launcher's cwd"
    )
    state.mkdir(parents=True, exist_ok=False, mode=0o700)
    os.environ["MEMGLIDE_HOME"] = str(state)
    console = Path(sys.executable).absolute().parent / "memglide-codex"
    assert console.is_file() and os.access(console, os.X_OK), "Installed console entrypoint missing"
    checks: dict[str, bool] = {"isolated_installed_package": True, "installed_entrypoint": True}

    with tempfile.TemporaryDirectory(prefix="memglide-installed-preload-", dir="/tmp") as temporary:
        repository = Path(temporary)
        subprocess.run(
            ["git", "-c", f"core.hooksPath={os.devnull}", "init", "-q", str(repository)],
            check=True,
            capture_output=True,
            timeout=15,
        )
        source = repository / "totals.py"
        initial = b"def calculate_total(values):\n    return sum(values) + 1\n"
        changed = b"def calculate_total(values):\n    return sum(values) + 2\n"
        assert len(initial) == len(changed)
        source.write_bytes(initial)
        marker = state / "untrusted-source-must-not-execute"
        untrusted = repository / "untrusted.py"
        untrusted.write_text(
            "from pathlib import Path\n"
            f"Path({str(marker)!r}).write_text('executed')\n"
            "raise RuntimeError('Indexed source must never execute')\n",
            encoding="utf-8",
        )
        oversized = repository / "large.py"
        oversized.write_text("# complete source that exceeds a tiny context budget\n" * 400)
        database_path = state / "approved.db"
        database = Database(database_path)
        project = Indexer(database).add_project(str(repository))
        project_id = project["id"]
        assert not marker.exists()
        checks["explicit_git_project_approval_and_indexing"] = True

        prompt = "Review 'totals' literally: $HOME `no-shell`; λ\r\nnext\rfinal\n"
        prepared = prepare_prompt(
            prompt,
            project_id=project_id,
            db_path=database_path,
            focus_paths=("totals.py", "untrusted.py"),
        )
        assert prepared["mode"] == "preloaded" and prepared["reason"] == "explicit_focus"
        sources = envelope(prepared)["memglide_sources"]
        actual = {entry["path"]: entry for entry in sources}
        assert actual["totals.py"]["content"].encode("utf-8") == initial
        assert actual["untrusted.py"]["content"] == untrusted.read_text()
        for entry in sources:
            assert entry["sha256"] == digest(entry["content"].encode("utf-8"))
            assert entry["start_line"] == 1
            assert entry["end_line"] == len(entry["content"].splitlines())
        assert prepared["task_prompt_sha256"] == digest(prompt.encode("utf-8"))
        assert prepared["prepared_prompt_sha256"] == digest(prepared["prompt"].encode("utf-8"))
        checks["complete_focused_sources_hashes_and_line_ranges"] = True

        before = source.stat()
        indexed_before = database.one(
            "SELECT hash FROM files WHERE project_id=? AND path=?", (project_id, "totals.py")
        )
        assert indexed_before is not None and indexed_before["hash"] == digest(initial)
        source.write_bytes(changed)
        os.utime(source, ns=(before.st_atime_ns, before.st_mtime_ns))
        assert source.stat().st_size == before.st_size
        assert source.stat().st_mtime_ns == before.st_mtime_ns
        refreshed = prepare_prompt(
            prompt, project_id=project_id, db_path=database_path, focus_paths=("totals.py",)
        )
        assert refreshed["mode"] == "preloaded"
        fresh_source = envelope(refreshed)["memglide_sources"][0]
        assert fresh_source["content"].encode("utf-8") == changed
        assert fresh_source["sha256"] == digest(changed) != digest(initial)
        assert database.one(
            "SELECT hash FROM files WHERE project_id=? AND path=?", (project_id, "totals.py")
        ) == indexed_before
        checks["fresh_source_despite_unchanged_size_mtime_and_old_index"] = True

        for label, focus, budget in (
            ("oversized_complete_source_fallback", ("large.py",), 500),
            ("missing_source_fallback", ("missing.py",), 12_000),
        ):
            fallback = prepare_prompt(
                prompt,
                project_id=project_id,
                db_path=database_path,
                focus_paths=focus,
                max_context_bytes=budget,
            )
            assert fallback["mode"] == "passthrough"
            assert fallback["prompt"].encode("utf-8") == prompt.encode("utf-8")
            assert fallback["files"] == [] and fallback["context_bytes"] == 0
            checks[label] = True
        absent_database = state / "never-created" / "state.db"
        bypass = prepare_prompt(
            prompt,
            project_id=project_id,
            db_path=absent_database,
            focus_paths=("totals.py",),
            context_supplied=True,
        )
        assert bypass["mode"] == "passthrough" and bypass["reason"] == "context_supplied"
        assert bypass["prompt"].encode("utf-8") == prompt.encode("utf-8")
        assert not absent_database.parent.exists()
        checks["context_supplied_bypasses_nonexistent_database"] = True

        fake_directory = state / "fake-installed-bin"
        fake_directory.mkdir(mode=0o700)
        fake_executable = fake_directory / "codex"
        native_capture = state / "actual-native-stdin.bin"
        native_marker = state / "exactly-one-native-exec.txt"
        native_observation = state / "actual-native-observation.json"
        preparation_report = state / "console-preparation.json"
        fake_executable.write_text(
            f"#!{sys.executable}\n"
            "import hashlib,json,sys\nfrom pathlib import Path\n"
            f"marker=Path({str(native_marker)!r})\n"
            "with marker.open('x') as stream: stream.write('one native execution\\n')\n"
            "payload=sys.stdin.buffer.read()\n"
            f"capture=Path({str(native_capture)!r})\n"
            "with capture.open('xb') as stream: stream.write(payload)\n"
            f"report=Path({str(preparation_report)!r})\n"
            "observed={'argv':sys.argv,'stdin_sha256':hashlib.sha256(payload).hexdigest(),"
            "'stdin_bytes':len(payload),'report_written_before_native_exec':report.is_file()}\n"
            f"with Path({str(native_observation)!r}).open('x') as stream: "
            "json.dump(observed,stream)\n"
            "raise SystemExit(29)\n",
            encoding="utf-8",
        )
        fake_executable.chmod(0o700)
        result = subprocess.run(
            [
                str(console),
                "--codex-executable",
                str(fake_executable),
                "--project",
                project_id,
                "--db",
                str(database_path),
                "--focus",
                "totals.py",
                "--preparation-report",
                str(preparation_report),
                "--",
                "--json",
                "--model",
                "synthetic-native-no-model-request",
                "-c",
                'model_reasoning_effort="low"',
            ],
            cwd="/tmp",
            input=prompt.encode("utf-8"),
            capture_output=True,
            timeout=30,
        )
        assert result.returncode == 29, result.stderr.decode("utf-8", errors="replace")
        assert not result.stdout and not result.stderr
        assert native_marker.read_text() == "one native execution\n"
        observation = json.loads(native_observation.read_text())
        delivered = native_capture.read_bytes()
        launch_report = json.loads(preparation_report.read_text())
        assert observation["report_written_before_native_exec"]
        assert observation["argv"] == [
            str(fake_executable),
            "exec",
            "--json",
            "--model",
            "synthetic-native-no-model-request",
            "-c",
            'model_reasoning_effort="low"',
            "--",
            "-",
        ]
        assert launch_report["exec_argv"] == observation["argv"]
        assert delivered == refreshed["prompt"].encode("utf-8")
        assert launch_report["preparation"]["prepared_prompt_sha256"] == digest(delivered)
        assert launch_report["preparation"]["task_prompt_sha256"] == digest(prompt.encode("utf-8"))
        assert launch_report["native_stdin"]["sha256"] == observation["stdin_sha256"] == digest(delivered)
        assert launch_report["native_stdin"]["bytes"] == observation["stdin_bytes"] == len(delivered)
        assert launch_report["native_stdin"]["transport"] == "temporary_file_descriptor"
        assert launch_report["preparation"]["files"] == refreshed["files"]
        assert preparation_report.stat().st_mode & 0o777 == 0o600
        assert "calculate_total" not in preparation_report.read_text()
        assert not marker.exists()
        checks["installed_console_exact_utf8_stdin_and_native_arguments"] = True
        checks["private_report_precedes_exec_and_matches_consumed_stdin"] = True
        checks["single_native_exec_and_exit_status_preserved"] = True
        checks["indexed_untrusted_source_never_executed"] = True

    summary = {
        "passed": True,
        "package_location": str(package),
        "python_prefix": sys.prefix,
        "console_entrypoint": str(console),
        "isolated_imports": True,
        "native_working_directory": "/tmp",
        "functional_checks": checks,
        "functional_check_count": len(checks),
        "native_execs_observed": 1,
        "native_exit_status": 29,
        "model_requests": 0,
        "network_requests": 0,
        "prepared_stdin_bytes": len(delivered),
        "prepared_stdin_sha256": digest(delivered),
        "scope": "Installed-wheel API/console smoke with trusted fake native Codex; no token/cost measurements. Functional checks are not additional pytest cases.",
    }
    with report.open("x", encoding="utf-8") as output:
        json.dump(summary, output, sort_keys=True, separators=(",", ":"))
        output.write("\n")
    print(json.dumps(summary, sort_keys=True, separators=(",", ":")))


if __name__ == "__main__":
    main()
