REPO=Path(__file__).resolve().parents[2]
class InvalidEvidence(ValueError): pass

def require(condition,message):
    if not condition: raise InvalidEvidence(message)

def digest(data): return hashlib.sha256(data).hexdigest()
def finite_number(value):
    return isinstance(value,(int,float)) and not isinstance(value,bool) and math.isfinite(value) and value>=0

def read_file(name,evidence,private=False):
    path=REPO/name
    require(path.is_file() and not path.is_symlink(),f'Missing/nonregular evidence: {name}')
    body=path.read_bytes()
    evidence[name]={'path':name,'bytes':len(body),'sha256':digest(body),'public_copy_allowed':not private}
    return body

def read_json(name,evidence,private=False,prefixed=False):
    body=read_file(name,evidence,private).decode('utf-8')
    if prefixed:
        start=body.find('{');require(start>=0,f'No JSON object in {name}');body=body[start:]
    def invalid_constant(value): raise InvalidEvidence(f'Nonfinite JSON number in {name}')
    return json.loads(body,parse_constant=invalid_constant)

def junit(name,evidence,clean=True):
    tree=ET.fromstring(read_file(name,evidence))
    cases=list(tree.iter('testcase'));require(cases,f'No JUnit test cases: {name}')
    counts={'tests':len(cases),'passed':0,'failures':0,'errors':0,'skipped':0}
    grouped=collections.defaultdict(lambda:{'tests':0,'passed':0,'failures':0,'errors':0,'skipped':0,'testcase_seconds':0.0})
    identities=[]
    for case in cases:
        cls=case.attrib.get('classname','');file=case.attrib.get('file')
        if not file: file=(cls.replace('.','/')+'.py') if cls.startswith('tests.') else cls or '(unattributed)'
        status=next((status for status,tag in [('errors','error'),('failures','failure'),('skipped','skipped')] if case.find(tag) is not None),'passed')
        duration=float(case.attrib.get('time',0));require(finite_number(duration),f'Invalid test duration: {name}')
        counts[status]+=1;grouped[file]['tests']+=1;grouped[file][status]+=1;grouped[file]['testcase_seconds']+=duration
        identities.append({'file':file,'classname':cls,'case_identity_sha256':digest((cls+'::'+case.attrib.get('name','')).encode('utf-8')),'status':status,'seconds':duration})
    suites=[tree] if tree.tag=='testsuite' else list(tree.findall('testsuite'))
    seconds=sum(float(s.attrib.get('time',0)) for s in suites)
    require(finite_number(seconds),f'Invalid suite duration: {name}')
    if clean: require(counts['passed']==counts['tests'],f'Failing/error/skipped final suite: {name}: {counts}')
    return {'receipt':name,**counts,'reported_suite_seconds':seconds,'per_file':dict(sorted(grouped.items())),'cases':identities}

def browser(name,evidence):
    raw=read_json(name,evidence,private=True,prefixed=True)
    tests=[]
    def walk(suites,parents=()):
        for suite in suites:
            title=suite.get('title','');chain=(*parents,title) if title else parents
            for spec in suite.get('specs',[]):
                for test in spec.get('tests',[]):
                    results=[]
                    for item in test.get('results',[]):
                        errors=item.get('errors',[])
                        results.append({'status':item.get('status'),'duration_ms':item.get('duration'),'retry':item.get('retry',0),'errors_count':len(errors) if isinstance(errors,list) else 1,'error_present':bool(item.get('error'))})
                    tests.append({'title_path':[*chain,spec.get('title','')],'file':spec.get('file'),'line':spec.get('line'),'project_name':test.get('projectName',''),'expected_status':test.get('expectedStatus'),'reported_status':test.get('status'),'results':results})
            walk(suite.get('suites',[]),chain)
    walk(raw.get('suites',[]))
    require(tests,f'No browser tests: {name}')
    stats=raw.get('stats',{})
    require(not raw.get('errors'),f'Browser global errors present: {name}')
    for field in ('unexpected','skipped','flaky'): require(stats.get(field)==0,f'Browser {field} is nonzero or missing')
    require(stats.get('expected')==len(tests),'Browser stats/test-count mismatch')
    for test in tests:
        require(test['expected_status']=='passed' and test['reported_status']=='expected','Browser test did not pass cleanly')
        require(len(test['results'])==1,'Browser retry or missing result')
        item=test['results'][0]
        require(item['status']=='passed' and item['retry']==0 and item['errors_count']==0 and not item['error_present'],'Browser result failed, retried or had diagnostics')
        require(finite_number(item['duration_ms']),'Invalid browser duration')
    clean={'schema_version':1,'source_sha256':evidence[name]['sha256'],'source_bytes':evidence[name]['bytes'],'scope':'Allowlisted Playwright outcomes only; raw original stays private.','omission_note':'Confidential browser configuration/environment and arbitrary error message/stack/body fields are deliberately omitted. Test outcomes, counts, durations and retries are preserved. This is not omission of failed outcomes.','stats':{key:stats.get(key) for key in ('startTime','duration','expected','skipped','unexpected','flaky')},'global_error_count':0,'tests':tests}
    return clean

def check_receipts(name,evidence,required_names,allow_historical=False):
    data=read_json(name,evidence)
    rows=data.get('checks') if isinstance(data,dict) else data
    require(isinstance(rows,list) and rows,f'No command receipts: {name}')
    selected=[]
    by_name={row.get('name'):row for row in rows}
    require(len(by_name)==len(rows),f'Duplicate command receipt names: {name}')
    for label in required_names:
        require(label in by_name,f'Missing required command {label}: {name}')
        require(by_name[label].get('returncode')==0,f'Failed required command {label}: {name}')
    for row in rows:
        require(isinstance(row,dict) and isinstance(row.get('name'),str),'Invalid command receipt')
        code=row.get('returncode');require(type(code) is int,'Missing actual command returncode')
        if not allow_historical: require(code==0,f'Nonzero final command receipt: {row["name"]}')
        wall=row.get('wall_seconds')
        reason=row.get('wall_unavailable_reason') or row.get('wall_seconds_unavailable_reason')
        require(finite_number(wall) or (wall is None and isinstance(reason,str) and reason),'Missing command duration needs explicit unavailable reason')
        item={key:row.get(key) for key in ('name','returncode','wall_seconds','log','exit_provenance') if key in row}
        if wall is None:item['wall_unavailable_reason']=reason
        log=row.get('log')
        if log:
            read_file(log,evidence,private=log.endswith('browser.json'))
            if row.get('log_sha256'):require(evidence[log]['sha256']==row['log_sha256'],f'Command log hash mismatch: {log}')
        selected.append(item)
    return data,selected

def snapshot(root,extension=None):
    base=REPO/root
    require(base.is_dir(),f'Missing source directory: {root}')
    paths=sorted(p for p in base.rglob('*') if p.is_file() and (extension is None or p.suffix==extension))
    require(paths,f'Empty source scope: {root}')
    hashes={str(p.relative_to(REPO)):digest(p.read_bytes()) for p in paths}
    aggregate=digest(''.join(path+'\0'+value+'\n' for path,value in hashes.items()).encode())
    return {'files':hashes,'file_count':len(hashes),'aggregate_sha256':aggregate,'aggregate_algorithm':'sha256 of sorted relative_path + NUL + file_sha256 + LF, UTF-8'}
