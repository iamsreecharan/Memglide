"""Supplemental independent check of late-added native stream framing guards."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

BASE = Path('/workspace/Memglide/.local/daily-v1/formal-20-v1')
OUTPUT = Path('/workspace/Memglide/.local/daily-v1/review/formal-framing-audit-v1.json')


def main():
    metadata = json.loads((BASE / 'metadata.json').read_text())
    if not metadata.get('finished_at'):
        raise RuntimeError('Experiment not finished')
    rows = json.loads((BASE / 'runs.json').read_text())
    observations = []
    for row in rows:
        path = BASE / 'runs' / row['run_id'] / 'events.jsonl'
        if path.is_symlink() or not path.is_file():
            raise ValueError('Missing or unsafe native stream')
        prefix = []
        with path.open(encoding='utf-8') as stream:
            for _ in range(128):
                line = stream.readline(65_537)
                if not line:
                    break
                if len(line) > 65_536:
                    raise ValueError('Oversized native stream header')
                if not line.strip():
                    continue
                prefix.append(json.loads(line))
                if len(prefix) == 2:
                    break
        framing = bool(len(prefix) == 2 and isinstance(prefix[0], dict)
                       and prefix[0].get('type') == 'thread.started'
                       and isinstance(prefix[0].get('thread_id'), str)
                       and prefix[0]['thread_id'].strip()
                       and isinstance(prefix[1], dict) and prefix[1].get('type') == 'turn.started')
        native_types = [value.get('type') for value in prefix]
        recorded = row.get('event_stream_framing') or {}
        agrees = recorded.get('valid') is framing and recorded.get('prefix_types') == native_types
        completed_flags = (row.get('status') == 'completed' and row.get('treatment_valid') is True
                           and row.get('preload_treatment_fidelity', {}).get('valid') is True
                           and row.get('native_execution_validity', {}).get('valid') is True
                           and row.get('grader', {}).get('success') is True)
        observations.append({'run_id': row['run_id'], 'valid_framing': framing,
                             'prefix_types': native_types, 'recorded_framing_agrees': agrees,
                             'all_completed_flags_true': completed_flags})
    value = {'audited_at_utc': datetime.now(timezone.utc).isoformat(),
             'scope': 'Supplement to the unchanged successful 9,884-assertion primary audit; checks the late-added header and explicit completion flags',
             'pass': len(observations) == 80 and all(row['valid_framing'] and row['recorded_framing_agrees'] and row['all_completed_flags_true'] for row in observations),
             'original_count': len(observations),
             'valid_framing_count': sum(row['valid_framing'] for row in observations),
             'matching_recorded_framing_count': sum(row['recorded_framing_agrees'] for row in observations),
             'all_completion_flags_count': sum(row['all_completed_flags_true'] for row in observations),
             'auditor_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
             'runs': observations}
    with OUTPUT.open('x', encoding='utf-8') as stream:
        json.dump(value, stream, indent=2); stream.write('\n')
    print(json.dumps({key: item for key, item in value.items() if key != 'runs'}))
    return int(not value['pass'])


if __name__ == '__main__':
    raise SystemExit(main())
