#!/usr/bin/env bash
set -euo pipefail
cd /workspace/Memglide
export UV_CACHE_DIR=/workspace/Memglide/.cache/uv
package_venv=/workspace/Memglide/.local/refined-wheel-env
package_state=/workspace/Memglide/.local/refined-wheel-state
package_dist=/workspace/Memglide/.local/refined-dist
if [ -e "$package_venv" ] || [ -e "$package_state" ] || [ -e "$package_dist" ]; then
  echo "Use fresh output paths; package validation will not overwrite prior environments/artifacts." >&2
  exit 2
fi
uv run python scripts/bundle_frontend.py
uv build --out-dir "$package_dist"
uv run python .local/refinement-package/audit_artifacts.py --repository "$PWD" --dist "$package_dist" --report .local/refinement-package/audit.json
uv export --frozen --no-dev --no-emit-project --format requirements-txt --output-file .local/refinement-package/requirements.txt
uv venv --python 3.12.14 "$package_venv"
uv pip install --python "$package_venv/bin/python" --require-hashes --requirements .local/refinement-package/requirements.txt
uv pip install --python "$package_venv/bin/python" --no-deps "$package_dist"/*.whl
cd /tmp
"$package_venv/bin/python" -I /workspace/Memglide/.local/refinement-package/verify_installed.py --state "$package_state" --report /workspace/Memglide/.local/refinement-package/smoke.json
