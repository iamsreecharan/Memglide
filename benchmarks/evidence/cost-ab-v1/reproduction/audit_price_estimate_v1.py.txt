"""Independent post-finish public-price-equivalent calculation, never actual billing.

Reads only finished approved experiment JSON metadata/native usage and a passing
run-audit receipt. Does not import application/runner code or execute any model,
repository code, test, network request, credential or database access.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import math
import statistics
from datetime import datetime, timezone
from decimal import Decimal
from pathlib import Path

ROOT = Path('/workspace/Memglide')
ALLOWED = ROOT / '.local/cost-v1'
RATE_SHA = '4e89f76cb9d0d18058279bf00400034b5febd16f55488477eeef6fc43fbd8634'
MILLION = Decimal(1000000)
KEYS = ('input_tokens', 'cached_input_tokens', 'cache_write_input_tokens', 'output_tokens')


def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path, base):
    path, base = Path(path), Path(base).resolve()
    if not path.is_absolute():
        path = base / path
    if not path.resolve().is_relative_to(base):
        raise ValueError('Input escapes approved root')
    for part in (path, *path.parents):
        if part == base:
            break
        if part.is_symlink() or part.name in {'auth.json', 'codex-state', 'memglide.db'}:
            raise ValueError('Private runtime state or symlinks excluded')
    if not path.is_file() or path.stat().st_size > 128 * 1024 * 1024:
        raise ValueError('Input is not a bounded regular file')
    return path.read_bytes()


def document(path, base):
    return json.loads(read(path, base))


def cost(usage, rates):
    i, c, w, o = (usage[k] for k in KEYS)
    if not all(type(n) is int and n >= 0 for n in (i, c, w, o)) or i < c + w:
        raise ValueError('Usage input/cache-read/cache-write partition is invalid')
    return (Decimal(i - c - w) * rates['uncached_input']
            + Decimal(c) * rates['cached_input']
            + Decimal(w) * rates['cache_write_input']
            + Decimal(o) * rates['output']) / MILLION


def stats(values):
    known = sorted(v for v in values if isinstance(v, Decimal))
    return {'n': len(known), 'missing': len(values) - len(known),
            'mean': str(sum(known) / len(known)) if known else None,
            'median': str(statistics.median(known)) if known else None,
            'p95': str(known[max(0, math.ceil(len(known) * .95) - 1)]) if known else None,
            'min': str(min(known)) if known else None,
            'max': str(max(known)) if known else None,
            'known_total': str(sum(known)) if known else None,
            'total': str(sum(known)) if known and len(known) == len(values) else None}


def strict_lifecycle(events):
    """A completed usage stream may still belong to a failed coding task."""
    active, seen, count = False, False, 0
    for index, event in enumerate(events):
        kind = event.get('type')
        if kind == 'thread.started':
            if index or seen or not isinstance(event.get('thread_id'), str) or not event['thread_id']:
                return False
            seen = True
        elif kind == 'turn.started':
            if not seen or active:
                return False
            active = True
        elif kind == 'turn.completed':
            if not active:
                return False
            active, count = False, count + 1
        elif kind in {'item.started', 'item.updated', 'item.completed'}:
            if not active:
                return False
        else:
            return False
    return bool(seen and count and not active and events[-1].get('type') == 'turn.completed')


def calculate(experiment, run_audit, pricing):
    audit = document(run_audit, ALLOWED)
    if audit.get('pass') is not True or audit.get('observed_attempts') != 20:
        raise ValueError('A passing full twenty-attempt independent run audit is required')
    metadata = document(experiment / 'metadata.json', experiment)
    report_bytes = read(experiment / 'report.json', experiment)
    report = json.loads(report_bytes)
    if not metadata.get('finished_at') or audit['source_report_sha256'] != sha(report_bytes):
        raise ValueError('Experiment must be finished and match the passing independent audit')
    price_bytes = read(pricing, ROOT)
    if sha(price_bytes) != RATE_SHA:
        raise ValueError('Explicit official-rate artifact hash differs')
    price = json.loads(price_bytes)
    if price['model'] != metadata['model'] or price['service_tier'] != 'standard':
        raise ValueError('No exact cited model/standard rate match')
    rates = {k: Decimal(str(v)) for k, v in price['per_million_usd'].items()}
    evidence = {row['run_id']: row for row in audit['runs']}
    rows, notes, errors = [], [], []
    for original in report['runs']:
        if not original.get('attempted'):
            continue
        run_id = original['run_id']
        folder = experiment / 'runs' / run_id
        native = read(folder / 'events.jsonl', experiment)
        events = [json.loads(line) for line in native.splitlines() if line.strip()]
        completed = [e for e in events if e.get('type') == 'turn.completed']
        usage = {}
        for key in KEYS:
            counters = [e.get('usage', {}).get(key) for e in completed]
            usage[key] = sum(counters) if counters and all(type(n) is int and n >= 0 for n in counters) else None
        reasoning = [e.get('usage', {}).get('reasoning_output_tokens') for e in completed]
        reasoning_total = sum(reasoning) if reasoning and all(type(n) is int and n >= 0 for n in reasoning) else None
        for key in ('input_tokens', 'cached_input_tokens', 'output_tokens'):
            if usage[key] != evidence[run_id]['metrics'][key]:
                errors.append({'code': 'native_usage_differs_from_run_audit', 'run_id': run_id, 'key': key})
        terminal_valid = (strict_lifecycle(events)
                          and original.get('execution_outcome_unknown') is not True
                          and (original.get('telemetry') or {}).get('usage_coverage_complete') is True
                          )
        # Aggregate input below the boundary rules out any constituent request
        # exceeding it. Above it, aggregate usage does not reveal request sizes.
        short_proven = (type(usage['input_tokens']) is int
                        and usage['input_tokens'] <= price['context_threshold_tokens'])
        point = upper = lower = None
        if terminal_valid and short_proven:
            point = cost(usage, rates) if usage['cache_write_input_tokens'] is not None else None
            low_usage = {**usage, 'cache_write_input_tokens': usage['cache_write_input_tokens'] or 0}
            high_usage = {**usage, 'cache_write_input_tokens': usage['input_tokens'] - usage['cached_input_tokens']}
            # W=0 may be a parser default. The native-reported point and the
            # independently unknown hidden-write interval are kept separate.
            lower, upper = cost(low_usage, rates), cost(high_usage, rates)
        else:
            notes.append({'code': 'cost_unknown_invalid_native_or_per_request_tariff_not_classifiable', 'run_id': run_id})
        rows.append({'run_id': run_id, 'task_id': original['task_id'],
                     'configuration': original['configuration'], 'status': original['status'],
                     'successful_completion': evidence[run_id]['successful_completion'],
                     'native_events_sha256': sha(native), 'native_reported_usage': usage,
                     'native_reported_reasoning_output_tokens': reasoning_total,
                     'short_context_proven_from_aggregate_upper_bound': short_proven,
                     'native_reported_write_API_price_equivalent_usd': str(point) if point is not None else None,
                     'hypothetical_unknown_write_lower_usd': str(lower) if lower is not None else None,
                     'hypothetical_all_noncached_written_upper_usd': str(upper) if upper is not None else None,
                     'actual_billing_usd': None})
    arms = {}
    fields = ('native_reported_write_API_price_equivalent_usd',
              'hypothetical_unknown_write_lower_usd', 'hypothetical_all_noncached_written_upper_usd')
    for arm in ('A', 'B'):
        attempts = [row for row in rows if row['configuration'] == arm]
        successes = sum(row['successful_completion'] for row in attempts)
        summaries = {}
        for key in fields:
            values = [Decimal(row[key]) if row[key] is not None else None for row in attempts]
            summary = stats(values)
            summary['per_successful_task_usd'] = str(Decimal(summary['total']) / successes) if summary['total'] is not None and successes else None
            summaries[key] = summary
        arms[arm] = {'attempted': len(attempts), 'successful': successes, 'estimates': summaries,
                     'actual_billing_usd': None, 'actual_cost_per_successful_task_usd': None}
    point_key, low_key, high_key = fields
    def per_success(arm, key):
        raw = arms[arm]['estimates'][key]['per_successful_task_usd']
        return Decimal(raw) if raw is not None else None
    ap, bp = per_success('A', point_key), per_success('B', point_key)
    al, au = per_success('A', low_key), per_success('A', high_key)
    bl, bu = per_success('B', low_key), per_success('B', high_key)
    point_savings = Decimal(100) * (1 - bp / ap) if ap and bp is not None else None
    lower_savings = Decimal(100) * (1 - bu / al) if al and bu is not None else None
    upper_savings = Decimal(100) * (1 - bl / au) if au and bl is not None else None
    pairs = []
    for task in report['tasks']:
        pair = {row['configuration']: row for row in rows if row['task_id'] == task['id']}
        a, b = (Decimal(pair[arm][point_key]) if pair[arm][point_key] is not None else None for arm in ('A', 'B'))
        pairs.append({'task_id': task['id'], 'A_usd': str(a) if a is not None else None,
                      'B_usd': str(b) if b is not None else None,
                      'B_minus_A_usd': str(b - a) if a is not None and b is not None else None,
                      'both_successful': all(pair[arm]['successful_completion'] for arm in ('A', 'B'))})
    return {'schema_version': 1, 'created_at_utc': datetime.now(timezone.utc).isoformat(),
            'pass': not errors, 'errors': errors, 'notes': notes,
            'estimate_kind': 'Independent API list-price-equivalent estimate; not actual subscription/account billing',
            'pricing_sha256': sha(price_bytes), 'source_report_sha256': sha(report_bytes),
            'run_audit_sha256': sha(read(run_audit, ALLOWED)),
            'auditor_sha256': sha(Path(__file__).read_bytes()), 'pricing': price,
            'arms': arms, 'runs': rows, 'pairs': pairs,
            'native_reported_write_cost_per_success_savings_percent': str(point_savings) if point_savings is not None else None,
            'independently_unknown_write_savings_interval_percent': {
                'lower': str(lower_savings) if lower_savings is not None else None,
                'upper': str(upper_savings) if upper_savings is not None else None},
            'limitations': ['All attempted model usage is charged in the numerator, including failed tasks; only validated successful completions enter the denominator.',
                            'Reported zero cache-write tokens may be parser defaults. Sensitivities are hypothetical independent billing scenarios, not measured charges.',
                            'No reasoning-output surcharge is added because output tokens already include it.',
                            'No billing, local-compute/hosting or external-tool invoices are available. Actual total cost and actual savings remain unknown.',
                            'Task aggregate above the long-context boundary cannot classify per-request billing and is left unknown.']}


def verify_assessment(independent, assessment_path):
    """Compare the separately implemented tracked estimator, never execute it."""
    raw = read(assessment_path, ROOT)
    actual = json.loads(raw)
    errors, checks = [], 0

    def check(condition, code, run_id=None):
        nonlocal checks
        checks += 1
        if not condition:
            errors.append({'code': code, **({'run_id': run_id} if run_id else {})})

    def number(actual_value, expected_value, code, run_id=None):
        if expected_value is None:
            check(actual_value is None, code, run_id)
            return
        if type(actual_value) not in (int, float) or not math.isfinite(actual_value):
            check(False, code, run_id)
            return
        expected = Decimal(str(expected_value))
        check(abs(Decimal(str(actual_value)) - expected) <= max(Decimal('1e-12'), abs(expected) * Decimal('1e-12')),
              code, run_id)

    check(actual.get('assessment_version') == 'codex-cost-1', 'assessment_version')
    check(actual.get('pricing') == independent['pricing'], 'exact_frozen_pricing_document')
    check(actual.get('actual_billed_cost_usd') is None, 'actual_billing_remains_unknown')
    check(actual.get('provenance', {}).get('primary_report_sha256') == independent['source_report_sha256'],
          'assessment_bound_to_exact_primary_report')
    check(actual.get('provenance', {}).get('pricing_document_sha256') == independent['pricing_sha256'],
          'assessment_bound_to_exact_pricing_document')
    check(actual.get('provenance', {}).get('estimator_sha256') == sha(read(ROOT / 'benchmarks/estimate_cost.py', ROOT)),
          'assessment_bound_to_exact_estimator_source')
    by_id = {row['run_id']: row for row in actual['runs']}
    check(len(actual['runs']) == len(by_id) == len(independent['runs']) == 20
          and set(by_id) == {row['run_id'] for row in independent['runs']}, 'all_twenty_attempts_preserved')
    fields = {'native_reported_write_API_price_equivalent_usd': 'native_reported_price_equivalent_usd',
              'hypothetical_unknown_write_lower_usd': 'possible_write_price_lower_usd',
              'hypothetical_all_noncached_written_upper_usd': 'possible_write_price_upper_usd'}
    for row in independent['runs']:
        run_id = row['run_id']
        other = by_id[run_id]
        check(other['status'] == row['status'] and other['success'] is row['successful_completion'],
              'success_denominator_requires_valid_completed_external_grade', run_id)
        check(other['events_sha256'] == row['native_events_sha256'], 'exact_native_stream_digest', run_id)
        check(other['cost']['actual_billed_cost_usd'] is None, 'individual_actual_billing_unknown', run_id)
        check(other['primary_usage_reconciled'] is True, 'primary_native_usage_reconciled', run_id)
        for key, value in row['native_reported_usage'].items():
            number(other['native_usage']['usage'][key], value, 'raw_native_usage:' + key, run_id)
        number(other['native_usage']['usage']['reasoning_output_tokens'],
               row['native_reported_reasoning_output_tokens'], 'reasoning_is_preserved_without_double_billing', run_id)
        for expected_key, actual_key in fields.items():
            number(other['cost'][actual_key], row[expected_key], 'individual_cost:' + actual_key, run_id)
    for arm in ('A', 'B'):
        other, expected = actual['configurations'][arm], independent['arms'][arm]
        check(other['attempts'] == other['assignments'] == expected['attempted'] == 10,
              'all_attempts_and_assignments_in_cost_numerator:' + arm)
        check(other['successful_assignments'] == other['successful_attempts'] == expected['successful'],
              'distinct_successful_assignments_in_denominator:' + arm)
        check(other['actual_billed_cost_usd'] is None, 'arm_actual_billing_unknown:' + arm)
        for expected_key, actual_key in fields.items():
            for field, value in expected['estimates'][expected_key].items():
                if field == 'per_successful_task_usd':
                    number(other['cost_per_successful_task_usd'][actual_key], value,
                           'cost_per_successful_task:' + arm + ':' + actual_key)
                else:
                    number(other['cost_statistics'][actual_key][field], value,
                           'arm_cost_statistics:' + arm + ':' + actual_key + ':' + field)
    number(actual['cost_per_success_saving_percent']['native_reported_price_equivalent_usd'],
           independent['native_reported_write_cost_per_success_savings_percent'], 'point_savings_percent')
    interval = independent['independently_unknown_write_savings_interval_percent']
    number(actual['write_uncertainty_saving_range']['lower_saving_percent'], interval['lower'], 'independent_write_lower_savings_scenario')
    number(actual['write_uncertainty_saving_range']['upper_saving_percent'], interval['upper'], 'independent_write_upper_savings_scenario')
    pairs = {row['task_id']: row for row in actual['pairs']}
    check(len(actual['pairs']) == len(pairs) == len(independent['pairs']) == 10, 'all_ten_task_pairs_preserved')
    for expected in independent['pairs']:
        other = pairs[expected['task_id']]
        for arm in ('A', 'B'):
            number(other['arms'][arm]['cost_statistics']['native_reported_price_equivalent_usd']['total'],
                   expected[arm + '_usd'], 'paired_arm_cost:' + arm, expected['task_id'])
        number(other['B_minus_A_usd'], expected['B_minus_A_usd'], 'paired_cost_delta', expected['task_id'])
    expected_failures = {row['run_id'] for row in independent['runs'] if not row['successful_completion']}
    check({row['run_id'] for row in actual['failures']} == expected_failures, 'every_failed_attempt_retained')
    return {'pass': not errors, 'checked_assertions': checks, 'errors': errors,
            'assessment_sha256': sha(raw), 'assessment_path': str(assessment_path)}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--after-finished', action='store_true')
    parser.add_argument('--experiment', type=Path, required=True)
    parser.add_argument('--run-audit', type=Path, required=True)
    parser.add_argument('--pricing', type=Path, default=ROOT / 'benchmarks/cost-pricing.json')
    parser.add_argument('--assessment', type=Path)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    if not args.after_finished:
        parser.error('Explicit --after-finished required; no experimental streams opened')
    if not args.output.resolve().is_relative_to(ALLOWED) or args.output.exists():
        parser.error('A fresh ignored .local/cost-v1 output is required')
    if not args.experiment.resolve().is_relative_to(ALLOWED):
        parser.error('Only the approved .local/cost-v1 experiment is readable')
    try:
        value = calculate(args.experiment.resolve(), args.run_audit, args.pricing)
        if args.assessment:
            value['assessment_verification'] = verify_assessment(value, args.assessment)
            value['pass'] = value['pass'] and value['assessment_verification']['pass']
    except Exception as error:
        value = {'schema_version': 1, 'pass': False, 'audit_exception_type': type(error).__name__,
                 'audit_exception_message': str(error)}
    with args.output.open('x') as stream:
        json.dump(value, stream, indent=2); stream.write('\n')
    print(json.dumps({'pass': value['pass'], 'output': str(args.output),
                      'exception': value.get('audit_exception_type')}))
    return int(not value['pass'])


if __name__ == '__main__':
    raise SystemExit(main())
