"""Add descriptive, post-measurement interpretation without changing raw evidence."""

import hashlib
import json
from decimal import Decimal
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
EXPERIMENT = ROOT / ".local/cost-v1/formal-10-v1"
ASSESSMENT = ROOT / "benchmarks/cost-assessment.json"
PRIMARY = ROOT / "benchmarks/codex-cost-ab.json"


def sha(content):
    return hashlib.sha256(content).hexdigest()


def read(path):
    return json.loads(path.read_text())


def number(value):
    return Decimal(str(value))


primary = read(PRIMARY)
assessment = read(ASSESSMENT)
assert sha(PRIMARY.read_bytes()) == assessment["provenance"]["primary_report_sha256"]
assert sha(PRIMARY.read_bytes()) == "de4067e198d3ad573ef633b8d2a47c45134092b5284b6ec67dacc7202d2cb8af"
assert len(assessment["runs"]) == 20
assert len(primary["runs"]) == 20
assert all(row["success"] for row in assessment["runs"])
assert all(row["primary_usage_reconciled"] for row in assessment["runs"])
assert primary["metadata"]["resume_count"] == 0
assert primary["metadata"]["source_unchanged_during_experiment"] is True
plan = read(EXPERIMENT / "plan.json")
saved_runs = read(EXPERIMENT / "runs.json")
supplemental = read(EXPERIMENT / "supplemental-runs.json")
assert len(plan["runs"]) == 20
assert not supplemental
assert {row["run_id"] for row in saved_runs} == {row["run_id"] for row in primary["runs"]}
assert not any(row.get("supplemental") for row in primary["runs"])
assert all(row["original_run_id"] == row["run_id"] for row in primary["runs"])

raw_index = {(row["task_id"], row["configuration"]): row for row in primary["runs"]}
controls = [task for task in primary["tasks"] if task["context_supplied"]]
assert len(controls) == 2
assert all(raw_index[(task["id"], "A")]["actual_model_prompt_sha256"] ==
           raw_index[(task["id"], "B")]["actual_model_prompt_sha256"] for task in controls)

rates = {key: number(value) for key, value in assessment["pricing"]["per_million_usd"].items()}
exact_cost = {}
for row in assessment["runs"]:
    usage = row["native_usage"]["usage"]
    i, c, w, o = (usage[key] for key in ("input_tokens", "cached_input_tokens", "cache_write_input_tokens", "output_tokens"))
    exact_cost[row["run_id"]] = ((i-c-w)*rates["uncached_input"]+c*rates["cached_input"]+w*rates["cache_write_input"]+o*rates["output"])/1_000_000
pair_deltas = []
for pair in assessment["pairs"]:
    a, b = (exact_cost[raw_index[(pair["task_id"], arm)]["run_id"]] for arm in "AB")
    pair_deltas.append({"task_id": pair["task_id"], "B_minus_A_usd": b-a})
net_delta = sum((pair["B_minus_A_usd"] for pair in pair_deltas), Decimal(0))
best = min(pair_deltas, key=lambda pair: pair["B_minus_A_usd"])
worst = max(pair_deltas, key=lambda pair: pair["B_minus_A_usd"])
assert net_delta == Decimal("-0.0089312")
strata = assessment["strata"]["selection_stratum"]
focused = strata["explicit_focus"]
supplied = strata["caller_context_supplied"]
focused_delta = sum((exact_cost[row["run_id"]] * (1 if row["configuration"] == "B" else -1)
                    for row in assessment["runs"] if row["selection_stratum"] == "explicit_focus"), Decimal(0))
controls_delta = net_delta-focused_delta
assert focused_delta == Decimal("0.014298")
assert controls_delta == Decimal("-0.0232292")

scope = {
    "started_at": primary["metadata"]["started_at"],
    "finished_at": primary["metadata"]["finished_at"],
    "frozen_git_head": primary["metadata"]["frozen_manifest"]["git_head"],
    "resume_count": primary["metadata"]["resume_count"],
    "source_unchanged_during_experiment": True,
    "planned_original_assignments": len(plan["runs"]),
    "provided_attempts": len(assessment["runs"]),
    "saved_original_attempts": len(saved_runs),
    "supplemental_attempts": len(supplemental),
    "failures": 0,
    "retries": 0,
    "valid_complete_reconciled_native_streams": 20,
    "native_cache_write_field_present_runs": sum(row["metrics"]["cache_write_input_tokens"] is not None for row in assessment["runs"]),
    "native_reasoning_field_present_runs": sum(row["metrics"]["reasoning_output_tokens"] is not None for row in assessment["runs"]),
    "controls_have_byte_identical_model_prompts": True,
    "focused_tasks": 8,
    "supplied_context_control_tasks": 2,
    "model_requests_for_accounting": 0,
    "attempt_coverage_artifacts": {
        name: {"bytes": len((EXPERIMENT/name).read_bytes()), "sha256": sha((EXPERIMENT/name).read_bytes())}
        for name in ("plan.json", "runs.json", "supplemental-runs.json", "state.json")
    },
    "attempt_coverage_scope": "All 20 planned originals match durable saved runs and the primary report; supplementary ledger is empty. Separate independent runtime audit checks the durable event chain.",
}
assessment["measurement_scope"] = scope
assessment["interpretation"] = {
    "conclusion": "The pooled native-reported API-price equivalent is 2.28% lower for B, but Memglide-preloaded tasks are 4.43% more expensive. Identical-prompt controls account for the pooled saving. This study does not establish money savings caused by Memglide.",
    "overall_B_minus_A_total_usd": float(net_delta),
    "focused_B_minus_A_total_usd": float(focused_delta),
    "controls_B_minus_A_total_usd": float(controls_delta),
    "controls_interpretation": "The two controls bypass Memglide source preparation and deliver byte-identical prompts. Their 33.27% price-equivalent difference demonstrates uncontrolled run/cache variation; it cannot be credited to retrieval or source preparation.",
    "focused_interpretation": "Across eight preloaded tasks, mean price equivalent is $0.04209265 with Memglide versus $0.0403054 with bare Codex, a 4.43% increase. Four single-source pairs are descriptively 24.14% cheaper while four integration pairs are 26.08% dearer; one trial per task cannot establish those mechanisms or predict future work.",
    "paired_interpretation": "B is cheaper in five pairs and dearer in five. All pairs are retained. Individual favorable and unfavorable deltas exceed the net pooled saving in magnitude, so the small aggregate is sensitive to individual tasks.",
    "largest_favorable_pair": {
        "task_id": best["task_id"],
        "B_minus_A_usd": float(best["B_minus_A_usd"]),
        "fraction_of_net_saving_percent": float(best["B_minus_A_usd"]/net_delta*100),
    },
    "largest_unfavorable_pair": {
        "task_id": worst["task_id"],
        "B_minus_A_usd": float(worst["B_minus_A_usd"]),
        "offset_as_percent_of_net_saving_magnitude": float(worst["B_minus_A_usd"]/abs(net_delta)*100),
    },
    "B_minus_A_cost_component_usd": {
        key: float(sum((exact_cost[row["run_id"]]*0 for row in []), Decimal(0)))
        for key in ()
    },
    "cost_change_explanation": "902 fewer output tokens reduce the price equivalent by $0.00902. 134 more ordinary uncached tokens add $0.000268; 1,792 fewer cached tokens remove $0.0001792. These sum to $0.0089312 less over ten successes, mainly the output difference. Native-reported writes are zero in every run, subject to producer-default ambiguity.",
    "recommendation": "Treat compact source preparation as an experimental option; do not advertise proven cost savings. Repeat fresh paired workloads, especially integrations and identical-context controls, before choosing a default or making causal economic claims.",
    "statistical_scope": "One balanced repeat of ten small synthetic tasks; no statistical evidence of general savings. Independent write-attribution sensitivity spans a 15.47% increase through a 16.99% saving and is not a confidence interval.",
}
assessment["interpretation"]["B_minus_A_cost_component_usd"] = {
    "uncached_input": 0.000268,
    "cached_input": -0.0001792,
    "cache_write_input": 0,
    "output": -0.00902,
}
assessment["provenance"]["interpretation_script_sha256"] = sha(Path(__file__).read_bytes())
ASSESSMENT.write_text(json.dumps(assessment,indent=2,ensure_ascii=False)+"\n")

markdown_path=ASSESSMENT.with_suffix(".md")
markdown=markdown_path.read_text()
interpretation="""The pooled native-reported estimate is **2.28% lower**, but the **eight tasks that actually preload source cost 4.43% more**. The two controls with byte-identical prompts cost 33.27% less in B and account for the pooled saving. Their difference reflects uncontrolled run/cache variation and cannot be credited to Memglide source preparation. This study does not establish money savings caused by Memglide.

| Scope | A USD / successful task | B USD / successful task | B versus A |
|---|---:|---:|---:|
| All 10 tasks | $0.03922680 | $0.03833368 | 2.28% lower |
| 8 source-preloaded tasks | $0.04030540 | $0.04209265 | 4.43% higher |
| 2 identical-prompt controls | $0.03491240 | $0.02329780 | 33.27% lower |

All 20 assignments completed and passed independent grading; all native usage streams reconcile with the primary report. There were no failed attempts, retries, resumptions or supplementary runs. Measurement finished at 2026-10-18 16:41:32 UTC with frozen source unchanged.

B costs less in five pairs and more in five. The largest favorable pair, `ts-cost-ledger-signed-cents`, saves $0.0183468, **205.42% of the $0.0089312 net saving**. The largest unfavorable pair, `ts-cost-geometry-normalized-bounds`, adds $0.0234092, offsetting 262.11% of that net amount. Every pair remains included. One repeat on ten small synthetic tasks provides no statistical proof of general savings.

The dollar difference mainly comes from 902 fewer output tokens ($0.00902 less). Uncached input rises by 134 tokens ($0.000268 more), while cached input falls by 1,792 tokens ($0.0001792 less). Native-reported cache writes are zero in every run; an omitted upstream value can decode to zero. Allowing independent possible writes produces a range from **15.47% more expensive to 16.99% cheaper**, under the same Standard tariff. This is a scenario range, not measured billing or a confidence interval.

Four single-source pairs are descriptively 24.14% cheaper in B; four integration pairs are 26.08% more expensive. Repeated trials are needed to separate source preparation from model/cache variation. Keep the integration experimental and collect stronger evidence before claiming cost savings.

"""
needle="Native usage is independently parsed"
assert needle in markdown
markdown_path.write_text(markdown.replace(needle,interpretation+needle,1))
print(json.dumps({"assessment_sha256":sha(ASSESSMENT.read_bytes()),"markdown_sha256":sha(markdown_path.read_bytes()),"net_saving_usd":str(-net_delta),"scope":scope},indent=2))
