#!/usr/bin/env python3
"""Package explicit, already-public cost-study evidence into a fresh ZIP.

No model requests, tests, private state, recursive source copies or overwrites.
The manifest and scanner hashes must be supplied from the publisher's independent
receipt, and all selected bytes are verified and scanned before opening a ZIP.
"""

from __future__ import annotations

import argparse
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import re

import sys
import types
import zipfile

ROOT = Path(__file__).resolve().parents[2]
EVIDENCE = ROOT / "benchmarks/evidence/cost-ab-v1"
OUTPUT = ROOT / "benchmarks/memglide-cost-ab-report.zip"
RECEIPT = ROOT / "benchmarks/cost-report-bundle.json"
FROZEN = "UNCONFIGURED"
APP_SHA = "UNCONFIGURED"
REPORTS = (
    "benchmarks/codex-cost-ab.md", "benchmarks/codex-cost-ab.json",
    "benchmarks/cost-assessment.md", "benchmarks/cost-assessment.json",
    "benchmarks/cost-pricing.json", "benchmarks/cost-pricing-research.md", "benchmarks/cost-pricing-research.json",

)

HISTORICAL = ()
FROZEN_FILES = (

    "benchmarks/run_cost.py", "benchmarks/corpus_cost.py", "benchmarks/tasks_cost.json",
    "benchmarks/cost-pricing.json", "benchmarks/cost-protocol.md",
    "benchmarks/run_cache_formats.py", "benchmarks/corpus_cache.py", "benchmarks/run_daily.py", "benchmarks/corpus_daily.py",

    "benchmarks/telemetry.py", "benchmarks/compare_components.py", "pyproject.toml", "uv.lock",
    "tests/test_benchmark_cost_runner.py", "tests/test_benchmark_corpus_cost.py",
)

ACCOUNTING_FILES = ("benchmarks/estimate_cost.py", "tests/test_benchmark_cost_estimate.py")

def fail(message: str) -> None:
    raise SystemExit(message)


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def encoded(value: object) -> bytes:
    return (json.dumps(value, indent=2, sort_keys=True) + "\n").encode()


def unique_json(data: bytes) -> object:
    def pairs(items: list[tuple[str, object]]) -> dict:
        result = {}
        for key, value in items:
            if key in result:
                fail("Duplicate JSON key")
            result[key] = value
        return result
    return json.loads(data, object_pairs_hook=pairs)


def safe_name(value: str) -> str:
    p = PurePosixPath(value)
    if not value or "\\" in value or p.is_absolute() or str(p) != value or any(
        item in {"", ".", "..", "auth.json", "codex-state", "worktrees", "repositories", "state"}
        for item in p.parts
    ) or p.suffix.lower() in {".db", ".sqlite", ".sqlite3", ".tar", ".tgz", ".gz", ".zip", ".whl", ".source"}:
        fail("Unsafe archive member name")
    return value


def reject_symlinks(path: Path) -> None:
    if not path.is_relative_to(ROOT):
        fail("Selected path outside repository")
    for part in (path, *path.parents):
        if part.is_symlink():
            fail("Symlink in selected path")
        if part == ROOT:
            return
    fail("Selected path does not terminate at repository root")


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--manifest-sha256", required=True)
    parser.add_argument("--expected-head", required=True)
    parser.add_argument("--expected-app", required=True)
    parser.add_argument("--primary-sha256", required=True)
    parser.add_argument("--primary-md-sha256", required=True)
    parser.add_argument("--assessment-sha256", required=True)
    parser.add_argument("--scanner-sha256", required=True)
    parser.add_argument("--evidence-files", type=int, required=True)
    parser.add_argument("--evidence-bytes", type=int, required=True)
    parser.add_argument("--check-only", action="store_true")
    args = parser.parse_args()
    global FROZEN, APP_SHA
    FROZEN, APP_SHA = args.expected_head, args.expected_app
    if not re.fullmatch(r"[0-9a-f]{40}", FROZEN):
        fail("Expected full frozen Git SHA")
    if not all(re.fullmatch(r"[0-9a-f]{64}", value) for value in (args.manifest_sha256, args.scanner_sha256, args.expected_app, args.primary_sha256, args.primary_md_sha256, args.assessment_sha256)):
        fail("Expected full SHA-256 values")
    if OUTPUT.exists() or OUTPUT.is_symlink() or RECEIPT.exists() or RECEIPT.is_symlink():
        fail("Refusing to overwrite an existing ZIP or receipt")
    reject_symlinks(OUTPUT.parent)
    manifest_path = EVIDENCE / "manifest.json"
    reject_symlinks(manifest_path)
    manifest_bytes = manifest_path.read_bytes()
    if sha(manifest_bytes) != args.manifest_sha256:
        fail("Evidence manifest hash differs from independent publisher receipt")
    manifest = unique_json(manifest_bytes)
    records = manifest["files"]
    if len(records) != args.evidence_files or sum(item["bytes"] for item in records) != args.evidence_bytes:
        fail("Evidence count/bytes differ from independent publisher receipt")
    scanner_relative = "reproduction/publish_evidence.py.txt"
    scanner_record = [item for item in records if item["path"] == scanner_relative]
    if len(scanner_record) != 1 or scanner_record[0]["sha256"] != args.scanner_sha256:
        fail("Scanner identity differs from independent publisher receipt")
    scanner_path = EVIDENCE / scanner_relative
    reject_symlinks(scanner_path)
    scanner_source = scanner_path.read_bytes()
    if sha(scanner_source) != args.scanner_sha256 or len(scanner_source) != scanner_record[0]["bytes"]:
        fail("Pinned scanner bytes differ")
    # In-memory compilation prevents __pycache__ writes to immutable evidence.
    guard = types.ModuleType("frozen_cost_bundle_scanner")
    guard.__file__ = str(scanner_path)
    sys.modules[guard.__name__] = guard
    exec(compile(scanner_source, str(scanner_path), "exec"), guard.__dict__)
    guard.ROOT = ROOT
    if manifest["source_freeze"]["git_head"] != FROZEN or manifest["source_freeze"]["application_source_tree_sha256"] != APP_SHA:
        fail("Manifest frozen source identity differs")
    guard.FROZEN_COMMIT = FROZEN
    guard.FROZEN_APP = APP_SHA
    guard.XML_FROZEN_COMMIT = FROZEN
    exceptions_config = manifest["whitelist"]["explicit_extra_artifacts"]
    guard.XML_APPROVALS = exceptions_config.get("xml_approvals", {})

    def scan(data: bytes, label: str) -> None:
        try:
            data.decode("utf-8")
        except UnicodeError:
            fail("Nontext selected artifact: " + label)
        if b"\0" in data or any(pattern.search(data) for pattern in guard.SECRET_PATTERNS):
            fail("Potential credential or binary content: " + label)

    scan(manifest_bytes, "evidence manifest")
    members: dict[str, bytes] = {}
    origins: dict[str, dict] = {}
    file_identities: dict[Path, tuple[int, str]] = {}

    def read(path: Path) -> bytes:
        reject_symlinks(path)
        guard.safe_source(path)
        data = path.read_bytes()
        file_identities[path] = (len(data), sha(data))
        return data

    def add(name: str, data: bytes, origin: dict, *, scanned: bool = False) -> None:
        name = safe_name(name)
        if name in members:
            fail("Duplicate selected ZIP member")
        if not scanned:
            scan(data, name)
        members[name] = data
        origins[name] = origin

    expected = {"manifest.json", "publication-audit.json"}
    for record in records:
        relative = safe_name(record["path"])
        if relative in expected:
            fail("Duplicate or reserved evidence manifest path")
        expected.add(relative)
        public_path = EVIDENCE / relative
        data = read(public_path)
        if len(data) != record["bytes"] or sha(data) != record["sha256"]:
            fail("Published evidence SHA/size mismatch: " + relative)
        # The pinned scanner checks XML's exact original SHA and masks only
        # proven frozen synthetic fixture literals inside testcase-name spans.
        guard.scan_text_bytes(ROOT / record["source"], data)
        add(str(public_path.relative_to(ROOT)), data,
            {"kind": "manifested_public_evidence", "source": str(public_path.relative_to(ROOT)),
             "original_source": record["source"]}, scanned=True)

    actual = set()
    for directory, dirs, files in os.walk(EVIDENCE, followlinks=False):
        for name in dirs:
            reject_symlinks(Path(directory) / name)
        for name in files:
            path = Path(directory) / name
            reject_symlinks(path)
            actual.add(str(path.relative_to(EVIDENCE)))
    if actual != expected:
        fail("Published evidence tree contains missing or unmanifested files")
    exceptions = sorted(guard.XML_SCAN_EXCEPTIONS.values(), key=lambda item: item["path"])
    if exceptions != sorted(manifest["synthetic_fixture_scan_exceptions"], key=lambda item: item["path"]):
        fail("Synthetic XML exceptions differ from publisher record")
    selected_manifest_bytes = read(manifest_path)
    if selected_manifest_bytes != manifest_bytes:
        fail("Pinned evidence manifest changed during selection")
    add(str(manifest_path.relative_to(ROOT)), selected_manifest_bytes, {"kind": "pinned_manifest", "sha256": args.manifest_sha256})
    audit_path = EVIDENCE / "publication-audit.json"
    audit_bytes = read(audit_path)
    audit = unique_json(audit_bytes)
    if audit.get("pass") is not True or audit["manifest_sha256"] != args.manifest_sha256 or audit["selected_files"] != args.evidence_files or audit["bytes"] != args.evidence_bytes:
        fail("Publication audit does not validate selected manifest")
    add(str(audit_path.relative_to(ROOT)), audit_bytes, {"kind": "publication_audit"})

    for relative in REPORTS + CURRENT_DOCS + HISTORICAL:
        data = read(ROOT / relative)
        if relative in HISTORICAL:
            historical = subprocess.check_output(["git", "show", f"{FROZEN}:{relative}"], cwd=ROOT, stderr=subprocess.PIPE)
            if historical != data:
                fail("Immutable historical report or receipt changed")
        add(relative, data, {"kind": "historical_summary_only" if relative in HISTORICAL else "finalized_report_or_docs", "source": relative})
    for relative in FROZEN_FILES:
        data = subprocess.check_output(["git", "show", f"{FROZEN}:{relative}"], cwd=ROOT, stderr=subprocess.PIPE)
        add("frozen-source/" + relative + ".txt", data, {"kind": "frozen_git_source_text", "commit": FROZEN, "path": relative})

    for relative in ACCOUNTING_FILES:
        add("accounting-source/" + relative + ".txt", read(ROOT / relative),
            {"kind": "reviewed_post_freeze_accounting_source_text", "source": relative,
             "note": "Derived accounting only; not loaded in measured model runs or claimed frozen in experiment commit"})

    report_bytes = members["benchmarks/codex-cost-ab.json"]
    report = unique_json(report_bytes)
    assessment_bytes = members["benchmarks/cost-assessment.json"]
    assessment = unique_json(assessment_bytes)
    if sha(report_bytes) != args.primary_sha256 or sha(members["benchmarks/codex-cost-ab.md"]) != args.primary_md_sha256:
        fail("Primary report changed from sealed measured output")
    if sha(assessment_bytes) != args.assessment_sha256:
        fail("Cost assessment differs from reviewed identity")
    if len(report["tasks"]) != 10 or len(report["runs"]) != 20:
        fail("Unexpected experiment scope")
    if report.get("configuration_labels") != {"A": "Standard Codex", "B": "Memglide compact-prefix"}:
        fail("Unexpected treatment labels")
    if report_bytes != members["benchmarks/evidence/cost-ab-v1/formal/report.json"] or members["benchmarks/codex-cost-ab.md"] != members["benchmarks/evidence/cost-ab-v1/formal/report.md"]:
        fail("Primary report copies differ from public immutable evidence")
    for phase in ("before", "after"):
        identity = report["metadata"]["memglide_source_" + phase]
        if identity["git_head"] != FROZEN or identity["source_tree_sha256"] != APP_SHA:
            fail("Primary report source identity differs")
    provenance = assessment["provenance"]
    if provenance["primary_report_sha256"] != args.primary_sha256 or provenance["pricing_document_sha256"] != sha(members["benchmarks/cost-pricing.json"]):
        fail("Assessment does not bind the selected report and pricing document")
    if provenance["estimator_sha256"] != sha(members["accounting-source/benchmarks/estimate_cost.py.txt"]):
        fail("Estimator differs from reviewed post-freeze accounting source")
    if assessment["actual_billed_cost_usd"] is not None:
        fail("Unavailable billing was replaced")
    for row in assessment["runs"]:
        if row["cost"]["actual_billed_cost_usd"] is not None:
            fail("Unavailable attempt billing was replaced")
        native_path = "benchmarks/evidence/cost-ab-v1/formal/runs/" + safe_name(row["run_id"]) + "/events.jsonl"
        if row["events_sha256"] != sha(members[native_path]):
            fail("Cost row native-event identity differs")
    for arm in ("A", "B"):
        if assessment["configurations"][arm]["actual_billed_cost_usd"] is not None:
            fail("Unavailable arm billing was replaced")
    source_frozen = report["metadata"].get("source_unchanged_during_experiment") is True
    if not source_frozen or manifest["independent_audit"]["pass"] is not True:
        fail("Frozen-source independent audit did not pass")
    price_proof = manifest["price_audit"]
    matches = [row for row in manifest["files"] if row["source"] == price_proof["source"]]
    if len(matches) != 1:
        fail("Final independent price comparison is missing or duplicated")
    price_member = str(EVIDENCE.relative_to(ROOT)) + "/" + matches[0]["path"]
    price_bytes = members[price_member]
    price_audit = unique_json(price_bytes)
    price_verification = price_audit.get("assessment_verification", {})
    if (sha(price_bytes) != price_proof["sha256"] or price_audit.get("pass") is not True
            or price_audit.get("errors") or price_verification.get("pass") is not True
            or price_verification.get("assessment_sha256") != args.assessment_sha256
            or price_proof["assessment_sha256"] != args.assessment_sha256
            or price_audit["source_report_sha256"] != args.primary_sha256
            or price_audit["pricing_sha256"] != provenance["pricing_document_sha256"]):
        fail("Final independent price comparison differs from finalized selected bytes")
    costs = assessment["configurations"]
    rates = assessment["pricing"]["per_million_usd"]
    savings = assessment["cost_per_success_saving_percent"]
    sensitivity = assessment["write_uncertainty_saving_range"]
    focus = assessment["strata"]["selection_stratum"]["explicit_focus"]
    controls = assessment["strata"]["selection_stratum"]["caller_context_supplied"]
    focus_increase = -focus["native_reported_cost_per_success_saving_percent"]
    controls_saving = controls["native_reported_cost_per_success_saving_percent"]
    def number(value, places=6):
        return f"{value:.{places}f}" if isinstance(value, (float, int)) else "unavailable"
    start = f"""# Memglide ten-task cost study: report and evidence

The two exact-prompt bypass controls drive the aggregate {number(savings['native_reported_price_equivalent_usd'], 2)}% apparent saving. These measurements do not establish that Memglide reduces model cost.**

Every dollar figure is an **API-list-price equivalent**, not an actual invoice or subscription charge.

A is standard native Codex and B is production compact-prefix Memglide followed by the same native Codex. The study uses 10 fresh coding tasks (5 Python, 5 TypeScript), one balanced repeat, 20 original assignments, identical starting commits, prompts, model `gpt-6.1-sol`, effort and native options. Memglide adds full current focused source in 8 B tasks and forwards the exact common prompt for 2 supplied-context controls. There are no configured native MCP tools in either arm. The deterministic preparation uses no auxiliary model. This does not measure MCP retrieval, cross-session memory or large unfamiliar repositories.

Start with [the cost assessment](benchmarks/cost-assessment.md), [the raw primary report](benchmarks/codex-cost-ab.md), [pricing provenance](benchmarks/cost-pricing-research.md), and [the protocol](benchmarks/cost-protocol.md).

| Result | A: standard Codex | B: Memglide |
|---|---:|---:|
| Successful assignments | {costs['A']['successful_assignments']} | {costs['B']['successful_assignments']} |
| Attempted runs | {costs['A']['attempts']} | {costs['B']['attempts']} |
| Estimated USD, all attempts | {number(costs['A']['cost_statistics']['native_reported_price_equivalent_usd']['total'])} | {number(costs['B']['cost_statistics']['native_reported_price_equivalent_usd']['total'])} |
| Estimated USD per successful task | {number(costs['A']['cost_per_successful_task_usd']['native_reported_price_equivalent_usd'])} | {number(costs['B']['cost_per_successful_task_usd']['native_reported_price_equivalent_usd'])} |
| Eight source-prepared tasks: USD per success | {number(focus['configurations']['A']['cost_per_successful_task_usd']['native_reported_price_equivalent_usd'])} | {number(focus['configurations']['B']['cost_per_successful_task_usd']['native_reported_price_equivalent_usd'])} |
| Two exact-prompt controls: USD per success | {number(controls['configurations']['A']['cost_per_successful_task_usd']['native_reported_price_equivalent_usd'])} | {number(controls['configurations']['B']['cost_per_successful_task_usd']['native_reported_price_equivalent_usd'])} |

The source-prepared subset is **{number(focus_increase, 2)}% more expensive** in B. The two bypass controls are **{number(controls_saving, 2)}% cheaper** despite identical forwarded task/source bytes; this variation cannot be attributed to source preloading. No control or outlier is excluded.

Aggregate native-reported price-equivalent savings per successful task: **{number(savings['native_reported_price_equivalent_usd'], 2)}%**; positive means B costs less. Independent worst-case possible cache-write attribution gives a **{number(sensitivity['lower_saving_percent'], 2)}% to {number(sensitivity['upper_saving_percent'], 2)}%** savings scenario range. Never add cached input to total input or count reasoning twice. Native JSONL cache-write zero may reflect the upstream parser default rather than explicit provider zero, so both the native-reported point and write sensitivity are retained. Applicable tariff, realized tier, discounts and attributable charges remain unavailable.

The cost-per-success numerator includes every attempted original and supplemental run, including failures; the denominator is successful distinct assignments. Missing attempt usage keeps arm total/per-success unavailable rather than zero. Failed attempts retained: {len(assessment['failures'])}. The raw primary report keeps actual and computed cost fields null; the separately named assessment carries estimates and source hashes. Local measured CPU/RSS/time have no attributable dollar rate and are not fabricated into actual expenditure.

The ten tasks are small synthetic known-file work. There is one observation per arm per task; backend and operating-system caches are uncontrolled, with pair order balanced. The runtime audit passed 4,307 assertions. The final independent numeric comparison passed {price_verification['checked_assertions']} assertions and binds the final assessment, primary report and pricing hashes. An earlier price-auditor relative-path failure, its correction and pre-final draft comparisons remain preserved; no measured task was rerun for those tooling fixes.

All 20 original native JSONL streams, exact common prompts, executed commands, preparation reports, external graders, patches, checkpoints, durable attempt/result seals and independent audit are under `benchmarks/evidence/cost-ab-v1/`. Private native state, auth, databases, worktrees, fixture repositories and untracked-file contents are excluded. The common prompt is not B's full forwarded stdin; exact forwarded hashes, byte counts, full-source attestations and independent source reconstruction provide the delivery proof.

The frozen implementation and selected runtime benchmark, grader, tests, pricing and protocol Git blobs from `{FROZEN}` are reference text under `frozen-source/`. The post-freeze estimator and accounting regression tests are separately identified under `accounting-source/`; those files were not used by the measured runtime and are not labelled frozen Git blobs. Every earlier tracked benchmark and ZIP was compared byte-for-byte against the frozen commit and remains unchanged; historical ZIPs are not duplicated here.

    add("START-HERE.md", start, {"kind": "generated_final_guide"})
    builder_path = Path(__file__).resolve()
    add("reproduction/build_bundle.py.txt", read(builder_path), {"kind": "explicit_builder_source", "source": str(builder_path.relative_to(ROOT))})
    inventory = {
        "schema_version": 1, "scope": "Fresh ten-task standard Codex / compact-prefix Memglide API-price-equivalent cost comparison",
        "frozen_commit": FROZEN, "application_source_tree_sha256": APP_SHA,
        "published_evidence_manifest_sha256": args.manifest_sha256, "pinned_scanner_sha256": args.scanner_sha256,
        "self_excluded_from_members": "MEMBER-MANIFEST.json; external receipt records its hash",
        "synthetic_fixture_scan_exceptions": exceptions,
        "files": [{"path": name, "bytes": len(data), "sha256": sha(data), "origin": origins[name]} for name, data in sorted(members.items())],
    }
    inventory_bytes = encoded(inventory)
    add("MEMBER-MANIFEST.json", inventory_bytes, {"kind": "generated_member_inventory"})
    for path, identity in file_identities.items():

        current = path.read_bytes()
        if (len(current), sha(current)) != identity:

    if args.check_only:

    with zipfile.ZipFile(OUTPUT, "x", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for name, data in sorted(members.items()):

        "scope": "Fresh ten-task/20-original-run standard Codex versus compact-prefix Memglide; USD values are estimates",
        "frozen_commit": FROZEN, "application_source_tree_sha256": APP_SHA,
