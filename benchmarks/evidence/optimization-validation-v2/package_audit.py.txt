"""Audit built artifacts without executing their contents; hashes live in an external sidecar."""
from __future__ import annotations
import argparse, hashlib, json, os, stat, tarfile, zipfile
from pathlib import Path, PurePosixPath

BANNED_PARTS = {'.local', '.cache', '.codex', '.aws', '.git', '.venv', '.agents', '.runtime', '.memglide', '.pytest_tmp', 'node_modules', '__pycache__', '.pytest_cache', '.mypy_cache', '.ruff_cache', 'test-results', 'playwright-report'}
BANNED_NAMES = {'auth.json', 'credentials.json', 'installation_id', '.env'}
BANNED_SUFFIXES = {'.db', '.sqlite', '.sqlite3', '.pyc', '.pyo', '.pem', '.key'}
SKIP_SOURCE_DIRS = BANNED_PARTS | {'dist', 'build', '*.egg-info'}

def forbidden(name: str) -> bool:
    p=PurePosixPath(name)
    return p.is_absolute() or '..' in p.parts or bool(set(p.parts) & BANNED_PARTS) or p.name in BANNED_NAMES or p.name.startswith('.env.') or p.suffix in BANNED_SUFFIXES or p.name.endswith(('.db-wal','.db-shm','.sqlite-wal','.sqlite-shm'))

def sources(repository: Path) -> list[Path]:
    found=[]
    for root_name in ['src','tests','benchmarks','docs','scripts','frontend']:
        for current, directories, names in os.walk(repository/root_name):
            directories[:]=[name for name in directories if name not in SKIP_SOURCE_DIRS and not name.endswith('.egg-info')]
            for name in names:
                path=Path(current)/name
                if path.suffix not in {'.pyc','.pyo'}:found.append(path)
    found.extend(repository/name for name in ['pyproject.toml','uv.lock','README.md','LICENSE','AGENTS.md','.python-version'] if (repository/name).is_file())
    return sorted(set(found))

def audit(repository: Path, dist: Path, report_path: Path) -> dict:
    wheels=list(dist.glob('*.whl')); sdists=list(dist.glob('*.tar.gz'))
    assert len(wheels)==len(sdists)==1, 'Expected exactly one wheel and one sdist'
    failures=[]; zip_contents={};tar_contents={}
    with zipfile.ZipFile(wheels[0]) as archive:
        wheel_names=archive.namelist()
        for info in archive.infolist():
            if forbidden(info.filename) or stat.S_ISLNK(info.external_attr>>16):failures.append('Forbidden wheel entry: '+info.filename)
            if not info.is_dir():zip_contents[info.filename]=archive.read(info)
    with tarfile.open(sdists[0]) as archive:
        tar_names=[]
        for info in archive.getmembers():
            parts=PurePosixPath(info.name).parts
            relative=str(PurePosixPath(*parts[1:]))
            if relative=='.':continue
            tar_names.append(relative)
            if forbidden(relative) or info.issym() or info.islnk():failures.append('Forbidden sdist entry: '+relative)
            if info.isfile():
                stream=archive.extractfile(info);assert stream is not None
                tar_contents[relative]=stream.read()
    python_sources=list((repository/'src/memglide').rglob('*.py'))
    for path in python_sources:
        relative=str(path.relative_to(repository/'src'))
        if zip_contents.get(relative)!=path.read_bytes():failures.append('Wheel source mismatch: '+relative)
    expected=sources(repository)
    for path in expected:
        relative=str(path.relative_to(repository))
        if path.is_symlink():failures.append('Source symlink: '+relative)
        if tar_contents.get(relative)!=path.read_bytes():failures.append('sdist source mismatch: '+relative)
    assets=[path for path in (repository/'src/memglide/web').rglob('*') if path.is_file()]
    assert assets and any(path.suffix=='.js' for path in assets), 'Frontend bundle missing'
    for path in assets:
        relative=str(path.relative_to(repository/'src'))
        if zip_contents.get(relative)!=path.read_bytes():failures.append('Wheel frontend mismatch: '+relative)
        if tar_contents.get('src/'+relative)!=path.read_bytes():failures.append('sdist frontend mismatch: '+relative)
    assert 'memglide/web/index.html' in zip_contents
    assert any(path.startswith('memglide/web/assets/') and path.endswith('.js') for path in zip_contents)
    assert any(path.startswith('memglide/web/assets/') and path.endswith('.css') for path in zip_contents)
    for required in ['benchmarks/context_performance.py','benchmarks/refinement_eval.py','tests/test_context_focus.py','docs/mcp.md','frontend/package-lock.json','scripts/bundle_frontend.py','uv.lock']:
        if required not in tar_contents:failures.append('Required sdist entry absent: '+required)
    summary={
        'passed':not failures,
        'scope':'Wheel/sdist source-byte equality, compiled frontend, required files and exclusion audit. Archive hashes are external sidecars only; this public summary includes no archive hash or self-hash.',
        'wheel_member_count':len(wheel_names),'sdist_member_count':len(tar_names),
        'wheel_python_files_verified':len(python_sources),'sdist_source_files_verified':len(expected),'frontend_files_verified':len(assets),
        'forbidden_state_and_symlink_entries':len([x for x in failures if x.startswith(('Forbidden','Source symlink'))]),
        'required_groups':['src','tests','benchmarks','docs','scripts','frontend','uv.lock','pyproject.toml'],
        'failures':failures,
    }
    report_path.write_text(json.dumps(summary,indent=2)+'\n')
    (dist/'SHA256SUMS').write_text(''.join(hashlib.sha256(p.read_bytes()).hexdigest()+'  '+p.name+'\n' for p in [wheels[0],sdists[0]]))
    assert summary['passed'],json.dumps(failures)
    return summary

if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('--repository',type=Path,required=True);parser.add_argument('--dist',type=Path,required=True);parser.add_argument('--report',type=Path,required=True)
    args=parser.parse_args();print(json.dumps(audit(args.repository.resolve(),args.dist.resolve(),args.report.resolve()),indent=2))
