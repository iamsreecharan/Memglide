"""Smoke an installed wheel using isolated imports and a real loopback HTTP server."""
from __future__ import annotations
import argparse, hashlib, json, os, re, socket, subprocess, sys, tempfile, time, urllib.request
from pathlib import Path
import memglide


def main() -> None:
    parser=argparse.ArgumentParser();parser.add_argument('--state',type=Path,required=True);parser.add_argument('--report',type=Path,required=True);args=parser.parse_args()
    state=args.state.resolve();state.mkdir(parents=True,exist_ok=True)
    package=Path(memglide.__file__).resolve();assert package.is_relative_to(Path(sys.prefix).resolve()),str(package)
    assert sys.flags.isolated
    env={**os.environ,'MEMGLIDE_HOME':str(state)}
    database=state/'memglide.db'
    def cli(*arguments: str) -> dict|list:
        result=subprocess.run([sys.executable,'-I','-m','memglide.cli','--db',str(database),*arguments],cwd='/tmp',env=env,text=True,capture_output=True,timeout=60)
        assert result.returncode==0,result.stderr
        return json.loads(result.stdout)
    doctor=cli('doctor');assert doctor['frontend_built'] and doctor['fts5'] and doctor['database_integrity']=='ok'
    with tempfile.TemporaryDirectory(prefix='memglide-refined-wheel-',dir='/tmp') as directory:
        fixture=Path(directory);subprocess.run(['git','init','-q',str(fixture)],check=True)
        source=fixture/'calculations.py';source.write_text('def calculate_total(values):\n    return sum(values)\n')
        marker=state/'indexed-source-must-not-run'
        (fixture/'untrusted.py').write_text(f"from pathlib import Path\nPath({str(marker)!r}).write_text('executed')\nraise RuntimeError('Indexed source must not run')\n")
        project=cli('project','add',str(fixture));assert not marker.exists()
        search=cli('search',project['id'],'calculate_total');assert any(row['path']=='calculations.py' for row in search)
        session=cli('session','start',project['id'],'--task','Verify calculate_total source evidence')
        context=cli('context',session['id'],'calculate_total','--level','2','--budget','4000')
        digest=hashlib.sha256(source.read_bytes()).hexdigest()
        assert any(item['path']=='calculations.py' and item['hash']==digest and 'return sum(values)' in item['snippet'] for item in context['items'])
        actual_estimate=(len(json.dumps(context,ensure_ascii=False).encode())+2)//3
        assert actual_estimate==context['estimated_tokens']<=4000
        passport=cli('passport',session['id'],context['retrieval_id']);assert passport['verified_source_hashes']['calculations.py']==digest
        source.write_text(source.read_text()+'# controlled freshness check\n')
        stale=cli('passport',session['id'],context['retrieval_id']);assert stale['status']=='Stale' and 'calculations.py' in stale['stale_files']
        refreshed=cli('context',session['id'],'calculate_total','--level','2','--budget','4000')
        assert refreshed['passport']['verified_source_hashes']['calculations.py']==hashlib.sha256(source.read_bytes()).hexdigest()
        with socket.socket() as available:available.bind(('127.0.0.1',0));port=available.getsockname()[1]
        origin=f'http://127.0.0.1:{port}'
        log=(state/'server.stderr.log').open('w')
        server=subprocess.Popen([sys.executable,'-I','-m','memglide.cli','--db',str(database),'serve','--port',str(port)],cwd='/tmp',env=env,stdout=subprocess.DEVNULL,stderr=log)
        opener=urllib.request.build_opener(urllib.request.ProxyHandler({}))
        def get(path: str):return opener.open(origin+path,timeout=3)
        try:
            deadline=time.monotonic()+20
            while True:
                assert server.poll() is None,'Installed wheel HTTP server exited'
                try:
                    with get('/api/health') as response:health=json.load(response)
                    break
                except OSError:
                    if time.monotonic()>deadline:raise
                    time.sleep(.1)
            assert health['status']=='ok' and health['local_only']
            with get('/') as response:html=response.read().decode()
            assets=re.findall(r'(?:src|href)="(/assets/[^\"]+)"',html);assert any(asset.endswith('.js') for asset in assets)
            fetched=[]
            for asset in assets:
                with get(asset) as response:
                    data=response.read();assert data and response.status==200 and 'text/html' not in response.headers.get('Content-Type','')
                    fetched.append({'path':asset,'bytes':len(data)})
            with get('/context') as response:assert response.read().decode()==html
            with get('/api/projects') as response:assert any(row['id']==project['id'] for row in json.load(response))
            assert not marker.exists()
            summary={'passed':True,'package_location':str(package),'isolated_imports':True,'cwd':'/tmp','doctor':'passed','indexed_files':project.get('index_stats',{}).get('file_count'),'search_hits':len(search),'context_items':len(context['items']),'estimated_planning_tokens':context['estimated_tokens'],'budget':4000,'source_hashes_match':True,'old_passport_detected_edit':True,'refreshed_context_hash_matches':True,'indexed_source_not_executed':True,'loopback_api_health':True,'bundled_spa_route':True,'production_assets':fetched,'scope':'Clean installed wheel with pinned locked dependencies; isolated CLI imports, real index/context/passport and loopback HTTP/compiled assets. No model requests or token/cost measurements.'}
            args.report.write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary,indent=2))
        finally:
            server.terminate()
            try:server.wait(timeout=10)
            except subprocess.TimeoutExpired:server.kill();server.wait(timeout=5)
            log.close()

if __name__=='__main__':main()
